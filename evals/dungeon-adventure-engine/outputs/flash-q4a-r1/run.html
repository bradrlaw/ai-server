"""
Text-Based Adventure Game Engine
=================================
A single-file, zero-dependency adventure game engine with a small dungeon world.
Only the Python standard library is used.
"""

from __future__ import annotations

import sys
from dataclasses import dataclass, field
from typing import Callable, Dict, List, Optional


# ---------------------------------------------------------------------------
# Data Classes
# ---------------------------------------------------------------------------

@dataclass
class Item:
    """Represents a collectible item in the game world.

    Attributes:
        item_id: Unique identifier for the item.
        name: Display name shown to the player.
        description: Flavor text printed when the player examines the item.
        on_use: Optional callback invoked when the player uses the item.
                Receives the current GameState and returns a message string.
    """

    item_id: str
    name: str
    description: str
    on_use: Optional[Callable[["GameState"], str]] = None


@dataclass
class Room:
    """Represents a location in the game world.

    Attributes:
        room_id: Unique identifier for the room.
        name: Display name of the room.
        description: Base description of the room.
        exits: Mapping of direction -> target room_id.
        items: List of item_ids present in this room.
        locked_exits: Mapping of direction -> required item_id to unlock.
        dynamic_descriptions: Mapping of flag_name -> alternate description
                              that replaces the base description when the flag is set.
    """

    room_id: str
    name: str
    description: str
    exits: Dict[str, str] = field(default_factory=dict)
    items: List[str] = field(default_factory=list)
    locked_exits: Dict[str, str] = field(default_factory=dict)
    dynamic_descriptions: Dict[str, str] = field(default_factory=dict)


# ---------------------------------------------------------------------------
# Game State
# ---------------------------------------------------------------------------

class GameState:
    """Tracks all mutable state for a single playthrough.

    Attributes:
        current_room: The room_id where the player currently stands.
        inventory: List of item_ids the player is carrying.
        visited_rooms: Set of room_ids the player has entered.
        flags: Arbitrary boolean flags for tracking events (e.g., lever pulled).
        history: Chronological log of all messages/events.
        won: Whether the secret ending has been achieved.
    """

    def __init__(self, starting_room: str) -> None:
        """Initialize the game state at the given starting room."""
        self.current_room: str = starting_room
        self.inventory: List[str] = []
        self.visited_rooms: set[str] = {starting_room}
        self.flags: Dict[str, bool] = {}
        self.history: List[str] = []
        self.won: bool = False

    def set_flag(self, flag: str) -> None:
        """Set a boolean flag to True."""
        self.flags[flag] = True

    def has_flag(self, flag: str) -> bool:
        """Return True if the flag has been set."""
        return self.flags.get(flag, False)

    def add_to_inventory(self, item_id: str) -> None:
        """Add an item to the player's inventory."""
        if item_id not in self.inventory:
            self.inventory.append(item_id)

    def remove_from_inventory(self, item_id: str) -> None:
        """Remove an item from the player's inventory."""
        if item_id in self.inventory:
            self.inventory.remove(item_id)

    def has_item(self, item_id: str) -> bool:
        """Return True if the player is carrying the given item."""
        return item_id in self.inventory

    def log(self, message: str) -> None:
        """Append a message to the game history."""
        self.history.append(message)


# ---------------------------------------------------------------------------
# Game Engine
# ---------------------------------------------------------------------------

class GameEngine:
    """Core engine that parses input, routes commands, and drives the game loop.

    Args:
        rooms: Mapping of room_id -> Room.
        items: Mapping of item_id -> Item.
        starting_room: The room_id where the game begins.
    """

    # Valid movement directions.
    DIRECTIONS: List[str] = ["north", "south", "east", "west"]

    # Primary commands the engine recognizes.
    COMMANDS: List[str] = [
        "look", "go", "move", "take", "drop", "inventory",
        "use", "help", "history", "quit",
    ]

    # Synonym map: input token -> canonical command.
    SYNONYMS: Dict[str, str] = {
        "examine": "look at",
        "inspect": "look at",
        "x": "look at",
        "grab": "take",
        "get": "take",
        "pick": "take",
        "pickup": "take",
        "i": "inventory",
        "inv": "inventory",
        "n": "north",
        "s": "south",
        "e": "east",
        "w": "west",
        "q": "quit",
        "exit": "quit",
        "leave": "quit",
        "l": "look",
        "h": "help",
        "?": "help",
        "hist": "history",
        "backpack": "inventory",
    }

    def __init__(
        self,
        rooms: Dict[str, Room],
        items: Dict[str, Item],
        starting_room: str,
    ) -> None:
        self.rooms: Dict[str, Room] = rooms
        self.items: Dict[str, Item] = items
        self.state: GameState = GameState(starting_room)
        self._separator: str = "─" * 56

    # ------------------------------------------------------------------
    # Public entry point
    # ------------------------------------------------------------------

    def run(self) -> None:
        """Start the interactive game loop. Exits on 'quit', EOF, or Ctrl+C."""
        self._print_banner()
        # Show the starting room.
        self._handle_look()

        while True:
            try:
                raw: str = input("\n> ").strip()
            except (EOFError, KeyboardInterrupt):
                self._print_goodbye()
                return

            if not raw:
                continue

            output: str = self._parse_and_execute(raw)
            print(self._separator)
            print(output)

            if self.state.won:
                self._print_victory()
                return

    # ------------------------------------------------------------------
    # Input parsing
    # ------------------------------------------------------------------

    def _parse_and_execute(self, raw: str) -> str:
        """Parse raw input and dispatch to the appropriate command handler."""
        self.state.log(f"> {raw}")

        tokens: List[str] = raw.lower().split()
        first: str = tokens[0]

        # Resolve synonyms.
        first = self._resolve_synonym(first)

        # Handle multi-word synonyms like "look at".
        if first == "look at" and len(tokens) > 1:
            return self._handle_look_at(" ".join(tokens[1:]))

        # Bare direction (already resolved to full word by synonym map).
        if first in self.DIRECTIONS:
            return self._handle_go(first)

        # Dispatch by command.
        handler_map: Dict[str, Callable[[List[str]], str]] = {
            "look": self._handle_look,
            "go": self._handle_go,
            "move": self._handle_go,
            "take": self._handle_take,
            "drop": self._handle_drop,
            "inventory": self._handle_inventory,
            "use": self._handle_use,
            "help": self._handle_help,
            "history": self._handle_history,
            "quit": self._handle_quit,
        }

        # Attempt partial match on the command.
        matched_cmd: Optional[str] = self._partial_match(first, list(handler_map.keys()))

        if matched_cmd is None:
            # Could be a bare direction typed in full.
            matched_dir: Optional[str] = self._partial_match(first, self.DIRECTIONS)
            if matched_dir:
                return self._handle_go(matched_dir)
            return self._error(f"I don't understand '{raw}'. Type 'help' for commands.")

        handler = handler_map[matched_cmd]
        args: List[str] = tokens[1:]

        # Special case: "look at <item>" when typed as separate tokens.
        if matched_cmd == "look" and args and args[0] == "at":
            return self._handle_look_at(" ".join(args[1:]))

        return handler(args)

    def _resolve_synonym(self, token: str) -> str:
        """Map a synonym token to its canonical form."""
        return self.SYNONYMS.get(token, token)

    def _partial_match(self, prefix: str, candidates: List[str]) -> Optional[str]:
        """Return the candidate that starts with *prefix*, or None."""
        matches: List[str] = [c for c in candidates if c.startswith(prefix)]
        if len(matches) == 1:
            return matches[0]
        return None

    # ------------------------------------------------------------------
    # Command handlers
    # ------------------------------------------------------------------

    def _handle_look(self, args: Optional[List[str]] = None) -> str:
        """Handle the 'look' command: describe the current room and visible items."""
        room: Room = self.rooms[self.state.current_room]
        desc: str = self._get_room_description(room)
        lines: List[str] = [f"[ {room.name.upper()} ]", desc]

        visible_items: List[str] = [
            self.items[iid].name
            for iid in room.items
            if iid not in self.state.inventory
        ]
        if visible_items:
            lines.append(f"\nYou notice: {', '.join(visible_items)}.")
        else:
            lines.append("\nYou see nothing of interest here.")

        exits_str: str = ", ".join(room.exits.keys()) if room.exits else "none"
        lines.append(f"Exits: {exits_str}")

        result: str = "\n".join(lines)
        self.state.log(result)
        return result

    def _handle_look_at(self, item_name: str) -> str:
        """Handle 'look at <item>': print the item's description."""
        item: Optional[Item] = self._find_item(item_name)
        if item is None:
            return self._error(f"You don't see any '{item_name}' here.")
        result: str = f"{item.name}: {item.description}"
        self.state.log(result)
        return result

    def _handle_go(self, args: List[str]) -> str:
        """Handle movement commands: go/move <direction> or bare direction."""
        # The first token might already be the direction.
        direction: str = args[0] if args else ""
        direction = self._resolve_synonym(direction)
        direction = self._partial_match(direction, self.DIRECTIONS) or direction

        if direction not in self.DIRECTIONS:
            return self._error(f"'{direction}' is not a valid direction. Try north/south/east/west.")

        room: Room = self.rooms[self.state.current_room]

        # Check if the direction exists as an exit.
        if direction not in room.exits:
            return self._error(f"You can't go {direction} from here.")

        target_room_id: str = room.exits[direction]

        # Check for locked doors.
        if direction in room.locked_exits:
            required_item_id: str = room.locked_exits[direction]
            if not self.state.has_item(required_item_id):
                req_name: str = self.items[required_item_id].name
                return self._error(
                    f"The way {direction} is locked. You need a {req_name} to proceed."
                )
            else:
                # Auto-use the key.
                self.state.log(f"You unlock the door {direction} with the {self.items[required_item_id].name}.")

        # Move the player.
        self.state.current_room = target_room_id
        self.state.visited_rooms.add(target_room_id)

        # Trigger any on-enter effects.
        enter_msg: str = self._on_enter_room(target_room_id)

        # Describe the new room.
        look_output: str = self._handle_look()
        if enter_msg:
            look_output = enter_msg + "\n" + look_output

        return look_output

    def _handle_take(self, args: List[str]) -> str:
        """Handle 'take <item>': pick up an item from the current room."""
        if not args:
            return self._error("Take what? Specify an item name.")
        item_name: str = " ".join(args)
        item: Optional[Item] = self._find_item_in_room(item_name)
        if item is None:
            return self._error(f"There is no '{item_name}' here to take.")
        if self.state.has_item(item.item_id):
            return self._error(f"You already have the {item.name}.")

        self.state.add_to_inventory(item.item_id)
        self.state.set_flag(f"{item.item_id}_taken")
        result: str = f"You take the {item.name}."
        self.state.log(result)
        return result

    def _handle_drop(self, args: List[str]) -> str:
        """Handle 'drop <item>': place an item from inventory into the room."""
        if not args:
            return self._error("Drop what? Specify an item name.")
        item_name: str = " ".join(args)
        item: Optional[Item] = self._find_item_in_inventory(item_name)
        if item is None:
            return self._error(f"You're not carrying any '{item_name}'.")

        self.state.remove_from_inventory(item.item_id)
        room: Room = self.rooms[self.state.current_room]
        if item.item_id not in room.items:
            room.items.append(item.item_id)
        result: str = f"You drop the {item.name}."
        self.state.log(result)
        return result

    def _handle_inventory(self, args: Optional[List[str]] = None) -> str:
        """Handle 'inventory' / 'i': list carried items."""
        if not self.state.inventory:
            result: str = "Your inventory is empty."
        else:
            names: List[str] = [self.items[iid].name for iid in self.state.inventory]
            result = "You are carrying: " + ", ".join(names) + "."
        self.state.log(result)
        return result

    def _handle_use(self, args: List[str]) -> str:
        """Handle 'use <item>': activate the item's on_use callback."""
        if not args:
            return self._error("Use what? Specify an item name.")
        item_name: str = " ".join(args)
        item: Optional[Item] = self._find_item_in_inventory(item_name)
        if item is None:
            return self._error(f"You're not carrying any '{item_name}'.")
        if item.on_use is None:
            result: str = f"You can't do anything useful with the {item.name}."
            self.state.log(result)
            return result

        result = item.on_use(self.state)
        self.state.log(result)
        return result

    def _handle_help(self, args: Optional[List[str]] = None) -> str:
        """Handle 'help': print all available commands."""
        lines: List[str] = [
            "AVAILABLE COMMANDS:",
            "  look              - Describe the current room",
            "  look at <item>    - Examine an item",
            "  go <direction>    - Move (north/south/east/west)",
            "  move <direction>  - Same as go",
            "  <direction>       - Shorthand: n, s, e, w",
            "  take <item>       - Pick up an item",
            "  drop <item>       - Put an item down",
            "  inventory / i     - List your items",
            "  use <item>        - Use/activate an item",
            "  history           - Show last 10 events",
            "  help              - Show this message",
            "  quit              - Exit the game",
        ]
        result: str = "\n".join(lines)
        self.state.log(result)
        return result

    def _handle_history(self, args: Optional[List[str]] = None) -> str:
        """Handle 'history': print the last 10 log entries."""
        recent: List[str] = self.state.history[-10:]
        if not recent:
            return "No history yet."
        lines: List[str] = ["--- Last 10 Events ---"]
        for entry in recent:
            # Truncate long entries for readability.
            display: str = entry if len(entry) <= 80 else entry[:77] + "..."
            lines.append(f"  • {display}")
        result: str = "\n".join(lines)
        self.state.log(result)
        return result

    def _handle_quit(self, args: Optional[List[str]] = None) -> str:
        """Handle 'quit': exit the game gracefully."""
        result: str = "Thanks for playing! Farewell, adventurer."
        print(self._separator)
        print(result)
        self._print_goodbye()
        sys.exit(0)

    # ------------------------------------------------------------------
    # Helper / utility methods
    # ------------------------------------------------------------------

    def _get_room_description(self, room: Room) -> str:
        """Return the room description, checking dynamic overrides."""
        # Check dynamic descriptions keyed by flag.
        for flag, alt_desc in room.dynamic_descriptions.items():
            if self.state.has_flag(flag):
                return alt_desc
        # Special: Armory changes if torch was taken.
        if room.room_id == "armory" and self.state.has_flag("torch_taken"):
            return room.dynamic_descriptions.get("torch_taken", room.description)
        return room.description

    def _find_item_in_room(self, name: str) -> Optional[Item]:
        """Find an item by fuzzy name match among items in the current room."""
        room: Room = self.rooms[self.state.current_room]
        for iid in room.items:
            if iid not in self.state.inventory and self._name_matches(name, iid):
                return self.items[iid]
        return None

    def _find_item_in_inventory(self, name: str) -> Optional[Item]:
        """Find an item by fuzzy name match in the player's inventory."""
        for iid in self.state.inventory:
            if self._name_matches(name, iid):
                return self.items[iid]
        return None

    def _find_item(self, name: str) -> Optional[Item]:
        """Find an item in the room or inventory by fuzzy name match."""
        return self._find_item_in_room(name) or self._find_item_in_inventory(name)

    def _name_matches(self, query: str, item_id: str) -> bool:
        """Check if *query* partially matches the item_id or item name."""
        item: Item = self.items[item_id]
        q: str = query.lower().strip()
        return q in item_id.lower() or q in item.name.lower()

    def _on_enter_room(self, room_id: str) -> str:
        """Trigger special messages when entering certain rooms."""
        if room_id == "garden":
            return "🌿 A gentle breeze carries the scent of roses.\n   A stone tablet reads: \"You've found peace.\""
        return ""

    def _error(self, message: str) -> str:
        """Format an error message and log it."""
        result: str = f"⚠  {message}"
        self.state.log(result)
        return result

    def _print_banner(self) -> None:
        """Print the styled ASCII banner at game start."""
        banner: str = (
            "\n"
            "  ╔══════════════════════════════════════════════════════╗\n"
            "  ║           ⚔  DUNGEON OF SHADOWS  ⚔              ║\n"
            "  ║        A Text Adventure — v1.0                  ║\n"
            "  ╠══════════════════════════════════════════════════════╣\n"
            "  ║  Explore rooms, solve puzzles, find treasure.    ║\n"
            "  ║  Type 'help' for a list of commands.            ║\n"
            "  ║  Type 'quit' to exit.                           ║\n"
            "  ╚══════════════════════════════════════════════════════╝\n"
        )
        print(banner)

    def _print_goodbye(self) -> None:
        """Print a farewell message."""
        print("\n  Until next time, adventurer. Goodbye! 👋\n")

    def _print_victory(self) -> None:
        """Print the victory banner when the secret ending is achieved."""
        victory: str = (
            "\n"
            "  ╔══════════════════════════════════════════════════════╗\n"
            "  ║                                                  ║\n"
            "  ║   ✦  YOU HAVE UNCOVERED THE SECRET ENDING!  ✦   ║\n"
            "  ║                                                  ║\n"
            "  ║   The ancient words echo through eternity...    ║\n"
            "  ║   You are the true heir of the Shadow Throne.  ║\n"
            "  ║                                                  ║\n"
            "  ║   ★  C O N G R A T U L A T I O N S  ★         ║\n"
            "  ║                                                  ║\n"
            "  ╚══════════════════════════════════════════════════════╝\n"
        )
        print(victory)
