/* ============================================================
 * Item
 * ============================================================ */
class Item {
  constructor(name, description, opts = {}) {
    this.name = name;
    this.description = description;
    this.readText = opts.readText || null;
    this.useText = opts.useText || null;
    this.onUse = opts.onUse || null;      // function(game) called when used
    this.usable = opts.usable || false;
    this.readable = opts.readable || false;
    this.icon = opts.icon || 'item';
  }
}

/* ============================================================ */
class Room {
  constructor(name, description, exits = {}, items = [], onEnter = null) {
    this.name = name;
    this.description = description;
    this.exits = exits;           // direction -> room id
    this.items = items;           // Item instances (removed when taken)
    this.onEnter = onEnter;       // optional function(game)
    this.hidden = false;
  }
}

  reset() {

  get room() { return this.rooms[this.currentRoomId]; }

  /* ---------- Typewriter console ---------- */
  print(text, cls = '') {
    if (this.typing) {
      this.typeQueue.push({ text, cls });
      return;
    }
    this.startTyping(text, cls);
  }

  startTyping(text, cls) {
    const line = document.createElement('div');
    line.className = 'line ' + cls;
    this.outputEl.appendChild(line);
    this.typing = true;
    let i = 0;
    const tick = () => {
      // if skipped externally, i === text.length
      if (i < text.length) {
        const chunk = Math.min(2, text.length - i);
        line.textContent += text.slice(i, i + chunk);
        i += chunk;
        this.autoScroll();
        this.typeTimer = setTimeout(tick, 14);
      } else {
        this.typing = false;
        this.typeTimer = null;
        if (this.typeQueue.length) {
          const next = this.typeQueue.shift();
          this.startTyping(next.text, next.cls);
        }
        if (this.typeDone) { const cb = this.typeDone; this.typeDone = null; cb(); }
        this.autoScroll();
      }
    };
    tick();
  }

  skipTyping() {
    if (!this.typing) return;
    clearTimeout(this.typeTimer);
    // finish current line instantly
    const lines = this.outputEl.querySelectorAll('.line');
    const last = lines[lines.length - 1];
    // We need the full text of the current line:
    if (this._currentFullText != null) {
      last.textContent = this._currentFullText;
    }
    this.typing = false;
    this.typeTimer = null;
    if (this.typeQueue.length) {
      const next = this.typeQueue.shift();
      this.startTyping(next.text, next.cls);
    }
    this.autoScroll();
  }

  // hook: track current full text
  _origStartTyping = null;

  autoScroll() {
    this.outputEl.scrollTop = this.outputEl.scrollHeight;
  }

  /* ---------- UI ---------- */
  bindUI() {
    // track current full text for skip
    const origStart = this.startTyping.bind(this);
    this.startTyping = (text, cls) => {
      this._currentFullText = text;
      origStart(text, cls);
    };

    this.inputEl.addEventListener('keydown', (e) => {
      if (e.key === 'Enter') {
        const raw = this.inputEl.value.trim();
        this.inputEl.value = '';
        if (raw) {
          this.history.push(raw);
          this.historyIndex = this.history.length;
          this.handleCommand(raw);
        }
      } else if (e.key === 'ArrowUp') {
        e.preventDefault();
        if (this.history.length === 0) return;
        this.historyIndex = Math.max(0, this.historyIndex - 1);
        this.inputEl.value = this.history[this.historyIndex] || '';
      } else if (e.key === 'ArrowDown') {
        e.preventDefault();
        if (this.history.length === 0) return;
        if (this.historyIndex < this.history.length - 1) {
          this.historyIndex++;
          this.inputEl.value = this.history[this.historyIndex];
        } else {
          this.historyIndex = this.history.length;
          this.inputEl.value = '';
        }
      }
    });

    document.getElementById('console-wrap').addEventListener('click', (e) => {
      if (e.target === this.outputEl || e.target.id === 'console-output' ||
          e.target.classList.contains('line') || e.target.closest('#console-output')) {
        this.skipTyping();
      }
      this.inputEl.focus();
    });

    document.getElementById('reset-btn').addEventListener('click', () => {
      this.reset();
      this.inputEl.focus();
    });

    const howto = document.getElementById('howto');
    document.getElementById('howto-toggle').addEventListener('click', () => {
      const open = howto.classList.toggle('open');
      document.getElementById('howto-toggle').setAttribute('aria-expanded', open);
    });

    this.inputEl.focus();
  }

  updateUI() {
    document.getElementById('stat-moves').textContent = this.moves;
    document.getElementById('stat-items').textContent = this.itemsFound;

    const list = document.getElementById('inventory-list');
    list.innerHTML = '';
    if (this.inventory.length === 0) {
      const li = document.createElement('li');
      li.className = 'inv-empty';
      li.textContent = 'Inventory is empty.';
      list.appendChild(li);
      return;
    }
    for (const item of this.inventory) {
      const li = document.createElement('li');
      const svg = document.createElementNS('http://www.w3.org/2000/svg', 'svg');
      svg.setAttribute('viewBox', '0 0 24 24');
      svg.setAttribute('fill', 'none');
      svg.setAttribute('stroke', 'currentColor');
      svg.setAttribute('stroke-width', '2');
      svg.setAttribute('stroke-linecap', 'round');
      svg.setAttribute('stroke-linejoin', 'round');
      const paths = {
        key: '<circle cx="7.5" cy="15.5" r="5.5"/><line x1="11.5" y1="11.5" x2="21" y2="2"/><line x1="16" y1="7" x2="19" y2="10"/><line x1="13" y1="10" x2="15.5" y2="12.5"/>',
        coin: '<circle cx="12" cy="12" r="9"/><path d="M12 7v10M9 9.5h4.5a2 2 0 1 1 0 4H9.5a2 2 0 1 0 0 4H15"/>',
        item: '<path d="M21 8l-9-5-9 5v8l9 5 9-5V8z"/><polyline points="3 8 12 13 21 8"/>'
      };
      svg.innerHTML = paths[item.icon] || paths.item;
      li.appendChild(svg);
      const span = document.createElement('span');
      span.textContent = item.name;
      li.appendChild(span);
      list.appendChild(li);
    }
  }

  /* ---------- Commands ---------- */
  handleCommand(raw) {
    const echo = document.createElement('div');
    echo.className = 'line cmd';
    echo.innerHTML = '<span class="prompt">&gt;</span> ' + this.escapeHtml(raw);
    this.outputEl.appendChild(echo);
    this.autoScroll();

    const result = this.parseCommand(raw.toLowerCase());
    if (typeof result === 'string') {
      result ? this.print(result) : null;
    }
    this.updateUI();
  }

  escapeHtml(s) {
    return s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
  }

  parseCommand(lower) {
    const tokens = lower.replace(/[.,!?]/g, '').split(/\s+/).filter(Boolean);
    if (tokens.length === 0) return null;

    const verb = tokens[0];
    const rest = tokens.slice(1).join(' ');

    // verb synonyms
    const v = this.normalizeVerb(verb);

    switch (v) {
      case 'look':
        return this.cmdLook();
      case 'go':
      case 'move': {
        const dir = this.resolveDirection(rest);
        if (!dir) return 'Which way? Try "go north", "go east", "go south", or "go west".';
        return this.cmdGo(dir);
      }
      case 'take':
        return this.cmdTake(rest);
      case 'drop':
        return this.cmdDrop(rest);
      case 'inventory':
        return this.cmdInventory();
      case 'use':
        return this.cmdUse(rest);
      case 'read':
        return this.cmdRead(rest);
      case 'unlock':
        return this.cmdUnlock(rest);
      case 'open':
        return this.cmdUnlock(rest);
      case 'help':
        return this.cmdHelp();
      case 'wait':
        return 'You wait. The dungeon does not. A moth circles your lantern, unbothered.';
      case 'quit':
        return 'There is no quitting a local dungeon. The hatch is very, very far up.';
      default:
        return 'I don\'t understand "' + verb + '". Type "help" to see what I can do.';
    }
  }

  normalizeVerb(w) {
    const map = {
      'look': 'look', 'examine': 'look', 'inspect': 'look', 'l': 'look',
      'lookat': 'look',
      'go': 'go', 'move': 'go', 'walk': 'go', 'north': 'go', 'south': 'go',
      'east': 'go', 'west': 'go', 'n': 'go', 's': 'go', 'e': 'go', 'w': 'go',
      'take': 'take', 'grab': 'take', 'get': 'take', 'pick': 'take', 'loot': 'take', 't': 'take',
      'drop': 'drop', 'put': 'drop', 'd': 'drop',
      'inventory': 'inventory', 'inv': 'inventory', 'i': 'inventory', 'items': 'inventory', 'carry': 'inventory',
      'use': 'use', 'u': 'use', 'pull': 'use', 'activate': 'use',
      'read': 'read', 'r': 'read', 'study': 'read',
      'unlock': 'unlock', 'open': 'open',
      'help': 'help', 'h': 'help', '?': 'help', 'commands': 'help'
    };
    // "look at x" -> look
    if (w === 'at') return 'look';
    return map[w] || w;
  }

  resolveDirection(s) {
    if (!s) return null;
    const dirs = {
      'north': 'north', 'n': 'north',
      'south': 'south', 's': 'south',
      'east': 'east', 'e': 'east',
      'west': 'west', 'w': 'west'
    };
    // allow "go the north door"
    const first = s.split(/\s+/)[0];
    return dirs[first] || dirs[s] || null;
  }

  findItemInRoom(query) {
    const q = this.normalizeQuery(query);
    return this.room.items.find(it =>
      this.normalizeQuery(it.name) === q ||
      this.normalizeQuery(it.name).includes(q) ||
      q.includes(this.normalizeQuery(it.name))
    ) || null;
  }

  findItemInInventory(query) {
    const q = this.normalizeQuery(query);
    return this.inventory.find(it =>
      this.normalizeQuery(it.name) === q ||
      this.normalizeQuery(it.name).includes(q) ||
      q.includes(this.normalizeQuery(it.name))
    ) || null;
  }

  normalizeQuery(s) {
    return s.toLowerCase().replace(/^(the|a|an)\s+/, '').trim();
  }

  /* ---------- Individual commands ---------- */
  describeRoom(full) {
    this.print('— ' + this.room.name + ' —', 'sys');
    this.room.items.forEach(it => {
      this.print('You can see a ' + it.name + ' here.', 'dim');
    });
    if (full) this.print(this.room.description);
  }

  cmdLook() {
    this.describeRoom(true);
    return null;
  }

  cmdGo(dir) {
    // special: locked armory north door
    if (this.room.id === undefined && this.currentRoomId === 'armory' && dir === 'north') {
      if (this.flags.armory_unlocked) {
        this.moveTo('treasure');
        return null;
      }
      return 'The north door is locked tight. The keyhole is small — a small key might fit.';
    }
    if (this.currentRoomId === 'armory' && dir === 'north') {
      if (this.flags.armory_unlocked) {
        this.moveTo('treasure');
        return null;
      }
      return 'The north door is locked tight. The keyhole is small — a small key might fit. (Try "unlock door" or "use rusted key".)';
    }
    const target = this.room.exits[dir];
    if (!target) {
      if (dir === 'west' && this.currentRoomId === 'library' && !this.flags.lever_pulled) {
        return 'The west wall is solid stone. Something behind it sounds like it wants to move, but refuses.';
      }
      return 'You cannot go ' + dir + ' from here. There is no passage that way.';
    }
    this.moveTo(target);
    return null;
  }

  moveTo(id) {
    this.currentRoomId = id;
    this.moves++;
    this.visited.add(id);
    if (this.room.onEnter) this.room.onEnter(this);
    this.describeRoom(true);
  }

  cmdTake(query) {
    if (!query) return 'Take what? Name an item, e.g. "take torch".';
    const item = this.findItemInRoom(query);
    if (!item) {
      if (this.findItemInInventory(query)) return 'You are already carrying the ' + query + '.';
      return 'There is no "' + query + '" here to take.';
    }
    const idx = this.room.items.indexOf(item);
    this.room.items.splice(idx, 1);
    this.inventory.push(item);
    this.itemsFound++;
    this.print('You pick up the ' + item.name + '.');
    this.print(item.description, 'dim');
    return null;
  }

  cmdDrop(query) {
    if (!query) return 'Drop what? Name an item.';
    const item = this.findItemInInventory(query);
    if (!item) return 'You are not carrying a "' + query + '".';
    this.inventory.splice(this.inventory.indexOf(item), 1);
    this.room.items.push(item);
    this.print('You set the ' + item.name + ' down on the stone floor.');
    return null;
  }

  cmdInventory() {
    if (this.inventory.length === 0) {
      this.print('Your pockets are empty. Just lint and a small sense of dread.');
    } else {
      this.print('You are carrying:');
      this.inventory.forEach(it => this.print(' • ' + it.name, 'dim'));
    }
    return null;
  }

  cmdUse(query) {
    if (!query) return 'Use what? Name an item, e.g. "use lever".';
    const item = this.findItemInInventory(query);
    if (!item) return 'You are not carrying a "' + query + '".';
    if (item.onUse) {
      const out = item.onUse(this);
      if (out) this.print(out);
      return null;
    }
    if (item.useText) {
      this.print(item.useText);
      return null;
    }
    return 'You fiddle with the ' + item.name + ' for a moment, but it does nothing useful here.';
  }

  cmdRead(query) {
    if (!query) return 'Read what? Name an item.';
    const item = this.findItemInInventory(query);
    if (!item) return 'You are not carrying a "' + query + '".';
    if (!item.readText) return 'The ' + item.name + ' has nothing written on it.';
    this.print(item.readText);
    return null;
  }

  cmdUnlock(query) {
    // only meaningful for the armory north door
    if (this.currentRoomId !== 'armory') {
      return 'There is nothing locked in this room.';
    }
    if (this.flags.armory_unlocked) {
      return 'The north door is already unlocked. Just "go north".';
    }
    const key = this.findItemInInventory('key');
    if (!key) {
      return 'You have no key. The lock is small and stubborn. (The entrance hall had a pedestal...)';
    }
    this.flags.armory_unlocked = true;
    this.rooms.armory.exits.north = 'treasure';
    this.print('The rusted key turns with a scrape of metal. The north door swings open into a warm, golden light.');
    return null;
  }

  cmdHelp() {
    this.print('Available commands:', 'sys');
    this.print('  look / examine        describe the current room', 'dim');
    this.print('  go <direction>        move (n, s, e, w, north, etc.)', 'dim');
    this.print('  take / grab <item>    pick up an item', 'dim');
    this.print('  drop <item>           put an item down', 'dim');
    this.print('  inventory / inv       list what you carry', 'dim');
    this.print('  use <item>            use an item (e.g. "use lever")', 'dim');
    this.print('  read <item>           read a readable item', 'dim');
    this.print('  unlock door           unlock a locked door with a key', 'dim');
    this.print('  help                  show this list', 'dim');
    this.print('Tip: ↑ and ↓ recall previous commands. Click the console to skip typing.', 'sys');
    return null;
  }
}

/* Room id accessor (rooms stored by id) */
Object.defineProperty(Room.prototype, 'id', {
  get() {
    const engine = window.__game;
    if (!engine) return null;
    for (const key in engine.rooms) {
      if (engine.rooms[key] === this) return key;
    }
    return null;
  }
});

window.__game = new GameEngine();
