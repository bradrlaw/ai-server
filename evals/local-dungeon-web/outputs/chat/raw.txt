    bindEvents() {
      this.inputEl.addEventListener('keydown', (e) => {
        if (e.key === 'Enter') {
          e.preventDefault();
          this.processCommand(this.inputEl.value.trim());
          this.inputEl.value = '';
        } else if (e.key === 'ArrowUp') {
          e.preventDefault();
          this.navigateHistory(-1);
        } else if (e.key === 'ArrowDown') {
          e.preventDefault();
          this.navigateHistory(1);
        }

    navigateHistory(direction) {
      if (this.commandHistory.length === 0) return;
      this.historyIndex += direction;
      if (this.historyIndex < 0) this.historyIndex = 0;
      if (this.historyIndex >= this.commandHistory.length) {
        this.historyIndex = this.commandHistory.length;
        this.inputEl.value = '';
        return;
      }
      this.inputEl.value = this.commandHistory[this.historyIndex];
    }

    printLine(text, className = '', speed = 12) {

      line.className = 'output-line ' + className;
      this.consoleEl.appendChild(line);

      if (className === 'system') {
        line.textContent = text;
        this.autoScroll();
        return;
      }

      this.typewrite(line, text, speed, () => {
        this.autoScroll();
      });
    }

    typewrite(element, text, speed, callback) {
      this.typewriterActive = true;
      this.skipTypewriter = false;
      let i = 0;

      const type = () => {
        if (this.skipTypewriter || i >= text.length) {
          element.textContent = text;
          this.typewriterActive = false;
          this.skipTypewriter = false;
          if (callback) callback();
          return;
        }

        // Handle HTML-like markers for styling
        element.textContent = text.substring(0, i + 1);
        i++;

        const delay = text[i - 1] === '.' || text[i - 1] === '!' || text[i - 1] === '?' ? speed * 8 :
                       text[i - 1] === ',' ? speed * 4 :
                       text[i - 1] === ' ' ? speed * 1.5 : speed;

        this.typewriterTimeout = setTimeout(type, delay);
      };

      type();
    }

    printLineInstant(text, className = '') {
      const line = document.createElement('div');
      line.className = 'output-line ' + className;
      line.textContent = text;
      this.consoleEl.appendChild(line);
      this.autoScroll();
    }

    autoScroll() {
      this.consoleEl.scrollTop = this.consoleEl.scrollHeight;
    }

    // ---- COMMAND PROCESSING ----
    processCommand(raw) {
      if (!raw) return;

      const cmd = raw.toLowerCase().trim();

      // Save to history
      this.commandHistory.push(raw);
      this.historyIndex = this.commandHistory.length;

      // Echo command
      this.printLineInstant('> ' + raw, 'system');

      const parts = cmd.split(/\s+/);
      const verb = parts[0];
      const noun = parts.slice(1).join(' ');

      this.execute(verb, noun);
      this.updateUI();
    }

        'north': 'go', 'south': 'go', 'east': 'go', 'west': 'go',
        'ne': 'go', 'nw': 'go', 'se': 'go', 'sw': 'go',
        'up': 'go', 'down': 'go', 'in': 'go', 'out': 'go',
        'l': 'look',
        'help': 'help', 'h': 'help', '?': 'help',
        'reset': 'reset', 'restart': 'reset',
        'quit': 'quit', 'exit': 'quit',
      };

      // Handle two-word verbs
      let resolvedVerb = vMap[verb] || verb;

      // Direction shorthand
      if (['n','s','e','w','ne','nw','se','sw','up','down','in','out'].includes(verb)) {
        resolvedVerb = 'go';
        noun = verb;
      }

      // Look
      if (resolvedVerb === 'look' || resolvedVerb === 'l') {
        this.look(noun);
        return;
      }

      if (resolvedVerb === 'lookat') {
        this.lookAt(noun);
        return;
      }

      // Go
      if (resolvedVerb === 'go') {
        this.go(noun);
        return;
      }

      // Take
      if (resolvedVerb === 'take' || resolvedVerb === 'grab' || resolvedVerb === 'pick') {
        this.take(noun);
        return;
      }

      // Drop
      if (resolvedVerb === 'drop' || resolvedVerb === 'discard') {
        this.drop(noun);
        return;
      }

      // Use
      if (resolvedVerb === 'use' || resolvedVerb === 'pull' || resolvedVerb === 'activate') {
        this.use(noun);
        return;
      }

      // Inventory
      if (resolvedVerb === 'inventory' || resolvedVerb === 'inv') {
        this.showInventory();
        return;
      }

      // Help
      if (resolvedVerb === 'help') {
        this.showHelp();
        return;
      }

      // Reset
      if (resolvedVerb === 'reset' || resolvedVerb === 'restart') {
        this.reset();
        return;
      }

      // Quit
      if (resolvedVerb === 'quit' || resolvedVerb === 'exit') {
        this.printLine('The dungeon awaits your return, explorer.', 'system');
        return;
      }

      // Unknown
      this.printLine('I don\'t understand that command. Type "help" for available commands.', 'error');
    }

    // ---- ACTIONS ----
    look(noun) {
      if (noun) {
        this.lookAt(noun);
        return;
      }

      const room = this.rooms[this.currentRoomId];
      this.printLine('── ' + room.name + ' ──', 'success');
      this.printLine(room.description);

      // Show items in room
      if (room.items.length > 0) {
        this.printLine('');
        const itemNames = room.items.map(i => i.name).join(', ');
        this.printLine('You can see: ' + itemNames);
      }

      // Show exits
      const exitNames = Object.keys(room.exits).map(e => e);
      this.printLine('Exits: ' + exitNames.join(', '));

      // Check locked exits
      if (room.lockedExits) {
        for (const [dir, lockId] of Object.entries(room.lockedExits)) {
          if (lockId === 'north_door_locked' && !this.flags.north_door_unlocked) {
            this.printLine('  A heavy iron door to the ' + dir + ' is locked.', 'system');
          }
        }
      }
    }

    lookAt(noun) {
      if (!noun) {
        this.printLine('Look at what? Be more specific.', 'error');
        return;
      }

      const room = this.rooms[this.currentRoomId];

      // Check room items
      for (const item of room.items) {
        if (this.partialMatch(noun, item.name)) {
          this.printLine(item.description, 'success');
          return;
        }
      }

      // Check inventory items
      for (const item of this.inventory) {
        if (this.partialMatch(noun, item.name)) {
          this.printLine(item.description, 'success');
          return;
        }
      }

      // Check specific objects
      if (this.partialMatch(noun, 'lever')) {
        this.printLine('A stone lever protrudes from the wall, carved with ancient runes. It looks sturdy enough to pull.');
        return;
      }
      if (this.partialMatch(noun, 'door') && this.currentRoomId === 'armory') {
        if (!this.flags.north_door_unlocked) {
          this.printLine('A heavy iron door, secured with a rusted lock. You\'ll need a key to open it.');
        } else {
          this.printLine('The door stands open, revealing a passage to the north.');
        }
        return;
      }
      if (this.partialMatch(noun, 'door') && this.currentRoomId === 'entrance') {
        this.printLine('The door you entered through. It\'s heavy and made of dark oak, its iron hinges creaking softly in the still air.');
        return;
      }
      if (this.partialMatch(noun, 'fountain')) {
        this.printLine('A small stone fountain trickles gently. The water is crystal clear and smells of rain.');
        return;
      }
      if (this.partialMatch(noun, 'bench')) {
        this.printLine('A weathered stone bench sits beneath a flowering vine. Someone has left a silver rose upon it.');
        return;
      }
      if (this.partialMatch(noun, 'wall') || this.partialMatch(noun, 'shelves') || this.partialMatch(noun, 'floor')) {
        this.printLine('The surfaces are covered in a thick layer of dust, disturbed only by the faint scuff marks of previous visitors.');
        return;
      }

      this.printLine('You don\'t see "' + noun + '" here.', 'error');
    }

    go(direction) {
      if (!direction) {
        this.printLine('Go where? Specify a direction: north, south, east, or west.', 'error');
        return;
      }

      const room = this.rooms[this.currentRoomId];
      const dirMap = {
        'n': 'north', 'north': 'north',
        's': 'south', 'south': 'south',
        'e': 'east', 'east': 'east',
        'w': 'west', 'west': 'west',
        'ne': 'northeast', 'ne': 'northeast',
        'nw': 'northwest', 'nw': 'northwest',
        'se': 'southeast', 'se': 'southeast',
        'sw': 'southwest', 'sw': 'southwest',
        'up': 'up', 'down': 'down',
        'in': 'in', 'out': 'out'
      };

      const dirKey = dirMap[direction] || direction;

      // Check if exit exists
      if (!room.exits[dirKey]) {
        this.printLine('You can\'t go that way. There\'s no passage to the ' + dirKey + '.', 'error');
        return;
      }

      // Check locked exits
      if (room.lockedExits && room.lockedExits[dirKey]) {
        const lockId = room.lockedExits[dirKey];
        if (lockId === 'north_door_locked' && !this.flags.north_door_unlocked) {
          this.printLine('The heavy iron door is locked. You need a key to open it.', 'error');
          return;
        }
        if (lockId === 'secret_door_locked' && !this.flags.secret_chamber_unlocked) {
          this.printLine('The wall to the west is solid stone. There\'s no way through.', 'error');
          return;
        }
      }

      // Move
      this.currentRoomId = room.exits[dirKey];
      this.visitedRooms.add(this.currentRoomId);
      this.moves++;

      const newRoom = this.rooms[this.currentRoomId];
      this.printLine('You travel ' + dirKey + '.');
      this.look();

      // Check for game win
      if (this.currentRoomId === 'garden' && this.visitedRooms.size >= 4) {
        this.flags.game_won = true;
        setTimeout(() => {
          this.printLine('');
          this.printLine('═══════════════════════════════════════', 'success');
          this.printLine('  You have found your way to the Garden', 'success');
          this.printLine('  and discovered the peace hidden', 'success');
          this.printLine('  beneath the dungeon.', 'success');
          this.printLine('  Thank you for playing The Local Dungeon.', 'success');
          this.printLine('═══════════════════════════════════════', 'success');
        }, 800);
      }
    }

    take(noun) {
      if (!noun) {
        this.printLine('Take what? Be specific.', 'error');
        return;
      }

      const room = this.rooms[this.currentRoomId];
      const itemIndex = room.items.findIndex(i => this.partialMatch(noun, i.name));

      if (itemIndex === -1) {
        this.printLine('You don\'t see "' + noun + '" here to take.', 'error');
        return;
      }

      const item = room.items[itemIndex];

      if (!item.takeable) {
        this.printLine('You can\'t take that.', 'error');
        return;
      }

      room.items.splice(itemIndex, 1);
      this.inventory.push(item);
      this.moves++;
      this.printLine('You take the ' + item.name + '.', 'success');
    }

    drop(noun) {
      if (!noun) {
        this.printLine('Drop what? Be specific.', 'error');
        return;
      }

      const itemIndex = this.inventory.findIndex(i => this.partialMatch(noun, i.name));

      if (itemIndex === -1) {
        this.printLine('You\'re not carrying "' + noun + '".', 'error');
        return;
      }

      const item = this.inventory[itemIndex];
      this.inventory.splice(itemIndex, 1);
      this.rooms[this.currentRoomId].items.push(item);
      this.moves++;
      this.printLine('You drop the ' + item.name + '.', 'success');
    }

    use(noun) {
      if (!noun) {
        this.printLine('Use what? Be specific.', 'error');
        return;
      }

      const itemIndex = this.inventory.findIndex(i => this.partialMatch(noun, i.name));

      if (itemIndex === -1) {
        this.printLine('You don\'t have "' + noun + '" to use.', 'error');
        return;
      }

      const item = this.inventory[itemIndex];

      // Use Rusted Key
      if (item.id === 'rusted_key') {
        if (this.currentRoomId === 'entrance' || this.currentRoomId === 'library' || this.currentRoomId === 'garden' || this.currentRoomId === 'secret_chamber') {
          this.printLine('There\'s nothing here that needs a key. Perhaps you should try the Armory?');
          return;
        }
        if (this.currentRoomId === 'armory') {
          if (!this.flags.north_door_unlocked) {
            this.flags.north_door_unlocked = true;
            this.moves++;
            this.printLine('You insert the rusted key into the lock. With a grinding click, the heavy iron door swings open, revealing a passage to the north.', 'success');
          } else {
            this.printLine('The door is already unlocked.');
          }
          return;
        }
        this.printLine('That doesn\'t seem to be the right place to use a key.', 'error');
        return;
      }

      // Use Lever
      if (item.id === 'lever') {
        if (this.currentRoomId === 'library') {
          if (!this.flags.lever_pulled) {
            this.flags.lever_pulled = true;
            this.flags.secret_chamber_unlocked = true;
            this.moves++;
            this.printLine('You pull the stone lever with all your strength. Deep within the wall, gears grind and stone shifts. A hidden passage opens to the west!', 'success');
          } else {
            this.printLine('The lever has already been pulled. The passage remains open.');
          }
          return;
        }
        this.printLine('You don\'t see a wall here that needs a lever.', 'error');
        return;
      }

      // Use Torch
      if (item.id === 'torch') {
        if (!this.flags.torch_lit) {
          this.flags.torch_lit = true;
          this.moves++;
          this.printLine('You strike a flint against the stone floor and light the torch. Flames leap to life, casting warm light across the chamber.', 'success');
        } else {
          this.printLine('The torch is already burning brightly.');
        }
        return;
      }

      // Use Golden Coin
      if (item.id === 'golden_coin') {
        this.printLine('You hold the golden coin up to the light. It\'s beautiful, but there\'s nothing here to spend it on.');
        return;
      }

      // Use Ancient Scroll
      if (item.id === 'ancient_scroll') {
        this.printLine('You carefully unroll the ancient scroll. The faded text speaks of a garden hidden beneath the dungeon — a place of peace and light, guarded by the weight of all the knowledge above.');
        return;
      }

      // Use Crystal Bottle
      if (item.id === 'bottle') {
        this.printLine('You uncork the crystal bottle. The shimmering liquid inside glows softly, casting a gentle blue light over everything around you.');
        return;
      }

      // Use Silver Rose
      if (item.id === 'rose') {
        this.printLine('You pick up the silver rose. The craftsmanship is extraordinary — every petal is perfectly formed. You feel a sense of quiet triumph.');
        return;
      }

      // Use Candle
      if (item.id === 'candle') {
        this.printLine('The candle is thick and sturdy. Perhaps you could light it with a torch?');
        return;
      }

      this.printLine('Using "' + item.name + '" here doesn\'t seem to do anything useful.', 'error');
    }

    showInventory() {
      if (this.inventory.length === 0) {
        this.printLine('Your inventory is empty.', 'system');
      } else {
        this.printLine('── Inventory ──', 'success');
        this.inventory.forEach(item => {
          this.printLine('  • ' + item.name);
        });
      }
    }

    showHelp() {
      this.printLine('═══ Available Commands ═══', 'success');
      this.printLine('  look                  — Examine your surroundings');
      this.printLine('  look at [object]      — Examine a specific item');
      this.printLine('  go [direction]        — Travel north, south, east, west');
      this.printLine('  take [item] / grab    — Pick up an object');
      this.printLine('  drop [item]           — Leave an item behind');
      this.printLine('  use [item]            — Use an item (key, lever, torch)');
      this.printLine('  inventory / i         — View what you\'re carrying');
      this.printLine('  help                  — Show this message');
      this.printLine('  reset                 — Restart the game');
    }

    // ---- UTILITIES ----
    partialMatch(query, target) {
      const q = query.toLowerCase().trim();
      const t = target.toLowerCase().trim();
      return q.length >= 2 && (t.includes(q) || q.includes(t));
    }

    updateUI() {
      // Stats
      this.statMovesEl.textContent = this.moves;
      this.statItemsEl.textContent = this.inventory.length;
      this.statRoomsEl.textContent = this.visitedRooms.size;

      // Inventory
      this.inventoryListEl.innerHTML = '';
      if (this.inventory.length === 0) {
        this.inventoryEmptyEl.style.display = 'block';
      } else {
        this.inventoryEmptyEl.style.display = 'none';
        this.inventory.forEach(item => {
          const li = document.createElement('li');
          li.innerHTML = '<span class="item-icon">◆</span> ' + item.name;
          this.inventoryListEl.appendChild(li);
        });
      }
    }

    reset() {
      // Clear typewriter
      if (this.typewriterTimeout) {
        clearTimeout(this.typewriterTimeout);
        this.typewriterTimeout = null;
      }
      this.typewriterActive = false;
      this.skipTypewriter = false;

      // Clear console
      this.consoleEl.innerHTML = '';

      // Reset state
      this.currentRoomId = 'entrance';
      this.inventory = [];
      this.visitedRooms = new Set();
      this.moves = 0;
      this.commandHistory = [];
      this.historyIndex = -1;
      this.flags = {
        lever_pulled: false,
        secret_chamber_unlocked: false,
        north_door_unlocked: false,
        torch_lit: false,
        game_won: false,
        lever_used_in_room: false
      };

      // Rebuild world (restore items to rooms)
      this.buildWorld();

      // Restart
      this.start();
    }
  }

  // ==================== INITIALIZE ====================
  window.addEventListener('DOMContentLoaded', () => {
    window.game = new GameEngine();
  });
})();
