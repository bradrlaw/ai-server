'use strict';

/* ============================= Item ============================= */
class Item {
  constructor(id, name, description) {
    this.id = id;
    this.name = name;
    this.description = description;
  }
  matches(text) {
    const t = text.trim().toLowerCase();
    const n = this.name.toLowerCase();
    return t === n || n.includes(t) || (t.length > 2 && n.startsWith(t));
  }
}

/* ============================= Room ============================= */
class Room {
  constructor(id, name, description, exits = {}, features = {}) {
    this.id = id;
    this.name = name;
    this.description = description;   // function(game) -> string, or string
    this.exits = exits;               // { north: 'library', ... }
    this.items = [];                  // Item objects on the floor
    this.features = features;         // { lever: 'desc', torch: ..., ... } examine-able things
  }
  getDescription(game) {
    return typeof this.description === 'function' ? this.description(game) : this.description;
  }
  exitList(game) {
    const open = Object.keys(this.exits).filter(dir => !(this.id === 'armory' && dir === 'north' && !game.flags.treasure_door_open));
    if (!open.length) return 'There are no obvious exits. Impressive, really.';
    return 'Exits: ' + open.join(', ') + '.';
  }
  findItem(text) {
    return this.items.find(it => it.matches(text)) || null;
  }
}

  typeLines(lines, done) {

      if (this.skipRequested) {

      cursor.className = 'cursor-blink';
      cursor.innerHTML = '&nbsp;';
      this.outputEl.appendChild(span);
      this.outputEl.appendChild(cursor);
      this.scrollDown();

      let ci = 0;
      const tick = () => {
        if (this.skipRequested) {
          span.textContent = text;
          cursor.remove();
          this.scrollDown();
          // finish rest instantly
          for (let i = li; i < lines.length; i++) this.appendInstant(lines[i].text, lines[i].cls);
          this.finish();
          return;
        }
        if (ci < text.length) {
          span.textContent += text[ci++];
          this.scrollDown();
          setTimeout(tick, text[ci - 1] === '\n' ? 40 : 12);
        } else {
          cursor.remove();
          const br = document.createElement('br');
          this.outputEl.appendChild(br);
          this.scrollDown();
          setTimeout(nextLine, 60);
        }
      };
      if (text === '') {
        cursor.remove();
        this.outputEl.appendChild(document.createElement('br'));
        setTimeout(nextLine, 20);
      } else {
        tick();
      }
    };

    this.finish = () => {
      this.typing = false;
      this.skipRequested = false;
      this.scrollDown();
      if (done) done();
    };

    nextLine();
  }

  appendInstant(text, cls) {
    const span = document.createElement('span');
    if (cls) span.className = cls;
    span.textContent = text;
    this.outputEl.appendChild(span);
    this.outputEl.appendChild(document.createElement('br'));
    this.scrollDown();
  }

  echo(text) {
    this.appendInstant('> ' + text, 'echo');
    this.scrollDown();
  }

  scrollDown() { this.outputEl.scrollTop = this.outputEl.scrollHeight; }

  skip() { if (this.typing) this.skipRequested = true; }

  /* ---------- Input / history ---------- */
  setupInput() {
    this.inputEl.addEventListener('keydown', (e) => {
      if (e.key === 'Enter') {
        const raw = this.inputEl.value;
        this.inputEl.value = '';
        if (this.typing) { this.skip(); return; }
        if (raw.trim()) {
          this.echo(raw.trim());
          this.history.push(raw.trim());
          this.historyIndex = this.history.length;
        }
        this.handle(raw.trim());
      } else if (e.key === 'ArrowUp') {
        e.preventDefault();
        if (!this.history.length) return;
        this.historyIndex = Math.max(0, this.historyIndex - 1);
        this.inputEl.value = this.history[this.historyIndex] || '';
        this.caretEnd();
      } else if (e.key === 'ArrowDown') {
        e.preventDefault();
        if (!this.history.length) return;
        this.historyIndex = Math.min(this.history.length, this.historyIndex + 1);
        this.inputEl.value = this.history[this.historyIndex] || '';
        this.caretEnd();
      }
    });

    this.outputEl.addEventListener('click', () => this.inputEl.focus());
    document.getElementById('console-wrap').addEventListener('click', (e) => {
      if (e.target.id !== 'skip-btn') this.inputEl.focus();
    });
    this.skipBtn.addEventListener('click', (e) => { e.stopPropagation(); this.skip(); this.inputEl.focus(); });
  }

  caretEnd() {
    const v = this.inputEl.value;
    this.inputEl.setSelectionRange(v.length, v.length);
  }

  /* ---------- Parser ---------- */
  parse(raw) {
    const input = raw.trim().toLowerCase().replace(/\s+/g, ' ');
    if (!input) return null;
    const words = input.split(' ');
    const verb = words[0];
    let rest = words.slice(1).join(' ');

    // Direction shortcuts (whole input)
    const dirMap = { n: 'north', s: 'south', e: 'east', w: 'west', north: 'north', south: 'south', east: 'east', west: 'west' };
    if (dirMap[verb] && words.length === 1) return { cmd: 'go', target: dirMap[verb] };
    if (verb === 'n' || verb === 's' || verb === 'e' || verb === 'w') return { cmd: 'go', target: dirMap[verb] };

    // Verb synonyms with partial matching
    const verbTable = [
      { cmd: 'look',    verbs: ['look', 'l', 'examine', 'inspect', 'exam', 'x', 'study', 'read', 'check'] },
      { cmd: 'go',      verbs: ['go', 'walk', 'move', 'head', 'travel', 'run', 'enter', 'exit'] },
      { cmd: 'take',    verbs: ['take', 'grab', 'get', 'pick', 'steal', 'acquire', 'loot'] },
      { cmd: 'drop',    verbs: ['drop', 'leave', 'put', 'discard', 'place'] },
      { cmd: 'use',     verbs: ['use', 'pull', 'push', 'turn', 'operate', 'activate', 'unlock', 'open', 'light'] },
      { cmd: 'inventory', verbs: ['inventory', 'inv', 'i', 'items', 'bag', 'possessions', 'gear'] },
      { cmd: 'help',    verbs: ['help', 'commands', 'hint', 'hints', '?'] },
      { cmd: 'wait',    verbs: ['wait', 'idle', 'breathe', 'listen'] }
    ];

    let matched = null;
    for (const entry of verbTable) {
      if (entry.verbs.some(v => v === verb || (verb.length >= 2 && v.startsWith(verb)))) { matched = entry.cmd; break; }
    }
    if (!matched) return { cmd: 'unknown', raw: input };

    // "look at X" / "pick up X" cleanup
    rest = rest.replace(/^(at|around|about|up|on|the|a|an)\s+/i, rest.startsWith('at ') || rest.startsWith('around ') || rest.startsWith('up ') ? rest.replace(/^(at|around|up)\s+/i, '') : rest.replace(/^(the|a|an)\s+/i, ''));
    // strip leading articles
    rest = rest.replace(/^(the|a|an)\s+/i, '').trim();

    // "use key on door" style
    let extra = null;
    const onMatch = rest.split(/\s+on\s+/);
    if (matched === 'use' && onMatch.length === 2) { rest = onMatch[0].trim(); extra = onMatch[1].trim(); }

    return { cmd: matched, target: rest, extra };
  }

  /* ---------- Command handling ---------- */
  handle(raw) {
    if (this.typing) { this.skip(); return; }
    const parsed = this.parse(raw);
    if (!parsed) return;
    const { cmd, target, extra } = parsed;
    let lines = [];

    switch (cmd) {
      case 'look':
        if (target) lines = this.examine(target);
        else lines = this.describeRoom(true);
        break;
      case 'go':   lines = this.go(target); break;
      case 'take': lines = this.take(target); break;
      case 'drop': lines = this.drop(target); break;
      case 'use':  lines = this.use(target, extra); break;
      case 'inventory': lines = this.inventoryCmd(); break;
      case 'help': lines = this.helpCmd(); break;
      case 'wait': lines = [{ text: 'You wait. The dungeon waits back. It is, objectively, better at waiting than you are.', cls: 'flavor' }]; break;
      default:
        lines = [{ text: `The dungeon does not respond to "${raw}". Try: look, go <direction>, take <item>, use <item>, examine <thing>, inventory, or help.`, cls: 'error' }];
    }

    this.typeLines(lines, () => this.updateUI());
  }

  describeRoom(includeTitle) {
    const room = this.room;
    const firstVisit = !this.visited.has(room.id);
    this.visited.add(room.id);
    const lines = [];
    if (includeTitle || firstVisit) lines.push({ text: `[ ${room.name} ]`, cls: 'system' });
    room.getDescription(this).split('\n').forEach(l => lines.push({ text: l, cls: null }));
    lines.push({ text: room.exitList(this), cls: 'flavor' });
    if (room.items.length) {
      lines.push({ text: 'You notice: ' + room.items.map(i => i.name).join(', ') + '.', cls: 'flavor' });
    }
    const feats = Object.keys(room.features);
    if (feats.length && firstVisit) {
      lines.push({ text: 'Also visible: ' + feats.join(', ') + '. (Try: examine <thing>)', cls: 'flavor' });
    }
    return lines;
  }

  examine(target) {
    if (!target) return [{ text: 'Examine what? The darkness? Done. It examines you back.', cls: 'error' }];
    const room = this.room;
    // features (partial match on keys)
    for (const [key, desc] of Object.entries(room.features)) {
      if (key.includes(target) || target.includes(key)) {
        return [{ text: desc, cls: 'flavor' }];
      }
    }
    // items on floor
    const item = room.findItem(target);
    if (item) return [{ text: item.description, cls: 'flavor' }];
    // carried items
    const held = this.inventory.find(i => i.matches(target));
    if (held) return [{ text: held.description, cls: 'flavor' }];
    return [{ text: `You squint intently, but "${target}" is not here — or is not a thing, or both.`, cls: 'error' }];
  }

  go(dir) {
    if (!dir) return [{ text: 'Go where? Give a direction: north, south, east, or west.', cls: 'error' }];
    const dirMap = { n: 'north', s: 'south', e: 'east', w: 'west' };
    dir = dirMap[dir] || dir.replace(/ern$/, '').replace(/ward$/, '');
    const room = this.room;

    // Locked armory north door
    if (room.id === 'armory' && dir === 'north' && !this.flags.treasure_door_open) {
      return [{ text: 'The iron-bound door to the north is padlocked shut. The padlock is old, but older. You need a key of similarly ancient vintage. (Try: use rusted key on door)', cls: 'error' }];
    }
    // Secret chamber locked
    if (room.id === 'library' && dir === 'west' && !this.flags.secret_unlocked) {
      return [{ text: 'The western wall is solid stone — mostly. One seam runs a little too straight to be natural. Something here must open it.', cls: 'error' }];
    }

    // Torch darkness flavor
    const darkRooms = ['library', 'treasure', 'secret'];
    const hasTorch = this.inventory.some(i => i.id === 'torch');

    const destId = room.exits[dir];
    if (!destId) {
      const anyExit = Object.keys(room.exits);
      return [{ text: `You walk ${dir} until walking is no longer fun, and hit a wall. This room has exits to the ${anyExit.join(', ')} only.`, cls: 'error' }];
    }

    this.currentRoomId = destId;
    this.moves++;
    if (destId === 'garden') this.flags.garden_hinted = true;

    const lines = this.describeRoom(true);
    if (darkRooms.includes(destId) && !hasTorch) {
      lines.push({ text: 'It is very dark here. A torch would be a strong lifestyle choice. You stub your toe on history.', cls: 'flavor' });
    }
    return lines;
  }

  take(target) {
    if (!target) return [{ text: 'Take what? Your leave? Tempting. Be specific.', cls: 'error' }];
    const item = this.room.findItem(target);
    if (!item) {
      if (this.inventory.some(i => i.matches(target))) return [{ text: `You already have the ${item ? item.name : target}. Greed looks better on you than this hesitation.`, cls: 'error' }];
      return [{ text: `There is no "${target}" here to take. Look around first — try "look".`, cls: 'error' }];
    }
    this.room.items = this.room.items.filter(i => i !== item);
    this.inventory.push(item);
    this.itemsFound++;

    const lines = [{ text: `You take the ${item.name}.`, cls: 'system' }, { text: item.description, cls: 'flavor' }];
    if (item.id === 'coin') this.flags.treasure_taken = true;
    if (item.id === 'scroll') this.flags.scroll_taken = true;
    if (item.id === 'torch') this.flags.torch_taken = true;
    if (item.id === 'scroll') lines.push({ text: 'Something about the letter mentions a garden... and the writer sounds tired. Go south, someday. Be kinder than they were to themselves.', cls: 'flavor' });
    return lines;
  }

  drop(target) {
    if (!target) return [{ text: 'Drop what? Your insecurities are not yet inventory-manageable.', cls: 'error' }];
    const item = this.inventory.find(i => i.matches(target));
    if (!item) return [{ text: `You are not carrying any "${target}".`, cls: 'error' }];
    this.inventory = this.inventory.filter(i => i !== item);
    this.room.items.push(item);
    return [{ text: `You set the ${item.name} down on the cold stone floor. It joins the dust, which has been collecting things far longer than you.`, cls: 'system' }];
  }

  use(target, extra) {
    if (!target) return [{ text: 'Use what? Time? You have plenty. Name an item or feature.', cls: 'error' }];
    const room = this.room;
    const held = this.inventory.find(i => i.matches(target));
    const combined = (target + ' ' + (extra || '')).toLowerCase();

    // Lever (library)
    if (room.id === 'library' && (target.includes('lever') || combined.includes('lever'))) {
      if (!this.flags.lever_pulled) {
        this.flags.lever_pulled = true;
        this.flags.secret_unlocked = true;
        return [
          { text: 'You grip the iron lever and pull with your whole body. For one hopeful second, nothing — then the entire western wall GRINDS inward, shelves swinging aside on counterweights as a hidden doorway opens onto candlelit darkness.', cls: null },
          { text: '*** A SECRET CHAMBER is now accessible to the WEST. ***', cls: 'system' }
        ];
      }
      return [{ text: 'The lever is already thrown. You jiggle it anyway, like everyone who has ever used a vending machine.', cls: 'flavor' }];
    }

    // Key on armory door
    if (room.id === 'armory' && (combined.includes('key') || target.includes('key') || combined.includes('door') || combined.includes('lock') || combined.includes('padlock'))) {
      const key = this.inventory.find(i => i.id === 'key');
      if (!key) return [{ text: 'The padlock laughs at your bare hands. You need a key — the Entrance Hall had something rusty on the floor, if memory serves.', cls: 'error' }];
      if (this.flags.treasure_door_open) return [{ text: 'The door is already open. The key has found peace.', cls: 'flavor' }];
      this.flags.treasure_door_open = true;
      return [
        { text: 'You work the rusted key into the ancient padlock. It resists, complains in a language of metal and grievance — then CRACKS open and drops to the floor, its centuries-long shift finally over.', cls: null },
        { text: '*** The north door to the TREASURE ROOM is unlocked. ***', cls: 'system' }
      ];
    }

    // Light the torch (flavor)
    if (held && held.id === 'torch') {
      return [{ text: 'The torch is already burning, loyal as a dog. Its light makes the shadows file a formal complaint.', cls: 'flavor' }];
    }

    // Coin in garden?
    if (room.id === 'garden' && combined.includes('coin')) {
      return [{ text: 'You flip the golden coin into the stream. It glitters once, twice, and becomes a small bright fish that salutes you before disappearing. Probably.', cls: 'flavor' }];
    }

    return [{ text: `Using "${target}" here achieves nothing, except a small sense of adventure-related futility. Try examining it, or a different target.`, cls: 'error' }];
  }

  inventoryCmd() {
    if (!this.inventory.length) return [{ text: 'You pat your pockets. Nothing but lint, courage, and a tavern token for a pub that closed in 1342.', cls: 'flavor' }];
    const lines = [{ text: 'You are carrying:', cls: 'system' }];
    this.inventory.forEach(i => lines.push({ text: '  • ' + i.name, cls: 'flavor' }));
    return lines;
  }

  helpCmd() {
    return [
      { text: '--- AVAILABLE COMMANDS ---', cls: 'system' },
      { text: 'look / l ............ describe the room and exits', cls: 'flavor' },
      { text: 'look at X / examine X  inspect an object closely', cls: 'flavor' },
      { text: 'go north / n / s / e / w ... move between rooms', cls: 'flavor' },
      { text: 'take X / grab X .... pick up an item', cls: 'flavor' },
      { text: 'drop X ............. put an item down', cls: 'flavor' },
      { text: 'use X (on Y) ....... pull levers, unlock doors, etc.', cls: 'flavor' },
      { text: 'inventory / inv / i  list carried items', cls: 'flavor' },
      { text: 'Hint: the Dungeon Master retired for a reason, and the garden gate is somewhere south of where you started.', cls: null }
    ];
  }

  /* ---------- UI ---------- */
  updateUI() {
    this.movesEl.textContent = this.moves;
    this.itemsEl.textContent = this.itemsFound;
    this.locEl.textContent = this.room.name;

    this.invEl.innerHTML = '';
    if (!this.inventory.length) {
      const li = document.createElement('li');
      li.className = 'empty';
      li.textContent = 'Inventory is empty.';
      this.invEl.appendChild(li);
    } else {
      this.inventory.forEach(item => {
        const li = document.createElement('li');
        li.innerHTML = '<svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="#00c853" stroke-width="2.5" stroke-linecap="round" stroke-linejoin="round"><path d="M20 6 9 17l-5-5"/></svg>';
        li.appendChild(document.createTextNode(item.name));
        this.invEl.appendChild(li);
      });
    }
  }
}
