/* =========================================================
   Item
   ========================================================= */
class Item {
  constructor(name, description) {
    this.name = name;           // canonical name (lowercase)
    this.description = description;
  }
  matches(word) {
    // word is a single normalized token like "rusted" or "key"
    return this.name.includes(word) || word.startsWith(this.name.split(" ")[0]);
  }
}

/* =========================================================
   Room
   ========================================================= */
class Room {
  constructor(name, description, exits = {}, items = [], lockedExits = {}) {
    this.name = name;
    this.description = description;
    this.exits = exits;          // { direction: roomId }
    this.items = items;          // [Item]
    this.lockedExits = lockedExits; // { direction: { message, unlockCmd, check: fn } }
  }
  item(nameOrWord) {
    if (typeof nameOrWord === "string" && nameOrWord.includes(" ")) {
      const wanted = nameOrWord;
      return this.items.find(it => it.name === wanted || it.name.startsWith(wanted));
    }
    return this.items.find(it => it.matches(nameOrWord));
  }
  exitText() {
    const dirs = Object.keys(this.exits);
    const labels = { n: "north", s: "south", e: "east", w: "west" };
    return dirs.map(d => labels[d] || d).join(", ");
  }
}

    if (["take", "grab", "get", "pick", "loot"].includes(cmd)) {
      const rest = normalized.slice(tokens[0].length + 1).trim();
      if (!rest) { this.say("Take what? You'll need to name an item. Try: take torch", true); return; }
      this.take(rest);
      return;
    }

    if (["use", "operate", "pull", "work"].includes(cmd)) {
      const rest = normalized.slice(tokens[0].length + 1).trim();
      if (!rest) { this.say("Use what? Try: use lever", true); return; }
      this.use(rest);
      return;
    }

    if (["open", "unlock", "lock"].includes(cmd)) {
      const rest = normalized.slice(tokens[0].length + 1).trim();
      this.unlock(rest);
      return;
    }

    if (["inv", "inventory", "list", "items", "sack", "show"].includes(cmd)) {
      this.showInventory();
      return;
    }

    if (["help", "h", "?", "commands"].includes(cmd)) {
      this.help();
      return;
    }

    if (cmd === "go" || cmd === "walk" || cmd === "move" || cmd === "head" || cmd === "travel") {
      const rest = normalized.slice(tokens[0].length + 1).trim();
      const d = dirMap[rest];
      if (d) { this.go(d); }
      else { this.say(`Go ${rest || "where"}? Valid directions: north, south, east, west.`, true); }
      return;
    }

    const bareDir = dirMap[cmd];
    if (bareDir) { this.go(bareDir); return; }

    if (cmd === "wait" || cmd === "rest") {
      this.say("You wait. The dungeon waits with you. Nothing moves except the dust.", true);
      return;
    }

    if (cmd === "quit" || cmd === "exit") {
      this.say("The stairs behind you are always open. But you've come this far — try the garden to the south first.", true);
      return;
    }

    this.say(`I don't understand "${input}". Type 'help' to see what I do understand.`, true);
  }

  /* ---------------- Actions ---------------- */
  look() {
    const r = this.room;
    this.say(`You are in the ${r.name}.`, true);
    this.say(r.description, true);
    this.blank(true);
    this.say(`Exits: ${r.exitText()}.`, true, "dim");
    if (r.items.length) {
      this.say(`You can see: ${r.items.map(i => i.name).join(", ")}.`, true, "dim");
    }
  }

  lookAt(target) {
    const here = this.room.item(target);
    if (here) {
      this.say(`${here.name.charAt(0).toUpperCase() + here.name.slice(1)}: ${here.description}`, true);
      return;
    }
    const held = this.inventory.find(i => i.matches(target.split(" ")[0]));
    if (held) {
      this.say(`${held.name.charAt(0).toUpperCase() + held.name.slice(1)}: ${held.description}`, true);
      return;
    }
    this.say(`You don't see anything like "${target}" here.`, true);
  }

  take(target) {
    const item = this.room.item(target);
    if (item) {
      this.room.items = this.room.items.filter(i => i !== item);
      this.inventory.push(item);
      this.itemsFound++;
      this.say(`You take the ${item.name}. It fits easily in your pack.`, true);
    } else if (this.inventory.some(i => i.matches(target.split(" ")[0]))) {
      this.say(`You already have the ${target.split(" ")[0] === "key" ? "key" : target}.`, true);
    } else {
      this.say(`There's no "${target}" here to take.`, true);
    }
  }

  use(target) {
    if (target.includes("lever")) {
      if (this.currentId !== "library") {
        this.say("There's no lever here.", true);
        return;
      }
      if (this.flags.lever_pulled) {
        this.say("The lever is already thrown. It rests at a satisfying angle.", true);
        return;
      }
      this.flags.lever_pulled = true;
      this.say("You pull the lever. It swings with a heavy, deliberate CLUNK.", true);
      this.blank(true);
      this.say("Deep in the wall, gears you didn't know existed grind past each other. The sealed west doorway shivers — then slides open a crack, revealing a narrow passage into darkness.", true);
      this.blank(true);
      this.say("The secret chamber to the west is now open. Try: go west", true, "sys");
      return;
    }
    if (target.includes("torch")) {
      this.say("You strike the torch. It catches instantly and throws the room into warm, swinging light.", true);
      return;
    }
    if (target.includes("coin")) {
      this.say("You flip the golden coin across your knuckles. It makes a sound like a small, expensive bell.", true);
      return;
    }
    if (target.includes("scroll")) {
      this.say("You unroll the ancient scroll. In its faded ink: 'May the coin bring plenty, and the garden bring peace.'", true);
      return;
    }
    if (target.includes("key")) {
      this.say("A key is for doors. Try: unlock north (from the armory).", true);
      return;
    }
    this.say(`You don't have anything useful to call a "${target}".`, true);
  }

  unlock(rest) {
    const d = (dirMap[rest] || ["north","south","east","west"].find(x => rest.includes(x)));
    if (!d) {
      this.say("Unlock what, where? Give me a direction. Try: unlock north", true);
      return;
    }
    const lock = this.room.lockedExits[d];
    if (!lock) {
      this.say(`There's no locked door to the ${dirLabel(d)} here.`, true);
      return;
    }
    if (lock.unlockCmd === "rusted key") {
      if (this.inventory.some(i => i.name === "rusted key")) {
        this.flags.armory_door_unlocked = true;
        this.say("You slide the rusted key into the hasp. The lock grinds, resists, then gives way with a crack like a knuckle.", true);
        this.blank(true);
        this.say("The north door swings open. Try: go north", true, "sys");
      } else {
        this.say("You have no key that would open this lock. It needs something rusted. Look in the Entrance Hall.", true);
      }
    } else if (lock.unlockCmd === "lever") {
      if (this.flags.lever_pulled) {
        this.say("The door is already open.", true);
      } else {
        this.say("The brass catch refuses your hands. This is a mechanism, not a lock — it answers to something else in the room.", true);
      }
    }
  }

  showInventory() {
    if (!this.inventory.length) {
      this.say("Your pack is empty. Only lint and ambition.", true);
      return;
    }
    this.say("You are carrying:", true);
    this.inventory.forEach(i => this.say(`  - ${i.name}`, true));
  }

  go(dir) {
    const exitTo = this.room.exits[dir];
    if (!exitTo) {
      this.say(`You can't go ${dirLabel(dir)} from here — the wall is solid stone.`, true);
      return;
    }
    const lock = this.room.lockedExits[dir];
    if (lock) {
      const unlocked = (dir === "w" && this.currentId === "library")
        ? this.flags.lever_pulled
        : (dir === "n" && this.currentId === "armory")
          ? this.flags.armory_door_unlocked
          : false;
      if (!unlocked) {
        this.say(lock.message, true);
        return;
      }
    }
    this.currentId = exitTo;
    this.moves++;
    this.visited.add(exitTo);
    this.say(`You move ${dirLabel(dir)} into the ${this.room.name}.`, true);
    this.blank(true);
    if (this.room.items.length && this.room.items.length < 2) {
      this.say(`You notice: ${this.room.items[0].name}.`, true, "dim");
    }
    if (exitTo === "garden") {
      this.blank(true);
      const hasCoin = this.inventory.some(i => i.name === "golden coin");
      const hasScroll = this.inventory.some(i => i.name === "ancient scroll");
      if (hasCoin && hasScroll) {
        this.say("You sit on the warm stone bench, coin in one hand, scroll in the other. The fountain murmurs.", true);
        this.blank(true);
        this.say("★★★ YOU HAVE COMPLETED THE LOCAL DUNGEON ★★★", true, "sys");
        this.say("The first mayor's coin will fund the cheese shop; the scroll's blessing covers the rest. Press 'Reset Game' above whenever you're ready to run it back.", true, "sys");
      } else {
        this.say("The bench is inviting, but something tells you there's more below before this place is truly finished with you. (Hint: the golden coin and the ancient scroll.)", true);
      }
    }
  }

  help() {
    this.say("Commands:", true, "sys");
    this.say("  look / examine        describe the room", true, "sys");
    this.say("  look at <item>        examine a specific item", true, "sys");
    this.say("  <n/s/e/w> or go <dir>  move around", true, "sys");
    this.say("  take / grab <item>    pick something up", true, "sys");
    this.say("  use <item>            operate something (use lever)", true, "sys");
    this.say("  unlock <dir>          unlock a locked door with a key", true, "sys");
    this.say("  inv / inventory       list what you carry", true, "sys");
    this.say("  help                  show this list", true, "sys");
  }
}

function dirLabel(d) {
  return { n: "north", s: "south", e: "east", w: "west" }[d] || d;
}

/* =========================================================
   Typewriter console
   ========================================================= */
const consoleOut = document.getElementById("console-out");
let pendingQueue = [];
let typing = false;
let skipRequested = false;
let typeTimer = null;

function scrollTop() { consoleOut.scrollTop = consoleOut.scrollHeight; }

function print(text, instant = false, className = "") {
  pendingQueue.push({ text, instant, className });
  if (!typing) pumpQueue();
}

function pumpQueue() {
  if (!pendingQueue.length) { typing = false; skipRequested = false; return; }
  typing = true;
  const entry = pendingQueue.shift();
  const div = document.createElement("div");
  div.className = "line" + (entry.className ? " " + entry.className : "");
  consoleOut.appendChild(div);

  if (entry.instant || skipRequested || entry.text.length === 0) {
    div.textContent = entry.text;
    scrollTop();
    pumpQueue();
    return;
  }

  let i = 0;
  const step = () => {
    if (skipRequested) {
      div.textContent = entry.text;
      scrollTop();
      pumpQueue();
      return;
    }
    i++;
    div.textContent = entry.text.slice(0, i);
    scrollTop();
    if (i < entry.text.length) {
      typeTimer = setTimeout(step, 14);
    } else {
      pumpQueue();
    }
  };
  step();
}

function skipTyping() { skipRequested = true; }

function clearConsole() {
  if (typeTimer) clearTimeout(typeTimer);
  pendingQueue = [];
  typing = false;
  skipRequested = false;
  consoleOut.innerHTML = "";
}

const engine = new GameEngine({ print });

function submit() {
  const raw = input.value;
  input.value = "";
  if (!raw.trim()) return;
  print("&gt; " + raw, true, "cmd");
  engine.handle(raw);
  updateUI();
}

input.addEventListener("keydown", (e) => {
  if (e.key === "Enter") {
    submit();
  } else if (e.key === "ArrowUp") {
    e.preventDefault();
    if (engine.history.length === 0) return;
    if (engine.historyIndex > 0) {
      engine.historyIndex--;
      input.value = engine.history[engine.historyIndex];
    }
  } else if (e.key === "ArrowDown") {
    e.preventDefault();
    if (engine.historyIndex < engine.history.length - 1) {
      engine.historyIndex++;
      input.value = engine.history[engine.historyIndex];
    } else {
      engine.historyIndex = engine.history.length;
      input.value = "";
    }
  }
});

consoleWrap.addEventListener("click", (e) => {
  if (e.target === input) return;
  input.focus();
  if (pendingQueue.length || typing) skipTyping();
});

document.getElementById("skip-btn").addEventListener("click", (e) => {
  e.stopPropagation();
  skipTyping();
  input.focus();
});

document.getElementById("reset-btn").addEventListener("click", () => {
  clearConsole();
  engine.reset();
  updateUI();
  engine.intro();
  input.value = "";
  input.focus();
});
