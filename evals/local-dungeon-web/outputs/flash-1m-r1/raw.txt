Typewriter:

```js
processQueue(){
  if(this.typing) return;
  if(!this.queue.length) return;
  this.typing=true;
  const {text,cls}=this.queue.shift();
  const div=document.createElement('div');
  div.className='line '+cls;
  this.output.appendChild(div);
  let i=0;
  this.typeTimer=setInterval(()=>{
    if(this.skipRequested){ div.textContent=text; this.finishType(); return; }
    div.textContent=text.slice(0,++i);
    this.scrollBottom();
    if(i>=text.length) this.finishType();
  }, this.speed);
}
finishType(){ clearInterval(this.typeTimer); this.typing=false; this.skipRequested=false; this.scrollBottom(); this.processQueue(); }
skip(){ if(!this.typing) return; this.skipRequested=true; // flush queue immediately
  // flush all queued lines
  this.queue.forEach(q=>{const d=document.createElement('div'); d.className='line '+q.cls; d.textContent=q.text; this.output.appendChild(d);});
  this.queue=[];
  this.scrollBottom();
}
```

```html
  /* ---------------- State & Reset ---------------- */

  reset() {
    this.rooms = {};
    this.buildWorld();
    this.current = 'entrance';
    this.inventory = [];
    this.visited = new Set(['entrance']);
    this.flags = { lever_pulled:false, armory_unlocked:false, garden_announced:false };
    this.moves = 0;
    this.itemsFound = 0;

    this.stopTyping();
    this.queue = [];
    this.typing = false;
    this.output.innerHTML = '';

    this.line('THE LOCAL DUNGEON — v1.0', 'room');
    this.line('A single-file text adventure. Type HELP for commands.', 'sys');
    this.line('');
    this.enterRoom(true);
    this.updateUI();
    this.input.focus();
  }

  stopTyping() {
    if (this.typeTimer) { clearInterval(this.typeTimer); this.typeTimer = null; }
  }

  /* ---------------- Output / Typewriter ---------------- */

  line(text, cls) {
    this.queue.push({ text: text == null ? '' : String(text), cls: cls || '' });
    this.processQueue();
  }

  processQueue() {
    if (this.typing) return;
    const job = this.queue.shift();
    if (!job) return;
    this.typing = true;
    this.typeDiv = document.createElement('div');
    this.typeDiv.className = 'line' + (job.cls ? ' ' + job.cls : '');
    this.typeFull = job.text;
    this.typeIdx = 0;
    this.output.appendChild(this.typeDiv);
    if (!this.typeFull.length) { this.finishTyping(); return; }
    this.typeTimer = setInterval(() => {
      this.typeIdx++;
      this.typeDiv.textContent = this.typeFull.slice(0, this.typeIdx);
      this.scrollBottom();
      if (this.typeIdx >= this.typeFull.length) this.finishTyping();
    }, this.speed);
  }

  finishTyping() {
    this.stopTyping();
    this.typeDiv.textContent = this.typeFull;
    this.typing = false;
    this.scrollBottom();
    this.processQueue();
  }

  skip() {
    if (this.typing) {
      this.stopTyping();
      this.typeDiv.textContent = this.typeFull;
      this.typing = false;
    }
    while (this.queue.length) {
      const job = this.queue.shift();
      const d = document.createElement('div');
      d.className = 'line' + (job.cls ? ' ' + job.cls : '');
      d.textContent = job.text;
      this.output.appendChild(d);
    }
    this.scrollBottom();
  }

  scrollBottom() { this.output.scrollTop = this.output.scrollHeight; }

  /* ---------------- Input Handling ---------------- */

  bindEvents() {
    this.input.addEventListener('keydown', (e) => {
      if (e.key === 'Enter') {
        this.submit();
      } else if (e.key === 'ArrowUp') {
        e.preventDefault();
        if (!this.history.length) return;
        if (this.histIndex === this.history.length) this.draft = this.input.value;
        this.histIndex = Math.max(0, this.histIndex - 1);
        this.input.value = this.history[this.histIndex];
      } else if (e.key === 'ArrowDown') {
        e.preventDefault();
        if (!this.history.length) return;
        this.histIndex = Math.min(this.history.length, this.histIndex + 1);
        this.input.value = this.histIndex === this.history.length ? this.draft : this.history[this.histIndex];
      }
    });

    this.skipBtn.addEventListener('click', (e) => {
      e.stopPropagation();
      this.skip();
      this.input.focus();
    });

    this.console.addEventListener('click', (e) => {
      if (e.target.tagName === 'BUTTON') return;
      if (this.typing || this.queue.length) { this.skip(); }
      this.input.focus();
    });

    this.output.addEventListener('click', () => {
      if (this.typing || this.queue.length) this.skip();
    });

    this.resetBtn.addEventListener('click', () => this.reset());
  }

  submit() {
    if (this.typing || this.queue.length) this.skip();
    const raw = this.input.value.trim();
    this.input.value = '';
    if (!raw) return;
    this.history.push(raw);
    this.histIndex = this.history.length;
    this.draft = '';
    this.line('> ' + raw, 'cmd');
    this.execute(raw);
    this.updateUI();
  }

  /* ---------------- Command Parser ---------------- */

  execute(raw) {
    let input = raw.toLowerCase().replace(/\s+/g, ' ').trim();

    // Bare direction words / prefixes → treat as movement.
    const first = input.split(' ')[0];
    const dirOnly = this.resolveDirection(input);
    if (dirOnly) { this.doGo(dirOnly); return; }
    if (this.resolveDirection(first) && input.split(' ').length === 1) {
      this.doGo(this.resolveDirection(first)); return;
    }

    // Multi-word phrase normalization ("pick up" → "take", "look at" → "examine")
    for (const [from, to] of this.MULTI) {
      if (input.startsWith(from)) { input = to + input.slice(from.length).trim(); break; }
    }

    const words = input.split(' ').filter(Boolean);
    const verbTok = words[0];
    let target = words.slice(1).join(' ').replace(/^(the|a|an)\s+/, '').trim();
    if (target === 'up' && verbTok === 'take') target = ''; // "take up" edge

    const verb = this.resolveVerb(verbTok);
    switch (verb) {
      case 'go': {
        const dir = this.resolveDirection(target);
        if (!dir) { target ? this.line(`I don't know the direction "${target}". Try north, south, east, or west.`, 'err')
                           : this.line('Go where? Name a direction: north, south, east, west.', 'err'); }
        else this.doGo(dir);
        break;
      }
      case 'look':
        target ? this.doLookAt(target) : this.doLook();
        break;
      case 'read':
        if (!target) this.line('Read what?', 'err');
        else this.doLookAt(target);
        break;
      case 'take':
        target ? this.doTake(target) : this.line('Take what?', 'err');
        break;
      case 'drop':
        target ? this.doDrop(target) : this.line('Drop what?', 'err');
        break;
      case 'use':
        if (!target || target === 'lever') this.doUseLever();
        else if (target.includes('key')) this.doUseKey();
        else this.line(`There's no obvious way to use the ${target} around here.`, 'err');
        break;
      case 'inventory':
        this.doInventory();
        break;
      case 'help':
        this.doHelp();
        break;
      default:
        this.line(`Unknown command: "${verbTok}". Type HELP to see what this dungeon understands.`, 'err');
    }
  }

  resolveVerb(word) {
    for (const [verb, syns] of Object.entries(this.VERBS)) {
      if (syns.includes(word)) return verb;
    }
    // Prefix match on longer synonyms (avoids 's' matching 'see')
    for (const [verb, syns] of Object.entries(this.VERBS)) {
      for (const s of syns) {
        if (s.length >= 4 && (word.startsWith(s) || (s.startsWith(word) && word.length >= 3))) return verb;
      }
    }
    return null;
  }

  resolveDirection(text) {
    if (!text) return null;
    const t = text.trim();
    const map = { n:'north', s:'south', e:'east', w:'west', north:'north', south:'south', east:'east', west:'west' };
    if (map[t]) return map[t];
    const full = ['north','south','east','west'].filter(d => d.startsWith(t));
    return full.length === 1 ? full[0] : null;
  }

  matchesName(text, name) {
    if (!text || !name) return false;
    if (name.includes(text) || text.includes(name)) return true;
    return text.split(' ').some(w => name.split(' ').some(nw => nw.startsWith(w) && w.length >= 2));
  }

  findItemIn(list, text) {
    for (const it of list) if (this.matchesName(text, it.name)) return it;
    return null;
  }

  hasItem(id) { return this.inventory.some(i => i.id === id); }

  room() { return this.rooms[this.current]; }

  /* ---------------- Command Implementations ---------------- */

  exitLabel(dir) {
    const r = this.room();
    if (r.id === 'library' && dir === 'west' && !this.flags.lever_pulled) return `${dir} (blocked)`;
    if (r.id === 'armory' && dir === 'north' && !this.flags.armory_unlocked) return `${dir} (locked)`;
    return dir;
  }

  doLook() {
    const r = this.room();
    this.line(`— ${r.name.toUpperCase()} —`, 'room');
    this.line(r.desc);
    this.printExits();
  }

  printExits() {
    const dirs = Object.keys(this.room().exits);
    if (!dirs.length) { this.line('There appear to be no exits.', 'sys'); return; }
    this.line('Exits: ' + dirs.map(d => this.exitLabel(d)).join(', '), 'sys');
  }

  doLookAt(text) {
    const r = this.room();
    if (/room|here|surroundings|around|self|myself/.test(text) && text.length < 9) { this.doLook(); return; }
    if (text.includes('lever') && r.id === 'library') { this.line(r.observables.lever); return; }
    if (text.includes('door') && r.id === 'armory') {
      this.line(r.observables.door);
      this.line(this.flags.armory_unlocked ? 'The lock hangs open now — the door stands ajar.' : 'It is firmly locked. The keyhole is crusted with rust.', 'sys');
      return;
    }
    if (text.includes('bookshelf') && r.id === 'library') {
      this.line(r.observables.bookshelf);
      if (this.flags.lever_pulled) this.line('The shelf has swung aside, revealing the passage to the west.', 'sys');
      return;
    }
    if (text.includes('passage') && r.id === 'library') {
      this.line(this.flags.lever_pulled ? 'A narrow passage breathes cool air from behind the swung-open bookshelf. West, if you dare.'
                                        : 'The west wall looks solid — except for that suspiciously enormous bookshelf.', 'sys');
      return;
    }
    const roomItem = this.findItemIn(r.items, text);
    if (roomItem) { this.line(roomItem.desc); return; }
    const held = this.findItemIn(this.inventory, text);
    if (held) { this.line(held.desc); return; }
    for (const [key, val] of Object.entries(r.observables)) {
      if (this.matchesName(text, key)) { this.line(val); return; }
    }
    this.line(`You squint at the ${text}, but it reveals nothing useful. Perhaps try another name.`, 'err');
  }

  doTake(text) {
    const r = this.room();
    if (this.findItemIn(this.inventory, text)) { this.line(`You already have the ${text}. Greed is unbecoming.`, 'err'); return; }
    const it = this.findItemIn(r.items, text);
    if (it) {
      r.items = r.items.filter(x => x !== it);
      this.inventory.push(it);
      this.itemsFound++;
      this.line(`You take the ${it.name}.`);
      if (it.id === 'key') this.line('It is oddly warm in your palm. You resolve not to think about that.', 'sys');
      if (it.id === 'torch') this.line('The amber flame steadies, as if glad of the company.', 'sys');
      return;
    }
    if (text.includes('lever') || text.includes('door') || text.includes('bookshelf') || text.includes('wall')) {
      this.line(`That is bolted to the dungeon itself. You respect the ambition, but no.`, 'err');
      return;
    }
    this.line(`You don't see any "${text}" here to take. Try LOOK to check your surroundings.`, 'err');
  }

  doDrop(text) {
    const it = this.findItemIn(this.inventory, text);
    if (!it) { this.line(`You aren't carrying any "${text}".`, 'err'); return; }
    this.inventory = this.inventory.filter(x => x !== it);
    this.room().items.push(it);
    this.line(`You drop the ${it.name}. It lands with a sound that suggests it will be judged for this later.`);
  }

  doUseLever() {
    if (this.current !== 'library') { this.line('There is no lever here. The library to the north of the Entrance Hall had one.', 'err'); return; }
    if (this.flags.lever_pulled) { this.line('The lever is already thrown. The bookshelf stays politely swung open.', 'err'); return; }
    this.flags.lever_pulled = true;
    this.line('You heave the brass lever. Somewhere inside the wall, three centuries of dust rearrange themselves.');
    this.line('With a grinding groan, the great oak bookshelf swings outward, revealing a hidden passage to the WEST!', 'win');
  }

  doUseKey() {
    if (!this.hasItem('key')) { this.line('You have no key to use. The Entrance Hall had a rusted one, you think.', 'err'); return; }
    if (this.current === 'armory') {
      if (this.flags.armory_unlocked) { this.line('The iron door is already unlocked. Save the key for emotional support.', 'err'); }
      else { this.line('You slide the rusted key into the iron door. Try GO NORTH and stop fiddling.', 'sys'); }
      return;
    }
    this.line('You jiggle the key theatrically. Nothing here fits it. The armory door to the east might, though.');
  }

  doInventory() {
    if (!this.inventory.length) { this.line('Your inventory is empty. Your pockets echo.', 'sys'); return; }
    this.line('You are carrying:');
    this.inventory.forEach(it => this.line('  • ' + it.name));
  }

  doHelp() {
    this.line('COMMANDS:', 'room');
    this.line('  look / l            Study the room. "look at [thing]" to inspect.');
    this.line('  examine [thing]     Same, but with more syllables.');
    this.line('  go [direction]      Travel n / s / e / w (or just type the direction).');
    this.line('  take [item]         Pick up something (grab / get also work).');
    this.line('  drop [item]         Put something back.');
    this.line('  use [thing]         Operate mechanisms — pull the library lever...');
    this.line('  inventory / inv     Check what you carry.');
    this.line('  Up/Down arrows cycle command history; click Skip to speed up text.');
  }

  /* ---------------- Movement ---------------- */

  doGo(dir) {
    const r = this.room();
    const dest = r.exits[dir];
    if (!dest) { this.line(`There is no exit to the ${dir}. The dungeon shakes its head.`, 'err'); return; }

    if (r.id === 'library' && dir === 'west' && !this.flags.lever_pulled) {
      this.line('The great oak bookshelf blocks the west wall, immovable and faintly smug. Perhaps a lever nearby?', 'err');
      return;
    }
    if (r.id === 'armory' && dir === 'north' && !this.flags.armory_unlocked) {
      if (this.hasItem('key')) {
        this.flags.armory_unlocked = true;
        this.line('You work the rusted key into the iron lock. It resists, then surrenders with a reluctant CLUNK.');
        this.line('The heavy door swings inward, exhaling cold treasure-scented air.', 'win');
      } else {
        this.line('A heavy iron door bars the north passage, locked fast. The keyhole is crusted with rust — you need a key.', 'err');
        return;
      }
    }

    this.current = dest;
    this.moves++;
    const firstVisit = !this.visited.has(dest);
    this.visited.add(dest);
    this.enterRoom(firstVisit);
  }

  enterRoom(firstVisit) {
    const r = this.room();
    this.line('');
    this.line(`— ${r.name.toUpperCase()} —`, 'room');
    this.line(firstVisit ? r.desc : 'You are back in the ' + r.name + '. The ' + r.desc.split('.')[0].toLowerCase().replace(/^you stand in a /, 'you still stand in a ') + '.');
    if (r.id === 'library' && this.flags.lever_pulled) this.line('The bookshelf still stands swung open; the western passage breathes cool air.', 'sys');
    if (r.id === 'armory' && this.flags.armory_unlocked) this.line('The unlocked iron door to the north stands ajar.', 'sys');
    if (firstVisit && r.id === 'garden' && !this.flags.garden_announced) {
      this.flags.garden_announced = true;
      this.line('★ You have reached the Moonlit Garden — the peaceful heart beneath the mill. ★', 'win');
      this.line('You sink onto a mossy bench. Somewhere above, the town sleeps on, never knowing how close you came to total relaxation.', 'win');
      this.line('This is a gentle ending to your descent — but the dungeon keeps its other secrets for another night.', 'sys');
    }
    if (firstVisit && r.id === 'treasure') this.line('Something glints in the center of the room. Your hands are already reaching.', 'sys');
    if (firstVisit && r.id === 'secret') this.line('You have found a room that officially does not exist. Tell no one. Tell everyone.', 'sys');
    if (r.items.length) this.line('You notice: ' + r.items.map(i => i.name).join(', ') + ' here.', 'sys');
    this.printExits();
  }

  /* ---------------- Sidebar UI ---------------- */

  updateUI() {
    this.elMoves.textContent = this.moves;
    this.elItems.textContent = this.itemsFound;

    this.elInv.innerHTML = '';
    this.elEmpty.style.display = this.inventory.length ? 'none' : 'block';
    for (const it of this.inventory) {
      const li = document.createElement('li');
      li.innerHTML = this.itemIcon(it.id) + '<span></span>';
      li.querySelector('span').textContent = it.name;
      this.elInv.appendChild(li);
    }

    let hint;
    if (this.flags.garden_announced) {
      hint = 'You found the Garden. The dungeon tips its mop to you. Try <strong style="color:var(--green)">Reset Game</strong> for a fresh run — or hunt the coin and scroll you may have missed.';
    } else if (!this.flags.lever_pulled && this.visited.has('library')) {
      hint = 'That brass lever in the Library looks pull-worthy. <strong style="color:var(--green)">use lever</strong>';
    } else if (this.flags.lever_pulled && !this.visited.has('secret')) {
      hint = 'The west passage in the Library is open. <strong style="color:var(--green)">go west</strong>';
    } else if (!this.flags.armory_unlocked && this.visited.has('armory') && !this.hasItem('key')) {
      hint = 'The iron door needs a key. Something rusty was lying around the Entrance Hall...';
    } else if (this.flags.armory_unlocked && !this.visited.has('treasure')) {
      hint = 'The north door stands open. Treasure awaits. <strong style="color:var(--green)">go north</strong>';
    } else if (!this.visited.has('garden')) {
      hint = 'Three ways lead from the Entrance Hall — north, east, and somewhere softer, to the south.';
    } else {
      hint = 'Keep exploring. <strong style="color:var(--green)">look</strong> often, <strong style="color:var(--green)">take</strong> everything not nailed down.';
    }
    this.elHint.innerHTML = hint;
  }

  itemIcon(id) {
    switch (id) {
      case 'key':
        return '<svg width="15" height="15" viewBox="0 0 24 24" fill="none"><circle cx="8" cy="8" r="4.5" stroke="#ffb000" stroke-width="2"/><path d="M11.5 11.5L21 21M17 17l2.5-2.5M14 20l2-2" stroke="#ffb000" stroke-width="2" stroke-linecap="round"/></svg>';
      case 'torch':
        return '<svg width="15" height="15" viewBox="0 0 24 24" fill="none"><path d="M12 2c1.6 2.4 3.6 3.8 3.6 6.4A3.6 3.6 0 0 1 12 12a3.6 3.6 0 0 1-3.6-3.6C8.4 5.8 10.4 4.4 12 2z" fill="#ffb000"/><rect x="11" y="12" width="2" height="10" rx="1" fill="#8a5a2b"/></svg>';
      case 'coin':
        return '<svg width="15" height="15" viewBox="0 0 24 24" fill="none"><circle cx="12" cy="12" r="9" stroke="#ffd700" stroke-width="2"/><circle cx="12" cy="12" r="5" stroke="#ffd700" stroke-width="2"/></svg>';
      case 'scroll':
        return '<svg width="15" height="15" viewBox="0 0 24 24" fill="none"><path d="M7 3h11a2 2 0 0 1 2 2v14a2 2 0 0 1-2 2H7" stroke="#55ffff" stroke-width="2" stroke-linejoin="round"/><path d="M7 3a2 2 0 0 0-2 2v14a2 2 0 0 0 2 2" stroke="#55ffff" stroke-width="2"/><path d="M10 8h7M10 12h7M10 16h4" stroke="#55ffff" stroke-width="1.6" stroke-linecap="round"/></svg>';
      default:
        return '<svg width="15" height="15" viewBox="0 0 24 24"><circle cx="12" cy="12" r="8" fill="none" stroke="#00ff00" stroke-width="2"/></svg>';
    }
  }
}
```
