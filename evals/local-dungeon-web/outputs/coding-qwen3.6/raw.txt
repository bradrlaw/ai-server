// ─── Item Class ──────────────────────────────────────────────
class Item {
  constructor(name, description, examineText, isKey = false) {
    this.name = name;
    this.description = description;
    this.examineText = examineText;
    this.isKey = isKey;
  }
}

// ─── Room Class ──────────────────────────────────────────────
class Room {
  constructor(id, name, description, exits = {}, items = [], searchDescription = '') {
    this.id = id;
    this.name = name;
    this.description = description;
    this.exits = { ...exits };
    this.items = items;
    this.searchDescription = searchDescription;
    this.searched = false;
  }

  hasItem(name) {
    return this.items.some(i => i.name.toLowerCase() === name.toLowerCase());
  }

  findItem(name) {
    return this.items.find(i => i.name.toLowerCase() === name.toLowerCase());
  }

  removeItem(name) {
    const idx = this.items.findIndex(i => i.name.toLowerCase() === name.toLowerCase());
    if (idx !== -1) {
      this.items.splice(idx, 1);
      return true;
    }
    return false;
  }
}

          return;
        }
        if (i < text.length) {
          line.textContent = text.substring(0, i + 1);
          i++;
          this.autoScroll();
          setTimeout(type, speed);
        } else {
          this.isTyping = false;
          this.skipBtn.classList.remove('visible');
          this.autoScroll();
          resolve();
        }
      };
      type();
    });
  }

    if (this.typewriterActive) {
      this.typewriterActive = false;
    }
  }

  // ── Command Processing ─────────────────────────────────────
  async processCommand(rawInput) {
    // Echo the command
    this.printInstant('&gt; ' + rawInput, 'system');

    const input = rawInput.toLowerCase().trim();
    const words = input.split(/\s+/);
    const verb = words[0];
    const noun = words.slice(1).join(' ');

    // Resolve verb synonyms
    const resolvedVerb = this.resolveVerb(verb);
    const resolvedNoun = this.resolveNoun(noun);

    switch (resolvedVerb) {
      case 'look':
        await this.cmdLook(noun);
        break;
      case 'go':
        await this.cmdGo(resolvedNoun);
        break;
      case 'take':
        await this.cmdTake(resolvedNoun);
        break;
      case 'inventory':
        this.cmdInventory();
        break;
      case 'use':
        await this.cmdUse(resolvedNoun);
        break;
      case 'examine':
        await this.cmdExamine(resolvedNoun);
        break;
      case 'search':
        await this.cmdSearch();
        break;
      case 'help':
        this.cmdHelp();
        break;
      default:
        await this.print(`I don't understand "${verb}". Type "help" for a list of commands.`, 'error');
    }

    this.updateUI();
    this.inputEl.focus();
  }

  resolveVerb(verb) {
    const map = {
      'look': 'look', 'l': 'look', 'x': 'examine', 'examine': 'examine', 'inspect': 'examine', 'check': 'examine',
      'go': 'go', 'move': 'go', 'walk': 'go', 'travel': 'go', 'goto': 'go',
      'north': 'go', 'n': 'go', 'south': 'go', 's': 'go', 'east': 'go', 'e': 'go', 'west': 'go', 'w': 'go',
      'take': 'take', 'get': 'take', 'grab': 'take', 'pick': 'take', 'acquire': 'take', 'grab': 'take',
      'i': 'inventory', 'inv': 'inventory', 'inventory': 'inventory', 'bag': 'inventory', 'stock': 'inventory',
      'use': 'use', 'apply': 'use', 'activate': 'use', 'pull': 'use', 'push': 'use', 'turn': 'use',
      'search': 'search', 'find': 'search', 'explore': 'search',
      'help': 'help', 'h': 'help', '?': 'help', 'commands': 'help',
    };
    return map[verb] || verb;
  }

  resolveNoun(noun) {
    const dirMap = { 'n': 'north', 's': 'south', 'e': 'east', 'w': 'west', 'north': 'north', 'south': 'south', 'east': 'east', 'west': 'west' };
    if (dirMap[noun]) return dirMap[noun];
    return noun;
  }

  async cmdLook(noun) {

    } else {

      await this.print(room.description, 'narrative');

    const exitNames = Object.keys(room.exits);
    if (exitNames.length > 0) {
      await this.print(`Exits: ${exitNames.map(e => e.charAt(0).toUpperCase() + e.slice(1)).join(', ')}`, 'system');
    }

    // Show items
    if (room.items.length > 0) {
      const itemNames = room.items.map(i => `<span class="item-name">${i.name}</span>`).join(', ');
      await this.print(`You can see: ${itemNames}`, 'narrative');

    if (room.id === 'armory' && !this.flags.keyUsed) {
      await this.print('The heavy iron door to the north is locked with a rusted padlock.', 'narrative');

      await this.print('The heavy iron door to the north is now open.', 'narrative');

    if (room.id === 'library' && this.flags.leverPulled) {
      await this.print('The lever has been pulled. A section of the wall to the west appears to have shifted.', 'narrative');
    }
  }

  async cmdGo(direction) {
    const room = this.currentRoom;
    const targetId = room.exits[direction];

    if (!targetId) {
      await this.print(`You can't go ${direction} from here.`, 'error');
      return;
    }

    // Check for locked doors
    if (room.id === 'armory' && direction === 'north' && !this.flags.keyUsed) {
      await this.print('The iron door to the north is locked tight. It needs a key to open.', 'error');
      return;
    }

    // Check for secret passage
    if (room.id === 'library' && direction === 'west' && !this.flags.leverPulled) {
      await this.print('There is nothing to the west—just a solid stone wall.', 'error');
      return;
    }

    this.moves++;
    const targetRoom = this.rooms.get(targetId);
    this.currentRoom = targetRoom;
    this.visitedRooms.add(targetRoom.id);

    if (!this.visitedRooms.has(targetRoom.id)) {
      await this.print(`\n${targetRoom.name}`, 'system');
      await this.print(targetRoom.description, 'narrative');
      this.visitedRooms.add(targetRoom.id);
    } else {
      await this.print(`\nYou head ${direction} and arrive at the ${targetRoom.name}.`, 'system');
    }

    // Show items
    if (targetRoom.items.length > 0) {
      const itemNames = targetRoom.items.map(i => `<span class="item-name">${i.name}</span>`).join(', ');
      await this.print(`You can see: ${itemNames}`, 'narrative');
    }

    // Special notes
    if (targetRoom.id === 'armory' && !this.flags.keyUsed) {
      await this.print('The heavy iron door to the north is locked with a rusted padlock.', 'narrative');
    }
    if (targetRoom.id === 'garden') {
      await this.print('\nYou feel a sense of peace wash over you. Perhaps this is where your journey ends—or where a new one begins.', 'success');
    }
  }

  async cmdTake(itemName) {
