<script>
"use strict";

/* ============================ Data Model ============================ */

class Item {
  constructor(id, name, description) {
    this.id = id;               // canonical short id, e.g. "key"
    this.name = name;           // display name, e.g. "rusted key"
    this.description = description;
  }
}

class Room {
  constructor(id, name, description, exits = {}) {
    this.id = id;
    this.name = name;
    this.description = description;
    this.exits = exits;         // { north: { to: "library" } | { to:"treasure", locked: true, key: "key" } }
    this.items = [];            // Item objects on the floor
    this.features = {};         // id -> { name, description, interactive }
  }
}

/* ============================ Game Engine ============================ */

class GameEngine {
  constructor(ui) {
    this.ui = ui;
    this.reset();
  }

  buildWorld() {
    const I = (a, b, c) => new Item(a, b, c);
    const rooms = {};

    rooms.entrance = new Room("entrance", "Entrance Hall",
      "You stand in a low-ceilinged Entrance Hall. Cobwebs drape a rotted portcullis, and a draft sneaks in from somewhere south, carrying the smell of wet soil and rosemary. A stairway climbs NORTH into shadow. A mossy archway opens EAST, where old weapon racks rattle.",
      { north: { to: "library" }, east: { to: "armory" }, south: { to: "garden" } });
    rooms.entrance.items.push(I("key", "rusted key",
      "A stubby iron key, orange with rust. Its bow is shaped like a snarling boar — the crest of the tavern above. It looks old enough to snap, but probably holds for one more lock."));

    rooms.library = new Room("library", "Drowned Library",
      "The Drowned Library. Shelves lean drunkenly around you, most of their books long since eaten by damp. One iron-bound tome remains on a lectern, chained shut. A heavy brass LEVER is bolted into the wall behind it. The way back is SOUTH.",
      { south: { to: "entrance" } });
    rooms.library.features.lever = {
      name: "brass lever",
      description: "A thick brass lever, green with verdigris, mounted on a plate engraved: 'PULL FOR QUIET COMPANY.' The chain running from it vanishes into the masonry.",
      interactive: true
    };

    rooms.armory = new Room("armory", "Ruined Armory",
      "The Ruined Armory. Rusted spear butts and dented bucklers litter the floor like discarded dinner plates. A wall torch sits in an empty bracket, dry and unused. A reinforced oak door with a boar-crested LOCK stands to the NORTH.",
      { west: { to: "entrance" }, north: { to: "treasure", locked: true, key: "key" } });
    rooms.armory.items.push(I("torch", "torch",
      "A pitch-soaked torch, still dry inside its bracket. The pitch smells sharp and fresh — whoever stocked this dungeon expected guests. It would burn for hours if lit."));

    rooms.treasure = new Room("treasure", "Treasure Vault",
      "The Treasure Vault. Most of it is a lie: rotted chests full of lead ingots the wizard used as ballast. But rolled into the far alcove, glinting with utter lack of shame, is a single real GOLDEN COIN — the wizard's 'lucky' coin, worth more than the tavern above.",
      { south: { to: "armory" } });
    rooms.treasure.items.push(I("coin", "golden coin",
      "A heavy golden coin stamped with a boar and the date 1123. Rubbing the grime away reveals a tiny inscription on the rim: 'HE WHO SPENDS THIS SHALL BE POOR BUT HAPPY.' You decide to keep it unspent."));

    rooms.secret = new Room("secret", "Secret Chamber",
      "A hidden chamber, no bigger than a wine cellar, revealed behind the library wall. Dust motes dance in your lantern light. On a stone pedestal, preserved by sealed air, rests an ANCIENT SCROLL — the wizard's own memoirs, apparently. Page one reads: 'I regret everything.'",
      { east: { to: "library" } });
    rooms.secret.items.push(I("scroll", "ancient scroll",
      "The wizard's sealed memoir, 'A Thousand Regrets, Abbreviated.' The final entry describes, in loving detail, a golden coin he left in the armory vault 'where no adventurer would think to look past the lead.' He clearly never met adventurers."));

    rooms.garden = new Room("garden", "Moonlit Garden",
      "You emerge into a walled Moonlit Garden — the dungeon's one act of gentleness. A dry stone bench faces a pond where lantern-fish drift like submerged stars. Somewhere, water trickles. The dungeon entrance is NORTH, and honestly, you've done enough adventuring for one evening.",
      { north: { to: "entrance" } });

    return rooms;
  }

  reset() {
    this.rooms = this.buildWorld();
    this.currentRoomId = "entrance";
    this.inventory = [];
    this.visited = new Set();
    this.flags = { lever_pulled: false };
    this.moves = 0;
    this.itemsFound = 0;
    this.history = [];
    this.historyIndex = -1;
    this.visited.add(this.currentRoomId);
  }

  get room() { return this.rooms[this.currentRoomId]; }

  /* ---------- Typewriter output (delegates to UI) ---------- */
  out(text, cls)  { this.ui.print(text, cls); }
  roomDesc()      { return this.room.description; }

  /* ---------- Command handling ---------- */
  handle(raw) {
    const text = raw.trim();
    if (!text) return;
    this.history.push(text);
    this.historyIndex = this.history.length;
    this.ui.echo(text);

    const tokens = text.toLowerCase().split(/\s+/);
    const verb = this.resolveVerb(tokens[0]);
    const rest = tokens.slice(1).join(" ");

    switch (verb) {
      case "look":     return this.cmdLook(rest);
      case "go":       return this.cmdGo(rest || tokens.slice(0).join(" "));
      case "take":     return this.cmdTake(rest);
      case "use":      return this.cmdUse(rest);
      case "inventory":return this.cmdInventory();
      case "help":     return this.cmdHelp();
      case "direction":return this.cmdGo(text.toLowerCase());
      default:
        this.out(`I don't know the command "${tokens[0]}". Type "help" to see what works down here.`, "err");
    }
  }

  resolveVerb(word) {
    const map = {
      look: "look", l: "look", inspect: "look", survey: "look",
      examine: "look", x: "look", check: "look",
      go: "go", move: "go", walk: "go", travel: "go", head: "go", g: "go",
      n: "direction", s: "direction", e: "direction", w: "direction",
      north: "direction", south: "direction", east: "direction", west: "direction",
      take: "take", grab: "take", get: "take", pick: "take", steal: "take",
      use: "use", pull: "use", activate: "use", operate: "use", unlock: "use", open: "use",
      inventory: "inventory", inv: "inventory", i: "inventory", items: "inventory", bag: "inventory",
      help: "help", commands: "help", "?": "help",
    };
    return map[word] || word;
  }

  resolveDirection(word) {
    const map = { n: "north", north: "north", s: "south", south: "south",
                  e: "east", east: "east", w: "west", west: "west" };
    return map[word] || null;
  }

  /* Find an item by partial name in a list */
  findByName(query, list) {
    if (!query) return null;
    const q = query.replace(/^(the|a|an)\s+/, "").trim();
    if (!q) return null;
    const words = q.split(/\s+/);
    let best = null, bestScore = 0;
    for (const it of list) {
      const name = it.name.toLowerCase();
      let score = 0;
      if (name === q) score = 100;
      else if (name.includes(q)) score = 50;
      else if (words.every(w => name.includes(w))) score = 30;
      if (score > bestScore) { best = it; bestScore = score; }
    }
    return best;
  }

  /* ---------- Commands ---------- */

  cmdLook(rest) {
    if (rest) {
      const q = rest.replace(/^at\s+/, "");
      // Features in room
      const featureList = Object.values(this.room.features)
        .map((f, i) => ({ id: Object.keys(this.room.features)[Object.values(this.room.features).indexOf(f)], name: f.name, description: f.description }));
      const feat = this.findByName(q, featureList);
      if (feat) { this.out(feat.description, "flavor"); return; }
      // Items in room
      const item = this.findByName(q, this.room.items);
      if (item) { this.out(item.description, "flavor"); return; }
      // Items in inventory
      const held = this.findByName(q, this.inventory);
      if (held) { this.out(held.description, "flavor"); return; }
      // Exits?
      const dir = this.resolveDirection(q);
      if (dir) {
        const exit = this.room.exits[dir];
        if (exit) {
          if (exit.hidden && !this.flags.lever_pulled) {
            this.out("You sense a draft from that direction, but see no doorway.", "flavor");
          } else {
            this.out(`The ${dir} passage leads ${exit.locked && !exit.open ? "to a locked door." : "onward."}`, "flavor");
          }
        } else this.out(`There is no way ${dir} from here.`, "flavor");
        return;
      }
      this.out(`You squint at "${q}" but it's not something you can make out down here.`, "err");
      return;
    }
    this.describeRoom(true);
  }

  cmdGo(rest) {
    if (!rest) { this.out("Go where? Specify a direction: north, south, east, or west.", "err"); return; }
    const dir = this.resolveDirection(rest.split(/\s+/).pop());
    if (!dir) { this.out(`"${rest}" isn't a direction. Try north, south, east, or west.`, "err"); return; }

    const exit = this.room.exits[dir];
    if (!exit) {
      this.out(`You bump into a solid wall to the ${dir}. The dungeon says no.`, "err");
      return;
    }
    if (exit.locked && !exit.open) {
      const key = this.inventory.find(i => i.id === exit.key);
      if (key) {
        exit.open = true;
        this.moves++;
        this.out(`You work the ${key.name} into the boar-crested lock. It resists, complains loudly, then turns with a satisfying CLUNK. The oak door swings open.`, "sys");
        this.moveTo(exit.to);
      } else {
        this.out(`The reinforced door is locked fast. The keyhole is shaped for a boar-crested key — perhaps one is lying around the dungeon.`, "err");
      }
      return;
    }
    this.moves++;
    this.moveTo(exit.to);
  }

  moveTo(roomId) {
    this.currentRoomId = roomId;
    const first = !this.visited.has(roomId);
    this.visited.add(roomId);
    if (first) this.out(`— You arrive in the ${this.room.name} for the first time. —`, "sys");
    this.describeRoom(false);
    if (roomId === "garden") {
      this.out("You sink onto the cold stone bench. Lantern-fish rise to meet you. Against all odds, the dungeon has given you something peaceful. THE END — or at least, the end of tonight. (Type 'help' if you can't resist exploring more.)", "sys");
    }
  }

  describeRoom(repeat) {
    this.out(this.room.description);
    const things = [
      ...this.room.items.map(i => i.name),
      ...Object.values(this.room.features).map(f => f.name)
    ];
    if (things.length) this.out(`You notice: ${things.join(", ")}.`, "flavor");
    const dirs = Object.entries(this.room.exits)
      .filter(([, e]) => !(e.hidden && !this.flags.lever_pulled))
      .map(([d]) => d);
    this.out(`Exits: ${dirs.join(", ")}.`, "flavor");
  }

  cmdTake(rest) {
    if (!rest) { this.out("Take what? Name an object, e.g. 'take key'.", "err"); return; }
    const item = this.findByName(rest, this.room.items);
    if (!item) {
      const inBag = this.findByName(rest, this.inventory);
      this.out(inBag ? `You already have the ${item?.name || rest} in your satchel.`
                     : `There's no "${rest}" here you can pick up.`, "err");
      return;
    }
    this.room.items = this.room.items.filter(i => i !== item);
    this.inventory.push(item);
    this.itemsFound++;
    this.out(`You pocket the ${item.name}. It's yours now, consequences and all.`, "sys");
    this.ui.updateSidebar();
  }

  cmdUse(rest) {
    if (!rest) { this.out("Use what? Try 'use lever' or 'use key'.", "err"); return; }

    // Lever in the library
    if (rest.includes("lever")) {
      if (!this.room.features.lever) { this.out("There's no lever here to use.", "err"); return; }
      if (this.flags.lever_pulled) { this.out("You jiggle the lever. It's already done its one trick.", "flavor"); return; }
      this.flags.lever_pulled = true;
      this.room.exits.west = { to: "secret" };
      this.moves++;
      this.out("You heave the brass lever. Somewhere inside the wall, a chain screams, counterweights crash, and a section of shelving grinds inward and pivots. A hidden doorway now leads WEST out of the library. The chained tome on the lectern falls open to a page that just says 'FINALLY.'", "sys");
      this.ui.updateSidebar();
      return;
    }

    // Key usage outside the door
    const keyItem = this.findByName(rest, this.inventory);
    if (keyItem && keyItem.id === "key") {
      this.out("You wave the rusted key around. The only lock it fits is on the armory's north door — go try it there.", "flavor");
      return;
    }
    if (this.findByName(rest, this.room.items)) {
      this.out(`The ${rest} doesn't seem usable on its own. Maybe take it first?`, "flavor");
      return;
    }
    this.out(`You can't figure out how to use "${rest}" — at least, not down here, not tonight.`, "err");
  }

  cmdInventory() {
    if (!this.inventory.length) { this.out("Your satchel is empty. Just lint, a button, and regret.", "flavor"); return; }
    this.out("In your satchel: " + this.inventory.map(i => i.name).join(", ") + ".", "sys");
  }

  cmdHelp() {
    this.out("Commands: look | examine <thing> | go <direction> (n/s/e/w) | take <item> | use <item> | inventory | help", "sys");
    this.out("Tip: partial names work — 'take key' grabs the rusted key, 'n' means go north.", "flavor");
  }
}

/* ============================ UI Layer ============================ */

class UI {
  constructor() {
    this.output   = document.getElementById("console-output");
    this.input    = document.getElementById("cmd-input");
    this.skipBtn  = document.getElementById("skip-btn");
    this.resetBtn = document.getElementById("reset-btn");
    this.movesEl  = document.getElementById("stat-moves");
    this.itemsEl  = document.getElementById("stat-items");
    this.invEl    = document.getElementById("inventory-list");
    this.roomEl   = document.getElementById("room-label");
    this.exitEl   = document.getElementById("exit-chips");

    this.queue = [];          // { text, cls }
    this.typing = false;
    this.skipFlag = false;
    this.currentP = null;
    this.typeTimer = null;
    this.engine = null;
  }

  bindEngine(engine) { this.engine = engine; }

  /* ---------- Typewriter ---------- */
  print(text, cls = "") {
    this.queue.push({ text, cls });
    if (!this.typing) this.runQueue();
  }

  runQueue() {
    if (!this.queue.length) { this.typing = false; this.removeCursor(); return; }
    this.typing = true;
    const { text, cls } = this.queue.shift();
    const p = document.createElement("p");
    if (cls) p.className = cls;
    this.output.appendChild(p);
    this.currentP = p;
    this.addCursor(p);
    let i = 0;
    const tick = () => {
      if (this.skipFlag) {
        p.textContent = text;
        this.finishNode(p);
        this.skipFlag = false;
        this.typeTimer = setTimeout(() => this.runQueue(), 10);
        this.scrollDown();
        return;
      }
      if (i < text.length) {
        this.insertChar(p, text[i]);
        i++;
        this.scrollDown();
        this.typeTimer = setTimeout(tick, text[i - 1] === "." ? 26 : 9);
      } else {
        this.finishNode(p);
        this.typeTimer = setTimeout(() => this.runQueue(), 90);
      }
    };
    tick();
  }

  addCursor(p) {
    this.removeCursor();
    const c = document.createElement("span");
    c.className = "cursor-blink";
    c.id = "type-cursor";
    p.appendChild(c);
  }
  insertChar(p, ch) {
    const cursor = document.getElementById("type-cursor");
    if (cursor) p.insertBefore(document.createTextNode(ch), cursor);
    else p.appendChild(document.createTextNode(ch));
  }
  finishNode(p) {
    const cursor = document.getElementById("type-cursor");
    if (cursor) cursor.remove();
  }
  removeCursor() {
    const c = document.getElementById("type-cursor");
    if (c) c.remove();
  }

  skip() {
    if (!this.typing) return;
    clearTimeout(this.typeTimer);
    // Finish current line instantly
    if (this.currentP) {
      this.skipFlag = true;
      this.runQueue(); // current tick fills it
    }
    // Flush remaining queue instantly
    this.skipFlag = true;
    while (this.queue.length) {
      const { text, cls } = this.queue.shift();
      const p = document.createElement("p");
      if (cls) p.className = cls;
      p.textContent = text;
      this.output.appendChild(p);
    }
    this.skipFlag = false;
    this.typing = false;
    this.removeCursor();
    this.scrollDown();
  }

  echo(text) { this.print("> " + text, "echo"); }

  scrollDown() { this.output.scrollTop = this.output.scrollHeight; }

  /* ---------- Sidebar ---------- */
  updateSidebar() {
    const e = this.engine;
    this.movesEl.textContent = e.moves;
    this.itemsEl.textContent = e.itemsFound;
    this.roomEl.textContent = e.room.name;

    this.exitEl.innerHTML = "";
    Object.entries(e.room.exits)
      .filter(([, ex]) => !(ex.hidden && !e.flags.lever_pulled))
      .forEach(([d]) => {
        const chip = document.createElement("span");
        chip.className = "chip";
        chip.textContent = d;
        this.exitEl.appendChild(chip);
      });

    this.invEl.innerHTML = "";
    if (!e.inventory.length) {
      const li = document.createElement("li");
      li.className = "empty";
      li.textContent = "Inventory is empty.";
      this.invEl.appendChild(li);
    } else {
      e.inventory.forEach(item => {
        const li = document.createElement("li");
        li.innerHTML = `<svg width="13" height="13" viewBox="0 0 24 24" fill="none" stroke="#ffb000" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><circle cx="8" cy="15" r="4"/><path d="M11 12L20 3M16 7l3 3M13 10l2 2"/></svg><span>${item.name}</span>`;
        this.invEl.appendChild(li);
      });
    }
  }

  /* ---------- Input ---------- */
  submit() {
    const val = this.input.value;
    if (!val.trim()) return;
    if (this.typing) this.skip();
    this.input.value = "";
    this.engine.handle(val);
    this.updateSidebar();
  }

  cycleHistory(delta) {
    const h = this.engine.history;
    if (!h.length) return;
    this.engine.historyIndex = Math.max(-1, Math.min(h.length - 1, this.engine.historyIndex + delta));
    this.input.value = this.engine.historyIndex === -1 ? "" : h[this.engine.historyIndex];
    if (this.engine.historyIndex === -1) this.input.value = "";
    else this.input.value = h[this.engine.historyIndex];
    // reposition caret at end
    requestAnimationFrame(() => this.input.setSelectionRange(this.input.value.length, this.input.value.length));
  }

  /* ---------- Boot / Reset ---------- */
  startIntro() {
    this.print("THE LOCAL DUNGEON — a text adventure", "sys");
    this.print("Deep beneath the village of Coppersmith Lane, a wizard's century-old hoard allegedly waits. Most of it is lead. One part of it is not.", "flavor");
    this.print("Type 'help' for commands, or just try 'look'.", "sys");
    this.engine.describeRoom(false);
    this.updateSidebar();
  }

  reset() {
    clearTimeout(this.typeTimer);
    this.queue = [];
    this.typing = false;
    this.removeCursor();
    this.output.innerHTML = "";
    this.engine.reset();
    this.input.value = "";
    this.startIntro();
    this.input.focus();
  }
}

/* ============================ Bootstrap ============================ */

const ui = new UI();
const engine = new GameEngine(ui);
ui.bindEngine(engine);
ui.startIntro();

ui.input.addEventListener("keydown", (ev) => {
  if (ev.key === "Enter") { ev.preventDefault(); ui.submit(); }
  else if (ev.key === "ArrowUp")   { ev.preventDefault(); ui.cycleHistory(-1); }
  else if (ev.key === "ArrowDown") { ev.preventDefault(); ui.cycleHistory(1); }
});

ui.skipBtn.addEventListener("click", (ev) => { ev.stopPropagation(); ui.skip(); });
ui.resetBtn.addEventListener("click", () => ui.reset());

// Clicking the console area focuses the input; clicking output also skips typing
document.getElementById("console-wrap").addEventListener("click", () => ui.input.focus());
ui.output.addEventListener("click", () => { if (ui.typing) ui.skip(); });

// Keep the input focused for keyboard-first play
document.addEventListener("keydown", (ev) => {
  if (document.activeElement !== ui.input && !ev.ctrlKey && !ev.metaKey && !ev.altKey
      && !["INPUT","BUTTON","SUMMARY"].includes(document.activeElement.tagName)) {
    ui.input.focus();
  }
});
</script>
