/* ============================================================
   ITEM
   ============================================================ */
class Item {
  constructor(name, description, keywords) {
    this.name = name;
    this.description = description;
    this.keywords = keywords || [name];
  }
  matches(text) {
    const t = text.trim();
    if (!t) return false;
    return this.keywords.some(k => k.startsWith(t) || t.startsWith(k));
  }
}

/* ============================================================
   ROOM
   ============================================================ */
class Room {
  constructor(id, name, description, exits) {
    this.id = id;
    this.name = name;
    this.description = description;
    this.exits = exits;        // { north: { to: 'library' }, ... }
    this.items = [];           // Item[]
  }
  findItem(text) {
    return this.items.find(i => i.matches(text)) || null;
  }
}

    this.print('═══════════════════════════════════════', 'title');
    this.print('Dust motes swirl in the trapdoor light behind you. It closes with a soft, final thud.', 'system');
    this.print('Type "help" at any time for a command list, or simply "look" to begin.', 'system');
    this.enterRoom(true);
    this.inputEl.value = '';
    this.inputEl.focus();
  }

  get room() { return this.rooms[this.currentRoomId]; }

  /* ---------- Entry / room description ---------- */
  enterRoom(firstTime) {
    const room = this.room;
    const fresh = !this.visited.has(room.id);
    this.visited.add(room.id);
    if (!firstTime) this.print('── ' + room.name.toUpperCase() + ' ──', 'title', true);
    else this.print('── ' + room.name.toUpperCase() + ' ──', 'title');
    this.print(room.description, 'normal');
    const here = room.items.map(i => i.name);
    if (here.length) this.print('You notice: ' + here.join(', ') + '.', 'system');
    this.print('Exits: ' + this.visibleExits().join(', ') + '.', 'system');

    if (room.id === 'garden' && !this.finished) {
      this.finished = true;
      this.print('', 'normal', true);
      this.print('★ YOU FOUND THE PEACEFUL ENDING! ★', 'win');
      this.print(`You settle onto the warm stone bench. With ${this.moves} moves and ${this.itemsFound} item(s) recovered, ` +
                 `you have bested The Local Dungeon without besting anyone. Well played.`, 'win');
      this.print('(The adventure continues if you like — the garden is always open. Or press "Reset Game" to start fresh.)', 'system');
      this.updateHint('🏆 Ending reached: "The Quiet Bench". Feel free to keep exploring!');
    }
  }

  visibleExits() {
    const list = [];
    for (const [dir, ex] of Object.entries(this.room.exits)) {
      if (ex.locked && !this.flags[ex.lockFlag]) list.push(dir + ' (blocked)');
      else list.push(dir);
    }
    return list.length ? list : ['none'];
  }

  /* ---------- Typewriter output ---------- */
  print(text, cls = 'normal', instant = false) {
    const div = document.createElement('div');
    div.className = 'line ' + cls;
    this.outEl.appendChild(div);
    if (instant || this.finished === 'skipAll') {
      div.textContent = text;
    } else {
      this.typeQueue.push({ div, text, i: 0 });
      if (!this.typing) this.tick();
    }
    this.autoScroll();
  }

  tick() {
    this.typing = true;
    this.skipBtn.style.display = 'inline-block';
    const step = () => {
      const head = this.typeQueue[0];
      if (!head) {
        this.typing = false;
        this.skipBtn.style.display = 'none';
        return;
      }
      if (head.i < head.text.length) {
        head.div.textContent += head.text[head.i++];
        this.autoScroll();
        this.typeTimer = setTimeout(step, this.TYPE_SPEED);
      } else {
        this.typeQueue.shift();
        this.typeTimer = setTimeout(step, 0);
      }
    };
    step();
  }

  skipTyping() {
    clearTimeout(this.typeTimer);
    while (this.typeQueue.length) {
      const head = this.typeQueue.shift();
      head.div.textContent = head.text;
    }
    this.typing = false;
    this.skipBtn.style.display = 'none';
    this.autoScroll();
  }

  autoScroll() { this.outEl.scrollTop = this.outEl.scrollHeight; }

  /* ---------- Event wiring ---------- */
  bindEvents() {
    this.inputEl.addEventListener('keydown', (e) => {
      if (e.key === 'Enter') {
        const raw = this.inputEl.value;
        if (raw.trim() === '') return;
        this.skipTyping();
        this.print('> ' + raw, 'echo', true);
        this.history.push(raw);
        this.historyIndex = this.history.length;
        this.inputEl.value = '';
        this.execute(raw);
        this.autoScroll();
      } else if (e.key === 'ArrowUp') {
        e.preventDefault();
        if (!this.history.length) return;
        this.historyIndex = Math.max(0, this.historyIndex - 1);
        this.inputEl.value = this.history[this.historyIndex];
        this.caretToEnd();
      } else if (e.key === 'ArrowDown') {
        e.preventDefault();
        if (!this.history.length) return;
        this.historyIndex = Math.min(this.history.length, this.historyIndex + 1);
        this.inputEl.value = this.historyIndex === this.history.length ? '' : this.history[this.historyIndex];
        this.caretToEnd();
      }
    });

    this.outEl.addEventListener('click', () => {
      if (this.typing) this.skipTyping();
      this.inputEl.focus();
    });

    this.skipBtn.addEventListener('click', (e) => {
      e.stopPropagation();
      this.skipTyping();
      this.inputEl.focus();
    });

    document.getElementById('reset-btn').addEventListener('click', () => {
      this.reset();
    });
  }

  caretToEnd() {
    const v = this.inputEl.value;
    this.inputEl.setSelectionRange(v.length, v.length);
  }

  /* ---------- Parser ---------- */
  partialMatch(word, list) {
    if (list.includes(word)) return word;
    const hits = list.filter(x => x.startsWith(word));
    return hits.length === 1 ? hits[0] : null;
  }

  parse(raw) {
    const clean = raw.trim().toLowerCase().replace(/\s+/g, ' ');
    if (!clean) return null;
    let parts = clean.split(' ');
    let verb = parts[0];
    let rest = parts.slice(1).join(' ').replace(/^(the|a|an)\s+/, '').trim();

    // Bare direction words: "north", "n", "s" ...
    const dirDirect = this.partialMatch(verb, this.DIRECTIONS);
    if (dirDirect && parts.length === 1) return { verb: 'go', noun: dirDirect };

    // "go n" / "go to north"
    if ((verb === 'go' || dirDirect) ) {
      if (dirDirect && rest) { // e.g., "n library" – treat dir as direction
        return { verb: 'go', noun: this.partialMatch(rest, this.DIRECTIONS) || rest };
      }
    }
    if (verb === 'go') {
      rest = rest.replace(/^to\s+/, '');
      const d = this.partialMatch(rest, this.DIRECTIONS);
      return { verb: 'go', noun: d || rest };
    }

    // Synonyms
    if (this.VERB_SYNONYMS[verb]) verb = this.VERB_SYNONYMS[verb];
    // "pick up key"
    if (verb === 'take' && rest.startsWith('up ')) rest = rest.slice(3);
    // Partial verb match: "inv" handled by synonym; "loo" -> "look"
    if (!this.VERBS.includes(verb)) {
      const m = this.partialMatch(verb, this.VERBS);
      if (m) verb = m;
    }
    // Strip leading article on nouns
    return { verb, noun: rest };
  }

  /* ---------- Command execution ---------- */
  execute(raw) {
    const cmd = this.parse(raw);
    if (!cmd) return;
    const { verb, noun } = cmd;

    switch (verb) {
      case 'go':         return this.cmdGo(noun);
      case 'look':       return this.cmdLook(noun);
      case 'take':       return this.cmdTake(noun);
      case 'drop':       return this.cmdDrop(noun);
      case 'use':        return this.cmdUse(noun);
      case 'inventory':  return this.cmdInventory();
      case 'help':       return this.cmdHelp();
      default:
        this.print(`I don't understand "${raw}". Try: look, go, take, drop, use, inventory, or help.`, 'error');
    }
  }

  cmdGo(dir) {
    if (!dir) { this.print('Go where? Specify a direction: north, south, east, or west.', 'error'); return; }
    if (!this.DIRECTIONS.includes(dir)) {
      this.print(`"${dir}" isn't a direction. Try north, south, east, or west.`, 'error'); return;
    }
    const exit = this.room.exits[dir];
    if (!exit) {
      this.print(`You walk firmly in the ${dir} direction and smack into a cold stone wall. There is no exit that way.`, 'error');
      return;
    }
    if (exit.locked && !this.flags[exit.lockFlag]) {
      this.print(exit.lockMsg, 'error');
      return;
    }
    this.moves++;
    this.currentRoomId = exit.to;
    this.updateSidebar();
    this.enterRoom(false);
  }

  cmdLook(noun) {
    if (!noun) { this.enterRoom(false); return; }
    // Examine: prefer inventory, then room
    const found = this.inventory.find(i => i.matches(noun)) || this.room.findItem(noun);
    if (found) { this.print(found.description, 'system'); return; }
    if (noun.includes('lever') && this.room.id === 'library') {
      this.print(this.flags.lever_pulled
        ? 'The brass lever is already thrown. Somewhere behind the western wall, stones remain politely settled.'
        : 'A stout brass lever on a iron mounting. Faint engraved words beneath it read: "FOR THE DRAFTY ARCHWAY".', 'system');
      return;
    }
    if (noun.includes('wall') ) {
      this.print('Damp stone, aggressively ordinary. One section behind the lectern sounds hollow — but you already knew that.', 'system');
      return;
    }
    if (noun.includes('door') && this.room.id === 'armory') {
      this.print('A heavy oak door, banded with iron. The keyhole is old, wolf-shaped, and waiting.', 'system');
      return;
    }
    this.print(`You stare intently at "${noun}", but it reveals no secrets. It isn't here (or you aren't holding it).`, 'error');
  }

  cmdTake(noun) {
    if (!noun) { this.print('Take what? There is plenty of damp air, but no room for it in your pockets.', 'error'); return; }
    if (noun.includes('lever')) { this.print('The lever is bolted to the wall. You are stronger than it, but not that stronger. Try "use lever".', 'error'); return; }
    const item = this.room.findItem(noun);
    if (!item) { this.print(`You don't see any "${noun}" here. (Try "look" to review the room.)`, 'error'); return; }
    this.room.items = this.room.items.filter(i => i !== item);
    this.inventory.push(item);
    this.itemsFound++;
    this.print(`You pocket the ${item.name}. ${item.description}`, 'normal');
    this.updateSidebar();
    this.updateHint();
  }

  cmdDrop(noun) {
    if (!noun) { this.print('Drop what? You resist the urge to dramatically cast your entire pack into a chasm.', 'error'); return; }
    const item = this.inventory.find(i => i.matches(noun));
    if (!item) { this.print(`You aren't carrying any "${noun}". Check your inventory with "inv".`, 'error'); return; }
    this.inventory = this.inventory.filter(i => i !== item);
    this.room.items.push(item);
    this.print(`You set the ${item.name} down on the floor. The dungeon accepts all donations.`, 'normal');
    this.updateSidebar();
  }

  cmdUse(noun) {
    if (!noun) { this.print('Use what? The dungeon offers no customer service desk.', 'error'); return; }
    if (noun.includes('lever')) {
      if (this.room.id !== 'library') { this.print('There is no lever here. The famous lever is in the library.', 'error'); return; }
      if (this.flags.lever_pulled) { this.print('You jiggle the lever. It has given its all. The west archway in the library stands open.', 'system'); return; }
      this.flags.lever_pulled = true;
      this.moves++;
      this.updateSidebar();
      this.print('You grip the cold brass lever and throw it with both hands. Deep inside the masonry, a counterweight GROANS, `+`'.replace('`+`',''));
      this.print('Deep inside the masonry, a counterweight GROANS. Somewhere to the west, stone grinds against stone. The sealed archway in the library is now open.', 'system');
      this.updateHint('The lever in the library did something to a sealed archway. West, perhaps?');
      return;
    }
    if (noun.includes('key')) {
      if (this.room.id !== 'armory') { this.print('You hold the rusted key up hopefully, but nothing nearby is lock-shaped.', 'error'); return; }
      if (!this.inventory.some(i => i.matches('key'))) { this.print('You reach for a key you do not have. The one in the Entrance Hall comes to mind.', 'error'); return; }
      if (this.flags.armory_unlocked) { this.print('The oak door already stands unlocked.', 'system'); return; }
      this.flags.armory_unlocked = true;
      this.moves++;
      this.print('The wolf-headed key turns with a howl of rust. The oak door swings inward, exhaling three centuries of held breath. The NORTH passage is open.', 'system');
      this.updateSidebar();
      this.updateHint('The armory\'s north door is unlocked. Treasure waits for the reasonably punctual.');
      return;
    }
    if (noun.includes('torch')) {
      if (!this.inventory.some(i => i.matches('torch'))) { this.print('You mime heroically holding a lit torch. It is an improvement on the unlit one in the sconce.', 'error'); return; }
      this.print('You light the torch from a stray candle. Warm orange light floods the room — the cobwebs everywhere else seem to lean away, respectfully.', 'system');
      this.updateHint('With the torch lit, even the dark corners of the dungeon feel merely untidy.');
      return;
    }
    if (noun.includes('scroll')) {
      if (!this.inventory.some(i => i.matches('scroll'))) { this.print('You cannot read a scroll you have not yet stolen from a pedestal.', 'error'); return; }
      this.print('You unroll the ancient scroll carefully. It reads: "Whoever reads this, the lease on the dungeon is renewable. — Management."', 'system');
      return;
    }
    this.print(`Using "${noun}" produces exactly nothing, which is itself a kind of magic.`, 'error');
  }

  cmdInventory() {
    if (!this.inventory.length) { this.print('Your pockets contain: lint, one (1) interesting pebble, and ambition. Inventory is empty.', 'system'); return; }
    this.print('You are carrying: ' + this.inventory.map(i => i.name).join(', ') + '.', 'system');
  }

  cmdHelp() {
    this.print('─── COMMANDS ───', 'title', true);
    this.print('look / l ............ describe the current room', 'normal', true);
    this.print('look at [thing] / x . examine an item or feature', 'normal', true);
    this.print('go [n|s|e|w] ........ move between rooms', 'normal', true);
    this.print('take [item] ......... pick something up (grab/get work too)', 'normal', true);
    this.print('drop [item] ......... set something down', 'normal', true);
    this.print('use [thing] ......... pull levers, turn keys, light torches', 'normal', true);
    this.print('inventory / inv / i . list what you carry', 'normal', true);
    this.print('Tip: the Up/Down arrow keys recall your previous commands.', 'system', true);
  }

  /* ---------- Sidebar ---------- */
  updateSidebar() {
    document.getElementById('stat-moves').textContent = this.moves;
    document.getElementById('stat-items').textContent = this.itemsFound;
    const list = document.getElementById('inventory-list');
    list.innerHTML = '';
    if (!this.inventory.length) {
      const li = document.createElement('li');
      li.className = 'empty';
      li.textContent = 'Inventory is empty.';
      list.appendChild(li);
      return;
    }
    this.inventory.forEach(item => {
      const li = document.createElement('li');
      li.innerHTML =
        '<svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="#00ff00" stroke-width="2" stroke-linecap="round" aria-hidden="true">' +
        '<path d="M20 12 l-8 8 -8 -8 4-4 h8 Z" transform="rotate(45 12 8)"/></svg>';
      li.appendChild(document.createTextNode(item.name));
      list.appendChild(li);
    });
  }

  updateHint(text) {
    const el = document.getElementById('hint-text');
    if (text) { el.textContent = text; return; }
    const total = 4; // collectible items
    if (this.itemsFound === 0) el.textContent = 'The Entrance Hall glints with something left behind. Loot first, ask questions never.';
    else if (this.itemsFound < total) el.textContent = `${this.itemsFound} of ${total} treasures recovered. The dungeon still has secrets — locked doors and hollow walls, specifically.`;
    else el.textContent = 'Every treasure claimed. A mossy archway to the south feels like it is calling you home.';
  }
}
