    this.id = id;
    this.name = name;
    this.description = description;
    this.exits = exits;          // { north: 'room_id', ... }
    this.items = [];             // Item[]

    const idx = this.items.findIndex(i => i.id === itemId);

    return this.items.splice(idx, 1)[0];
  }
  findItem(query) { return this.items.find(i => i.id === query) || null; }
}

      'A stout iron lever bolted into the masonry. Engraved beneath it, in cramped Latin, are the words: "Ad occultum" — "To the hidden place." It is definitely not meant to be taken.', false));
    library.exits = { south: 'entrance', west: 'secret' };

    const armory = new Room('armory', 'The Armory',
      'Racks of spears stand at attention in the gloom, their points furred with rust but unbroken. A row of empty breastplates lines the wall like silent sentries. ' +
      'A single unlit torch leans against the rack, oddly dry. The north doorway is sealed by a heavy oak door bound in iron — and it is firmly locked.');
    armory.addItem(new Item('torch', 'Torch',
      'A pitch-soaked torch, dry as kindling. Whoever left it here expected to come back for it. It would take a flame to light, but it makes an excellent improvised club or walking stick.'));
    armory.exits = { west: 'entrance', north: 'treasure' };

    const treasure = new Room('treasure', 'Treasure Room',
      'The treasury is... modest. A plinth of carved marble sits at the center of the room, lit by a shaft of pale light with no visible source. ' +
      'Shelves along the walls stand empty — this dungeon was robbed centuries ago. Everything except one thing.');
    treasure.addItem(new Item('golden_coin', 'Golden Coin',
      'A single gold coin the size of your palm, stamped with the face of a king no history book remembers. It is warm to the touch, and faintly hums when you hold it up to the light.'));
    treasure.exits = { south: 'armory' };

    const secret = new Room('secret', 'Secret Chamber',
      'Behind the wall lies a hexagonal chamber no larger than a wardrobe. Crystals in the ceiling glow with a soft, patient amber light. ' +
      'A stone pedestal holds a single scroll, sealed with wax the color of dried blood. The dust here has never been disturbed — until now.');
    secret.addItem(new Item('ancient_scroll', 'Ancient Scroll',
      'The scroll unfurls with a crackle. In faded ink you read: "Whoever reads this has already gone further than the wise. Well done, fool. The garden is south of the hall — rest there."'));
    secret.exits = { east: 'library' };

    const garden = new Room('garden', 'The Underground Garden',
      'You step onto soft, living grass. An impossible garden spreads beneath a ceiling lost in luminous mist — apple trees heavy with fruit, a stone fountain murmuring, beds of violets perfuming the dark air. ' +
      'Birds you cannot see sing somewhere in the branches. This is where the dungeon lets you go.');
    garden.exits = { north: 'entrance' };

    this.rooms = { entrance, library, armory, treasure, secret, garden };
  }

  /* ---------- State ---------- */
  resetState() {
    this.state = {
      currentRoom: 'entrance',
      inventory: [],
      visited: new Set(),
      flags: { lever_pulled: false, armory_unlocked: false, garden_end: false },
      moves: 0,
      itemsFound: 0
    };
    this.history = [];
    this.historyIndex = -1;
    // Rebuild pristine item placements
    this.initWorld();
  }

  /* ---------- Boot / Reset ---------- */
  start() {
    this.resetState();
    this.outputEl.innerHTML = '';
    this.typeQueue = [];
    this.typing = false;
    this.updateUI();
    this.print('THE LOCAL DUNGEON — v1.0', 'system');
    this.print('You descend the collapsed cellar stairs of the old bookshop. The dungeon smells of dust, old paper, and possibility.', 'story');
    this.print('Type "help" for a list of commands. Type "look" to begin.', 'system');
    this.enterRoom(this.state.currentRoom, true);
    this.focusInput();
  }

  reset() {
    this.start();
  }

  /* ---------- Typewriter engine ---------- */
  print(text, cls = 'story') {
    this.typeQueue.push({ text, cls });
    if (!this._running) this.runQueue();
  }

  async runQueue() {
    this._running = true;
    while (this.typeQueue.length) {
      const { text, cls } = this.typeQueue.shift();
      await this.typeLine(text, cls);
    }
    this._running = false;
  }

  async typeLine(text, cls) {
    const line = document.createElement('div');
    line.className = 'line ' + cls;
    const span = document.createElement('span');
    const cursor = document.createElement('span');
    cursor.className = 'cursor';
    cursor.innerHTML = '&nbsp;';
    line.appendChild(span);
    line.appendChild(cursor);
    this.outputEl.appendChild(line);
    this.scrollToBottom();

    this.typing = true;
    this.skipRequested = false;
    this.skipBtn.style.display = 'inline-block';

    for (let i = 0; i < text.length; i++) {
      if (this.skipRequested) {
        span.textContent = text;
        break;
      }
      const ch = text[i];
      if (ch === '\n') span.appendChild(document.createElement('br'));
      else span.textContent += ch;
      this.scrollToBottom();
      await new Promise(r => setTimeout(r, this.typeSpeed));
    }

    cursor.remove();
    this.typing = false;
    this.skipBtn.style.display = 'none';
    this.scrollToBottom();
  }

  skipTyping() {
    if (this.typing) this.skipRequested = true;
  }

  scrollToBottom() {
    this.outputEl.scrollTop = this.outputEl.scrollHeight;
  }

  /* ---------- UI updates ---------- */
  updateUI() {
    this.movesEl.textContent = this.state.moves;
    this.itemsEl.textContent = this.state.itemsFound;

    this.invEl.innerHTML = '';
    if (this.state.inventory.length === 0) {
      this.emptyInvEl.style.display = 'block';
    } else {
      this.emptyInvEl.style.display = 'none';
      for (const item of this.state.inventory) {
        const li = document.createElement('li');
        li.innerHTML = '<svg viewBox="0 0 24 24" fill="none" stroke="#00ff00" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><path d="M21 16V8a2 2 0 0 0-1-1.73l-7-4a2 2 0 0 0-2 0l-7 4A2 2 0 0 0 3 8v8a2 2 0 0 0 1 1.73l7 4a2 2 0 0 0 2 0l7-4A2 2 0 0 0 21 16z"/><polyline points="3.27 6.96 12 12.01 20.73 6.96"/><line x1="12" y1="22.08" x2="12" y2="12"/></svg>';
        const label = document.createElement('span');
        label.textContent = item.name;
        li.appendChild(label);
        this.invEl.appendChild(li);
      }
    }

    const room = this.rooms[this.state.currentRoom];
    this.locNameEl.textContent = room.name;
    const exits = Object.keys(room.exits)
      .filter(d => !(room.id === 'library' && d === 'west' && !this.state.flags.lever_pulled))
      .map(d => {
        if (room.id === 'armory' && d === 'north' && !this.state.flags.armory_unlocked) return 'north (locked)';
        return d;
      });
    this.locExitsEl.textContent = 'Exits: ' + (exits.length ? exits.join(', ') : 'none');
  }

  /* ---------- Room entry ---------- */
  enterRoom(roomId, first = false) {
    const room = this.rooms[roomId];
    this.state.currentRoom = roomId;
    const firstVisit = !this.state.visited.has(roomId);
    this.state.visited.add(roomId);

    this.print('══ ' + room.name.toUpperCase() + ' ══', 'room');
    this.print(room.description, 'story');

    if (firstVisit) {
      const takeables = room.items.filter(i => i.takeable);
      if (takeables.length) {
        this.print('You notice: ' + takeables.map(i => i.name).join(', ') + '.', 'system');
      }
    }

    if (roomId === 'library' && !this.state.flags.lever_pulled) {
      this.print('The western side of the library is nothing but a seamless wall of ancient stone. For now.', 'story');
    }

    if (roomId === 'garden' && !this.state.flags.garden_end) {
      this.state.flags.garden_end = true;
      this.print('', 'story');
      this.print('✦ PEACEFUL ENDING REACHED ✦', 'system');
      this.print('You sink into the cool grass beneath the apple trees. Above you — far, far above — the town goes about its evening, never knowing.', 'story');
      this.print('Total moves: ' + this.state.moves + ' — Items found: ' + this.state.itemsFound + ' of 4. You may keep exploring if you like; the garden will wait.', 'system');
    }

    this.updateUI();
  }

  /* ---------- Fuzzy item matching ---------- */
  findItemFuzzy(query, list) {
    if (!query) return null;
    const q = query.trim().toLowerCase();
    // exact id
    let hit = list.find(i => i.id === q);
    if (hit) return hit;
    // full name contains query, or query contains full name
    hit = list.find(i => i.name.toLowerCase().includes(q) || q.includes(i.name.toLowerCase()));
    if (hit) return hit;
    // word-level: every word of query appears in the name
    const words = q.split(/\s+/);
    hit = list.find(i => {
      const n = i.name.toLowerCase();
      return words.some(w => w.length > 2 && n.includes(w));
    });
    return hit || null;
  }

  /* ---------- Parser ---------- */
  matchVerb(sentence) {
    // Try two-word verbs first, then one-word verbs.
    const words = sentence.split(/\s+/);
    const two = words.slice(0, 2).join(' ');
    for (const [verb, syns] of Object.entries(this.verbMap)) {
      if (syns.some(s => s.includes(' ') && (two === s || s.startsWith(two) && two.length >= 3))) {
        return { verb, consumed: s.split(' ').length, synonym: s };
      }
    }
    const one = words[0];
    // exact match first
    for (const [verb, syns] of Object.entries(this.verbMap)) {
      if (syns.some(s => !s.includes(' ') && s === one)) return { verb, consumed: 1 };
    }
    // partial match (prefix)
    for (const [verb, syns] of Object.entries(this.verbMap)) {
      const s = syns.find(s => !s.includes(' ') && s.startsWith(one) && one.length >= 2);
      if (s) return { verb, consumed: 1 };
    }
    return null;
  }

  execute(raw) {
    const sentence = raw.trim().toLowerCase();
    if (!sentence) return;

    this.print('> ' + raw.trim(), 'echo');
    this.history.push(raw.trim());
    this.historyIndex = this.history.length;

    const m = this.matchVerb(sentence);
    if (!m) {
      this.print('I don\'t understand "' + raw.trim() + '". Try: look, go north, take key, use lever, inventory, or help.', 'error');
      return;
    }

    const rest = sentence.split(/\s+/).slice(m.consumed).join(' ').trim();

    switch (m.verb) {
      case 'look':      this.cmdLook(rest); break;
      case 'examine':   this.cmdLook(rest || 'room'); break;
      case 'go':        this.cmdGo(rest); break;
      case 'take':      this.cmdTake(rest); break;
      case 'drop':      this.cmdDrop(rest); break;
      case 'use':       this.cmdUse(rest); break;
      case 'inventory': this.cmdInventory(); break;
      case 'help':      this.cmdHelp(); break;
      case 'reset':     this.print('Resetting the adventure...', 'system'); setTimeout(() => this.reset(), 400); break;
    }
    this.updateUI();
  }

  /* ---------- Commands ---------- */
  cmdLook(target) {
    const room = this.rooms[this.state.currentRoom];

    if (!target || target === 'room' || target === 'around') {
      this.print(room.description, 'story');
      const takeables = room.items.filter(i => i.takeable);
      if (takeables.length) {
        this.print('Here you can see: ' + takeables.map(i => i.name).join(', ') + '.', 'system');
      } else {
        this.print('There is nothing here to take.', 'system');
      }
      const exits = Object.keys(room.exits)
        .filter(d => !(room.id === 'library' && d === 'west' && !this.state.flags.lever_pulled));
      this.print('Exits: ' + (exits.length ? exits.join(', ') : 'none') + '.', 'system');
      return;
    }

    // examine an item in the room or inventory
    const item = this.findItemFuzzy(target, room.items.concat(this.state.inventory));
    if (item) { this.print(item.description, 'story'); return; }

    // examine scenery
    if (/door/.test(target)) {
      if (room.id === 'armory') {
        this.print(this.state.flags.armory_unlocked
          ? 'The heavy oak door, banded in iron, now stands open on its hinges.'
          : 'A massive oak door bound in iron. The keyhole is shaped for a trefoil-headed key — three lobes, like a clover.', 'story');
      } else if (room.id === 'entrance') {
        this.print('The cellar door you came in by has warped shut. Wood and rust and time have welded the story closed. This way is out.', 'story');
      } else {
        this.print('No doors here but the ones that lead onward.', 'story');
      }
      return;
    }
    if (/wall/.test(target)) {
      if (room.id === 'library') {
        this.print(this.state.flags.lever_pulled
          ? 'Where a seamless wall stood an hour ago, a low passage now yawns open to the west, exhaling cool, dry air.'
          : 'Close-set courses of ancient stone. Running your fingers along them, you find no crack, no hinge, no seam — yet the wall feels hollow near the reading desk.', 'story');
      } else {
        this.print('Damp stone blocks, laid by masons who clearly hated decoration.', 'story');
      }
      return;
    }
    if (/lever|switch/.test(target)) {
      if (room.id === 'library') { this.print(this.rooms.library.items[0].description, 'story'); return; }
      this.print('There is no lever here.', 'error'); return;
    }
    if (/torch ?bracket/.test(target)) {
      this.print('Cold iron brackets, blackened where torches burned centuries ago. Whatever lit this dungeon went out a very long time ago.', 'story');
      return;
    }

    this.print('You peer closely at "' + target + '" but can make neither head nor tail of it. Try looking at something in the room.', 'error');
  }

  cmdGo(dirInput) {
    if (!dirInput) {
      const room = this.rooms[this.state.currentRoom];
      this.print('Go where? Available directions: ' + (Object.keys(room.exits).join(', ') || 'none') + '.', 'error');
      return;
    }
    // Allow "go to north", "north", "n"
    const cleaned = dirInput.replace(/^to\s+/, '').trim();
    const dir = this.dirMap[cleaned.split(/\s+/)[0]];
    if (!dir) {
      this.print('"' + cleaned + '" is not a direction. Try north, south, east, or west (n / s / e / w).', 'error');
      return;
    }

    const room = this.rooms[this.state.currentRoom];
    if (!room.exits[dir]) {
      this.print('You cannot go ' + dir + ' from here. There is only solid rock in that direction.', 'error');
      return;
    }

    // Special: sealed wall west of library
    if (room.id === 'library' && dir === 'west' && !this.state.flags.lever_pulled) {
      this.print('You walk west and walk into a wall. Solid, seamless, faintly smug stonework. Perhaps something in this library could open it.', 'error');
      return;
    }

    // Special: locked armory door
    if (room.id === 'armory' && dir === 'north') {
      if (!this.state.flags.armory_unlocked) {
        const key = this.state.inventory.find(i => i.id === 'rusted_key');
        if (key) {
          this.state.flags.armory_unlocked = true;
          this.print('You slide the rusted key into the trefoil keyhole. It resists, you resist back, and with a groan like an old man standing up, the lock turns.', 'story');
          this.print('The iron-banded door swings open northward, exhaling a breath of gold-tinged air.', 'story');
        } else {
          this.print('The north door is locked fast. The keyhole is shaped for a trefoil-headed key — three lobes, like a clover. You don\'t have the key. (Check the Entrance Hall — you may have missed something glinting.)', 'error');
          return;
        }
      }
    }

    this.state.moves++;
    this.updateUI();
    this.enterRoom(room.exits[dir]);
  }

  cmdTake(target) {
    if (!target) { this.print('Take what? Try "take key".', 'error'); return; }
    const room = this.rooms[this.state.currentRoom];
    const item = this.findItemFuzzy(target, room.items);
    if (!item) {
      const held = this.findItemFuzzy(target, this.state.inventory);
      if (held) { this.print('You are already carrying the ' + held.name.toLowerCase() + '.', 'error'); return; }
      this.print('You don\'t see any "' + target + '" here. Try "look" to see what\'s in the room.', 'error');
      return;
    }
    if (!item.takeable) {
      if (item.id === 'lever') this.print('You heave at the lever with both hands. It is bolted into the wall and has no intention of leaving. Try "pull lever" instead.', 'error');
      else this.print('The ' + item.name.toLowerCase() + ' is not coming with you.', 'error');
      return;
    }
    room.removeItem(item.id);
    this.state.inventory.push(item);
    this.state.itemsFound++;
    const flavor = {
      rusted_key: 'You pocket the rusted key. It\'s heavier than it looks, and faintly warm.',
      torch: 'You take the torch. It\'s dry, pitchy, and smells faintly of a bonfire not yet lit.',
      golden_coin: 'The golden coin leaps into your palm as if it had been waiting. The hum in your bones grows louder for a moment, then settles.',
      ancient_scroll: 'You lift the scroll from the pedestal. Somewhere far off, a stone slides — politely — back into place.'
    };
    this.print(flavor[item.id] || ('You take the ' + item.name.toLowerCase() + '.'), 'story');
    this.updateUI();
  }

  cmdDrop(target) {
    if (!target) { this.print('Drop what?', 'error'); return; }
    const item = this.findItemFuzzy(target, this.state.inventory);
    if (!item) {
      this.print('You aren\'t carrying any "' + target + '". Try "inventory" to see what you hold.', 'error');
      return;
    }
    const idx = this.state.inventory.indexOf(item);
    this.state.inventory.splice(idx, 1);
    this.rooms[this.state.currentRoom].addItem(item);
    this.print('You set the ' + item.name.toLowerCase() + ' down. It lands with a small, final sound.', 'story');
    this.updateUI();
  }

  cmdUse(target) {
    if (!target) { this.print('Use what? Try "pull lever" or "use key".', 'error'); return; }
    const room = this.rooms[this.state.currentRoom];

    // Lever logic
    if (/lever|switch/.test(target)) {
      if (room.id !== 'library') { this.print('There is no lever here.', 'error'); return; }
      if (this.state.flags.lever_pulled) {
        this.print('You jiggle the lever, but it has already done its one great work. The western passage stands open.', 'story');
        return;
      }
      this.state.flags.lever_pulled = true;
      this.print('You brace your boots and throw your weight against the lever. Deep inside the wall, something enormous shifts — counterweights, you think, older than the town above.', 'story');
      this.print('With a grinding sigh, the seamless wall to the WEST splits apart and folds inward, revealing a hidden passage bathed in faint amber light.', 'system');
      this.updateUI();
      return;
    }

    if (/key/.test(target)) {
      const key = this.state.inventory.find(i => i.id === 'rusted_key') || this.findItemFuzzy(target, room.items);
      if (!key) { this.print('You don\'t have a key on you. The Entrance Hall had one, if memory (and dust) serve.', 'error'); return; }
      if (room.id === 'armory') {
        if (this.state.flags.armory_unlocked) { this.print('The north door is already open.', 'story'); return; }
        this.print('You try the key in the trefoil keyhole — it fits! Now, rather than just standing here holding a turned key, try "go north".', 'system');
        return;
      }
      this.print('You look for a keyhole, but every door that matters is elsewhere. Try the Armory to the east.', 'story');
      return;
    }

    if (/torch/.test(target)) {
      this.print('You pat your pockets for flint and steel. You have neither. The torch remains stubbornly unlit, but it\'s still useful to carry.', 'story');
      return;
    }
    if (/scroll/.test(target)) {
      this.print('You read the scroll again: "The garden is south of the hall — rest there." Some advice is worth taking literally.', 'story');
      return;
    }
    if (/coin/.test(target)) {
      this.print('You flip the golden coin. It comes up king-side, as always, and hums approvingly. Spending it down here seems inadvisable.', 'story');
      return;
    }

    const item = this.findItemFuzzy(target, room.items.concat(this.state.inventory));
    if (item) this.print('You fiddle with the ' + item.name.toLowerCase() + ', but nothing happens. Perhaps "use" isn\'t its purpose.', 'story');
    else this.print('You don\'t see any "' + target + '" to use.', 'error');
  }

  cmdInventory() {
    if (this.state.inventory.length === 0) {
      this.print('Your hands are empty. The dungeon has not yet trusted you with anything.', 'system');
      return;
    }
    this.print('You are carrying: ' + this.state.inventory.map(i => i.name).join(', ') + '.', 'system');
  }

  cmdHelp() {
    this.print([
      'AVAILABLE COMMANDS:',
      '  look / l              — survey the room',
      '  look at <thing>       — examine something closely (or: examine, x)',
      '  go <dir>  /  n s e w  — move between rooms',
      '  take <item>           — pick something up (or: get, grab)',
      '  drop <item>           — put something down',
      '  use <item>            — use or pull something',
      '  inventory / inv / i   — list carried items',
      '  reset                 — start the adventure over',
      'Tip: partial words work ("n" = "north", "grab ke" = "take key").'
    ].join('\n'), 'system');
  }

  /* ---------- Events ---------- */
  bindEvents() {
    this.inputEl.addEventListener('keydown', (e) => {
      if (e.key === 'Enter') {
        const val = this.inputEl.value;
        if (this.typing) { this.skipTyping(); this.inputEl.value = ''; return; }
        this.inputEl.value = '';
        this.execute(val);
      } else if (e.key === 'ArrowUp') {
        e.preventDefault();
        if (!this.history.length) return;
        this.historyIndex = Math.max(0, this.historyIndex - 1);
        this.inputEl.value = this.history[this.historyIndex] || '';
      } else if (e.key === 'ArrowDown') {
        e.preventDefault();
        if (!this.history.length) return;
        this.historyIndex = Math.min(this.history.length, this.historyIndex + 1);
        this.inputEl.value = this.historyIndex === this.history.length ? '' : this.history[this.historyIndex];
      }
    });

    this.skipBtn.addEventListener('click', (e) => {
      e.stopPropagation();
      this.skipTyping();
    });

    this.frameEl.addEventListener('click', (e) => {
      if (e.target === this.skipBtn) return;
      if (this.typing) this.skipTyping();
      this.focusInput();
    });

    document.getElementById('resetBtn').addEventListener('click', () => this.reset());
  }

  focusInput() {
    this.inputEl.focus({ preventScroll: true });
  }
}
