'use strict';

/* ============================================================
   Item
   ============================================================ */
class Item {
  constructor(name, desc, takeable = true, unTakeMsg = '') {
    this.name = name;
    this.desc = desc;
    this.takeable = takeable;
    this.unTakeMsg = unTakeMsg;
  }
}

/* ============================================================
   Room
   ============================================================ */
class Room {
  constructor(id, name, desc) {
    this.id = id;
    this.name = name;
    this.desc = desc;
    this.exits = {};    // dir -> roomId (open)
    this.blocked = {};  // dir -> { target, flag, requires, lockedMsg, unlockMsg }
    this.items = [];
  }
}

/* ============================================================
   GameEngine
   ============================================================ */
class GameEngine {
  constructor() {
    this.consoleEl = document.getElementById('console');
    this.inputEl = document.getElementById('cmd');

    this.statFound = document.getElementById('statFound');

    this.invList = document.getElementById('invList');

    this.typeQueue = [];
    this.typeTimer = null;

    this.bindEvents();

  }

  /* ---------------- World ---------------- */
  buildWorld() {

    armory.exits = { south: 'entrance' };
    armory.blocked.north = {

    garden.exits = { north: 'entrance' };

    // Items
    const rustedKey = new Item('rusted key',

  /* ---------------- State / reset ---------------- */
  reset() {
    this.skipTyping();
    this.consoleEl.innerHTML = '';

    this.rooms = this.buildWorld();
    this.currentRoom = this.rooms.entrance;
    this.inventory = [];
    this.visited = new Set([this.currentRoom.id]);
    this.found = new Set();
    this.moves = 0;
    this.flags = { leverPulled: false, armoryDoorOpen: false };
    this.history = [];
    this.histIdx = 0;
    this.inputEl.value = '';

    this.write('The Local Dungeon', 'sys');
    this.write('A pocket-sized adventure in six rooms.');
    this.write("Type 'look' to begin, or 'help' for the command list.");
    this.write('');
    this.roomLine();

    this.updateUI();
  }

  /* ---------------- Typewriter console ---------------- */
  write(text, cls = 'sys') {
    const el = document.createElement('div');
    el.className = 'line ' + cls;
    this.consoleEl.appendChild(el);
    this.typeQueue.push({ el, text, pos: 0 });
    this.startTyping();
    this.scrollToBottom();
  }

  startTyping() {
    if (this.typeTimer) return;
    this.typeTimer = setInterval(() => {
      const cur = this.typeQueue[0];
      if (!cur) {
        clearInterval(this.typeTimer);
        this.typeTimer = null;
        return;
      }
      cur.pos = Math.min(cur.text.length, cur.pos + 2);
      cur.el.textContent = cur.text.slice(0, cur.pos);
      if (cur.pos >= cur.text.length) this.typeQueue.shift();
      this.scrollToBottom();
    }, 12);
  }

  skipTyping() {
    for (const q of this.typeQueue) q.el.textContent = q.text;
    this.typeQueue.length = 0;
    if (this.typeTimer) {
      clearInterval(this.typeTimer);
      this.typeTimer = null;
    }
  }

  scrollToBottom() {
    this.consoleEl.scrollTop = this.consoleEl.scrollHeight;
  }

  /* ---------------- Events ---------------- */
  bindEvents() {
    this.inputEl.addEventListener('keydown', (e) => {
      if (e.key === 'Enter') {
        const v = this.inputEl.value;
        this.inputEl.value = '';
        if (v.trim()) this.handle(v);
      } else if (e.key === 'ArrowUp') {
        e.preventDefault();
        if (this.histIdx > 0) {
          this.histIdx--;
          this.inputEl.value = this.history[this.histIdx] || '';
        }
      } else if (e.key === 'ArrowDown') {
        e.preventDefault();
        if (this.histIdx < this.history.length) {
          this.histIdx++;
          this.inputEl.value = this.history[this.histIdx] || '';
        }
      }
    });

    this.consoleEl.addEventListener('click', () => {
      this.skipTyping();
      this.inputEl.focus();
    });

    document.getElementById('skipBtn').addEventListener('click', (e) => {
      e.stopPropagation();
      this.skipTyping();
      this.inputEl.focus();
    });

    document.getElementById('resetBtn').addEventListener('click', () => {
      this.reset();
      this.inputEl.focus();
    });

    document.addEventListener('click', (e) => {
      if (!e.target.closest('.console-shell')) { /* leave focus alone elsewhere */ }
    });
  }

  /* ---------------- Command pipeline ---------------- */
  handle(raw) {
    const cmd = raw.trim().toLowerCase();
    if (!cmd) return;
    this.history.push(cmd);
    this.histIdx = this.history.length;
    this.write('> ' + cmd, 'player');
    this.parse(cmd);
    this.updateUI();
  }

  parse(cmd) {
    const parts = cmd.split(/\s+/);
    const verb = parts[0];
    const rest = parts.slice(1).join(' ').trim();

    const DIRS = ['north', 'south', 'east', 'west'];
    const DIR_ABBR = { n: 'north', s: 'south', e: 'east', w: 'west' };

    // Bare directions: "north", "n"
    if (DIRS.includes(verb)) return this.go(verb);
    if (DIR_ABBR[verb] && !rest) return this.go(DIR_ABBR[verb]);

    switch (verb) {
      case 'go':
      case 'move':
      case 'walk':
      case 'head': {
        if (!rest) return this.write('Go where? Try "go north".');
        const dir = DIRS.find(d => d.startsWith(rest));
        if (!dir) return this.write(`You can't find a way to go "${rest}".`);
        return this.go(dir);
      }
      case 'look':
      case 'l': {
        if (rest && rest.startsWith('at ')) return this.describe(rest.slice(3));
        return this.look();
      }
      case 'examine':
      case 'inspect':
      case 'study': {
        if (!rest) return this.look();
        return this.describe(rest);
      }
      case 'take':
      case 'grab':
      case 'get':
      case 'pick':
        return this.take(rest);
      case 'drop':
        return this.drop(rest);
      case 'inv':
      case 'inventory':
      case 'items':
      case 'pockets':
        return this.showInventory();
      case 'use':
      case 'pull':
      case 'operate':
        return this.use(rest);
      case 'read':
        return this.read(rest);
      case 'help':
      case 'commands':
        return this.showHelp();
      case 'reset':
        this.write('You take a deep breath, let go of everything, and begin again.');
        this.reset();
        return;
      default:
        return this.write(`I don't understand "${verb}". Type 'help' to see what the dungeon allows.`);
    }
  }

  /* ---------------- Items ---------------- */
  matchName(itemName, query) {
    const n = itemName.toLowerCase();
    if (!query) return false;
    if (n === query) return true;
    if (n.startsWith(query)) return true;
    return query.split(/\s+/).some(w => w.length > 2 && n.includes(w));
  }

  findItem(list, query) {
    return list.find(i => this.matchName(i.name, query)) || null;
  }

  take(query) {
    if (!query) return this.write('Take what? Name an object, like "take torch".');
    const item = this.findItem(this.currentRoom.items, query);
    if (!item) return this.write(`There's no ${query} here.`);
    if (!item.takeable) return this.write(item.unTakeMsg);
    this.currentRoom.items = this.currentRoom.items.filter(i => i !== item);
    this.inventory.push(item);
    this.found.add(item.name);
    this.write(`You take the ${item.name}.`);
  }

  drop(query) {
    if (!query) return this.write('Drop what? Name an object from your pockets.');
    const item = this.findItem(this.inventory, query);
    if (!item) return this.write(`You don't have a ${query}.`);
    this.inventory = this.inventory.filter(i => i !== item);
    this.currentRoom.items.push(item);
    this.write(`You set the ${item.name} down on the flagstones.`);
  }

  showInventory() {
    if (this.inventory.length === 0) {
      this.write('Your pockets are empty. The dungeon still has five objects to give away.');
    } else {
      this.write('You are carrying:\n' + this.inventory.map(i => '  - ' + i.name).join('\n'));
    }
  }

  /* ---------------- Rooms / movement ---------------- */
  look() {
    const r = this.currentRoom;
    let open = Object.keys(r.exits);
    for (const dir of Object.keys(r.blocked)) {
      if (this.flags[r.blocked[dir].flag]) open.push(dir);
    }
    let text = r.name + '\n' + r.desc + '\nExits: ' + (open.length ? open.join(', ') : 'nowhere');
    if (r.items.length) {
      text += '\nYou can see: ' + r.items.map(i => i.name).join(', ') + '.';
    } else {
      text += '\nThe room is bare of anything worth carrying.';
    }
    this.write(text);
  }

  roomLine() {
    this.look();
  }

  describe(query) {
    if (!query) return this.look();
    const item = this.findItem(this.currentRoom.items, query) || this.findItem(this.inventory, query);
    if (item) {
      const where = this.inventory.includes(item) ? 'In your pocket, ' : 'Here, ';
      this.write(where + item.name + '.\n' + item.desc);
      return;
    }
    this.write(`You look for a ${query} and find only dust and the suggestion of a draft.`);
  }

  go(dir) {
    const room = this.currentRoom;
    let target = room.exits[dir] || null;
    const blocked = room.blocked[dir];

    if (blocked && !target) {
      if (this.flags[blocked.flag]) {
        target = blocked.target;
      } else if (blocked.requires) {
        const idx = this.inventory.findIndex(i => i.name === blocked.requires);
        if (idx > -1) {
          this.inventory.splice(idx, 1);
          this.flags[blocked.flag] = true;
          this.write(room.unlockMsg);
          target = blocked.target;
        } else {
          this.write(blocked.lockedMsg);
          return;
        }
      } else {
        this.write(blocked.lockedMsg);
        return;
      }
    }

    if (!target) {
      this.write(`You can't go ${dir} from here. The stone is where it's always been.`);
      return;
    }

    const next = this.rooms[target];
    this.currentRoom = next;
    this.moves++;
    this.visited.add(next.id);
    this.roomLine();
  }

  /* ---------------- Interactions ---------------- */
  use(query) {
    if (!query) return this.write('Use what on what? Name the object.');

    if (this.matchName('lever', query)) {
      if (this.currentRoom.id !== 'library') {
        return this.write("There's no lever here. The lever is in the Library.");
      }
      if (this.flags.leverPulled) {
        return this.write('The lever is already caught in its new position, pointing at nothing in particular.');
      }
      this.flags.leverPulled = true;
      this.currentRoom.items = this.currentRoom.items.filter(i => i.name !== 'lever');
      this.write('You grip the brass handle and pull. A deep grind rolls through the walls like a stone rolling in a drum, then stops. A seam of shadow appears in the west wall, and a section of it slides open on silent runners, revealing a narrow passage.');
      return;
    }

    if (this.matchName('torch', query)) {
      if (!this.inventory.some(i => i.name === 'torch')) {
        return this.write("You don't have a torch to raise.");
      }
      if (this.currentRoom.id === 'garden') {
        return this.write('You hold up the torch to the daylight, then feel foolish, and put it away.');
      }
      this.write('You strike the torch against the stone. It catches with a soft bloom, and a ring of honest light pushes the gloom back a good two feet.');
      return;
    }

    const item = this.findItem(this.currentRoom.items, query) || this.findItem(this.inventory, query);
    if (item) {
      return this.write(`You fiddle with the ${item.name}, but it was not built for fiddling.`);
    }
    this.write(`You don't see anything here that a "use" would help.`);
  }

  read(query) {
    if (!query) return this.write('Read what? You are not a book.');
    if (this.matchName('ancient scroll', query) || this.matchName('scroll', query)) {
      if (!this.inventory.some(i => i.name === 'ancient scroll') &&
          !this.currentRoom.items.some(i => i.name === 'ancient scroll')) {
        return this.write('There is no scroll here. Scrolls in this dungeon are known to be particular about their rooms.');
      }
      if (this.currentRoom.id === 'garden') {
        this.write("You unroll the ancient scroll beneath the pale light of the sunken garden. The ink seems to stand up out of the page:\n\n\"Beyond the wall that breathes, where stone learns the sky, the wanderer rests. May they rest where the sun finds the stone.\"\n\nThe sun above finds the stone below. The fountain, the ferns, the quiet. The dungeon releases its breath and lets you go.\n\nYou have found the peaceful end of The Local Dungeon. Thank you for playing.");
        return;
      }
      this.write('You unroll the ancient scroll. The ink is old but legible:\n\n"Beyond the wall that breathes, where stone learns the sky, the wanderer rests."');
      return;
    }
    const item = this.findItem(this.currentRoom.items, query) || this.findItem(this.inventory, query);
    if (item) return this.write(`You try to read the ${item.name}, but it contains no words, only intent.`);
    this.write(`There's nothing to read by the name of ${query} here.`);
  }

  showHelp() {
    this.write(
      "Commands the dungeon understands:\n" +
      "  look / examine [thing]   describe the room or an object\n" +
      "  go <dir> / n s e w       move (north, south, east, west)\n" +
      "  take / grab / get        pick something up\n" +
      "  drop                     set something down\n" +
      "  inv / inventory          list what you carry\n" +
      "  use / pull [thing]       work the machinery\n" +
      "  read [thing]             for things written on\n" +
      "  help                     this list\n" +
      "  reset                    start over\n\n" +
      "Short forms are fine: 'examine' means 'look at', 'grab' means 'take', 'inv' means 'inventory', 'n' means 'north'."
    );
  }

  /* ---------------- UI ---------------- */
  updateUI() {
    this.statMoves.textContent = this.moves;
    this.statFound.textContent = this.found.size;
    this.statRoom.textContent = this.currentRoom.name.replace('The ', '');

    this.invList.innerHTML = '';
    const svg = '<svg width="12" height="12" viewBox="0 0 12 12" aria-hidden="true"><path d="M6 0.5 L11.5 6 L6 11.5 L0.5 6 Z" fill="none" stroke="#00ff00" stroke-width="1.4"/></svg>';
    if (this.inventory.length === 0) {
      const li = document.createElement('li');
      li.className = 'empty';
      li.textContent = 'Inventory is empty.';
      this.invList.appendChild(li);
    } else {
      for (const item of this.inventory) {
        const li = document.createElement('li');
        li.innerHTML = svg;
        const span = document.createElement('span');
        span.textContent = item.name;
        li.appendChild(span);
        this.invList.appendChild(li);
      }
    }
  }
}
