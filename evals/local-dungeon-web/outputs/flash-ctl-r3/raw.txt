/* ============================ Item ============================ */
class Item {
  constructor(id, name, description, takeable = true) {
    this.id = id;
    this.name = name;
    this.description = description;
    this.takeable = takeable;
  }
  matches(target) {
    const t = target.toLowerCase().trim();
    const n = this.name.toLowerCase();
    return t === n || n.includes(t) || t.includes(n) || n.startsWith(t);
  }
}

/* ============================ Room ============================ */
class Room {
  constructor(id, name, description, lookDetail) {
    this.id = id;
    this.name = name;
    this.description = description;
    this.lookDetail = lookDetail;
    this.exits = {};        // direction -> roomId
    this.items = [];        // Item[] currently in room
    this.features = [];     // { name, description, onUse(engine) }
    this.locks = {};        // direction -> { item, message }
    this.onEnter = null;    // optional fn(engine)
  }
  findItem(target) {
    return this.items.find(i => i.matches(target)) || null;
  }
  findFeature(target) {
    return this.features.find(f => {
      const t = target.toLowerCase().trim();
      return f.name.toLowerCase().includes(t) || t.includes(f.name.toLowerCase());
    }) || null;
  }
}

      "A scroll in a hand older than the town charter. The visible text reads: 'To whoever finds this — the garden is real. Go that way. Turn back while you can.' It is signed with a small drawing of a snail wearing a hat."));
    R.secret = secret;

    const garden = new Room("garden", "Walled Garden",
      "You step through a low arch and stop breathing. A walled garden stretches beneath a sky full of stars — impossibly, undeniably, a hundred feet underground. A fountain runs clear and cold. Jasmine. A nightingale, rehearsing. The blue torchlight does not follow you past the arch.",
      "The dungeon ends here, at a mossy bench beside the fountain. There are no other exits. There do not need to be.");
    garden.exits = { };
    garden.onEnter = (engine) => {
      if (engine.flags.garden_ending) return;
      engine.flags.garden_ending = true;
      engine.print("· · ·  P E A C E  · · ·", "sys");
      engine.print("You have escaped The Local Dungeon. Most visitors only ever find the gift shop. Sit on the bench, if you like — the nightingale doesn't mind, and neither does management.", "sys");
      engine.print("(You've reached the peaceful ending. Feel free to keep wandering, or hit 'Reset Game' up top for another run.)", "dim");
    };
    R.garden = garden;

    this.rooms = R;
    // The rusted key lives in the Entrance Hall — keep a master copy so reset restores it.
    this.masterItems = {
      entrance: () => [new Item("rusted_key", "rusted key",
        "A heavy iron key, its teeth worn smooth by centuries of use. A brass tag reads: 'ARMORY — IF FOUND, RETURN TO MANAGEMENT. MANAGEMENT IS NOT FUNNY.'")],
      library: () => [],
      armory: () => [new Item("torch", "torch",
        "A pitch-heavy wooden torch. The label taped to it reads: 'EMERGENCY USE ONLY — DO NOT LIGHT IN THE LIBRARY (see incident report #47).' It is, somehow, still perfectly good torch.")],
      treasure: () => [new Item("golden_coin", "golden coin",
        "A golden coin, bright and warm, as if someone only just this moment decided you should have it. Stamped on one face: a doorway. On the other: a small, satisfied smile.")],
      secret: () => [new Item("ancient_scroll", "ancient scroll",
        "A scroll in a hand older than the town charter. The visible text reads: 'To whoever finds this — the garden is real. Go that way. Turn back while you can.' It is signed with a small drawing of a snail wearing a hat.")],
      garden: () => []
    };
  }

  /* ---------- State / Reset ---------- */
  reset() {
    this.currentRoomId = "entrance";
    this.inventory = [];
    this.visited = new Set();
    this.flags = { lever_pulled: false, garden_ending: false };
    this.moves = 0;
    this.itemsFound = 0;
    // Restore room items fresh
    for (const id in this.rooms) {
      this.rooms[id].items = this.masterItems[id]();
      if (id === "library") { delete this.rooms.library.exits.west; }
    }
    this.typeQueue = [];
    this.typing = false;
    this.skipRequested = false;
    this.outputEl.innerHTML = "";
    this.history = [];
    this.historyIndex = -1;
    this.updateUI();

    this.print("THE LOCAL DUNGEON — v1.0", "room-title");
    this.print("A text adventure beneath the Mire Street car park. Type 'help' for commands.", "sys");
    this.print("", "dim");
    this.enterRoom("entrance", true);
    this.inputEl.focus();
  }

  get room() { return this.rooms[this.currentRoomId]; }

  enterRoom(id, first = false) {
    this.currentRoomId = id;
    const room = this.rooms[id];
    const firstVisit = !this.visited.has(id);
    this.visited.add(id);

    if (!first) this.print("", "dim");
    this.print("— " + room.name + " —", "room-title");
    this.print(room.description);
    if (firstVisit && room.items.length) {
      this.print("You notice: " + room.items.map(i => i.name).join(", ") + ".", "sys");
    }
    if (room.onEnter) room.onEnter(this);
    this.updateUI();
  }

  /* ---------- UI ---------- */
  updateUI() {
    document.getElementById("movesCount").textContent = this.moves;
    document.getElementById("itemsCount").textContent = this.itemsFound;
    document.getElementById("locationName").textContent = this.room.name;
    const list = document.getElementById("inventoryList");
    list.innerHTML = "";
    if (this.inventory.length === 0) {
      const li = document.createElement("li");
      li.className = "empty-note";
      li.textContent = "Inventory is empty.";
      list.appendChild(li);
    } else {
      for (const item of this.inventory) {
        const li = document.createElement("li");
        li.innerHTML = '<svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="#ffb000" stroke-width="2" stroke-linecap="round"><circle cx="8" cy="15" r="5"/><path d="M11.5 11.5L20 3l2 2-2 2 2 2-3 3-2-2-1.5 1.5"/></svg>';
        li.appendChild(document.createTextNode(item.name));
        list.appendChild(li);
      }
    }
  }

  /* ---------- Typewriter output ---------- */
  print(text, cls = "") {
    this.typeQueue.push({ text, cls });
    if (!this.typing) this.processQueue();
  }

  processQueue() {
    if (this.typeQueue.length === 0) {
      this.typing = false;
      this.skipRequested = false;
      return;
    }
    this.typing = true;
    const { text, cls } = this.typeQueue.shift();
    const div = document.createElement("div");
    div.className = "line" + (cls ? " " + cls : "");
    this.outputEl.appendChild(div);

    if (this.skipRequested || text.length === 0) {
      div.textContent = text;
      this.scrollToBottom();
      this.processQueue();
      return;
    }

    let i = 0;
    const tick = () => {
      if (this.skipRequested) {
        div.textContent = text;
        this.scrollToBottom();
        this.processQueue();
        return;
      }
      i++;
      div.textContent = text.slice(0, i);
      this.scrollToBottom();
      if (i < text.length) setTimeout(tick, 12);
      else this.processQueue();
    };
    setTimeout(tick, 0);
  }

  skipTyping() { if (this.typing) this.skipRequested = true; }
  scrollToBottom() { this.outputEl.scrollTop = this.outputEl.scrollHeight; }

  /* ---------- Command parsing ---------- */
  static DIRS = {
    n: "north", north: "north", northerly: "north",
    s: "south", south: "south", southerly: "south",
    e: "east", east: "east", easterly: "east",
    w: "west", west: "west", westerly: "west"
  };

  handle(rawInput) {
    const raw = rawInput.trim();
    if (!raw) return;
    this.history.push(raw);
    this.historyIndex = this.history.length;
    this.print("> " + raw, "echo");

    let input = raw.toLowerCase();

    // Multi-word prefixes first
    const multiPrefixes = [
      ["pick up ", "take "], ["put down ", "drop "], ["look at ", "examine "],
      ["look around ", "examine "], ["head ", "go "]
    ];
    for (const [from, to] of multiPrefixes) {
      if (input.startsWith(from)) { input = to + input.slice(from.length); break; }
    }

    const tokens = input.split(/\s+/);
    const verb = tokens[0];
    const rest = tokens.slice(1).join(" ").replace(/\b(the|a|an|up|down)\b/g, " ").replace(/\s+/g, " ").trim();

    if (GameEngine.DIRS[verb]) { this.go(GameEngine.DIRS[verb]); return; }

    switch (true) {
      case ["go","move","walk","travel","head"].includes(verb): {
        const dir = GameEngine.DIRS[rest.split(" ")[0] || ""];
        if (!dir) { this.print("Go where? Try north, south, east, or west (n/s/e/w also work).", "err"); return; }
        this.go(dir);
        return;
      }
      case ["l","look"].includes(verb): {
        if (!rest || rest === "around") {
          this.print(this.room.description);
          this.print(this.room.lookDetail);
          const exits = this.availableExits();
          this.print("Exits: " + (exits.length ? exits.join(", ") : "none."), "sys");
          if (this.room.items.length)
            this.print("You see here: " + this.room.items.map(i => i.name).join(", ") + ".", "sys");
          return;
        }
        this.examine(rest); return;
      }
      case ["x","examine","inspect","read","check","describe"].includes(verb): {
        if (!rest) { this.print("Examine what? Name a thing, e.g. 'examine lever'.", "err"); return; }
        this.examine(rest); return;
      }
      case ["take","get","grab","steal","collect"].includes(verb): { this.take(rest); return; }
      case ["drop","put","leave","discard"].includes(verb): { this.drop(rest); return; }
      case ["use","pull","push","activate","press","turn","open"].includes(verb): { this.use(rest); return; }
      case ["i","inv","inventory","items","bag","carrying"].includes(verb): {
        if (this.inventory.length === 0) this.print("Your pockets are empty. Your hands are also, technically, inventory.", "sys");
        else this.print("You are carrying: " + this.inventory.map(i => i.name).join(", ") + ".", "sys");
        return;
      }
      case ["h","help","commands","?"].includes(verb): {
        this.print("COMMANDS: look · examine <thing> · go <direction> (n/s/e/w) · take <item> · drop <item> · use <thing> · inventory · reset", "sys");
        this.print("Tip: you can abbreviate — 'x lever', 'n', 'inv', 'grab torch'.", "dim");
        return;
      }
      case ["stats","score"].includes(verb): {
        this.print(`Moves: ${this.moves} · Items found: ${this.itemsFound} · Rooms visited: ${this.visited.size} of ${Object.keys(this.rooms).length}`, "sys");
        return;
      }
      case ["reset"].includes(verb): {
        this.print("Restarting your run…", "sys");
        setTimeout(() => this.reset(), 300);
        return;
      }
      default: {
        this.print(`Unknown command: "${tokens.join(" ")}". Type 'help' to see what the dungeon understands.`, "err");
      }
    }
  }

  availableExits() {
    return Object.keys(this.room.exits);
  }

  /* ---------- Actions ---------- */
  go(dir) {
    const room = this.room;
    const destId = room.exits[dir];
    if (!destId) {
      if (dir === "west" && room.id === "library" && !this.flags.lever_pulled) {
        this.print("West of here is nothing but an enormous, extremely shut bookcase. Perhaps something around here opens it.", "err");
      } else {
        this.print(`You cannot go ${dir} from here. Exits: ${this.availableExits().join(", ") || "none"}.`, "err");
      }
      return;
    }
    const lock = room.locks[dir];
    if (lock) {
      const hasKey = this.inventory.some(i => i.id === lock.item);
      if (!hasKey) { this.print(lock.blocked, "err"); return; }
      this.print(lock.unlocked, "sys");
      delete room.locks[dir];
    }
    this.moves++;
    this.enterRoom(destId);
  }

  take(target) {
    if (!target) { this.print("Take what? Try 'take key'.", "err"); return; }
    const item = this.room.findItem(target);
    if (!item) { this.print(`You don't see any "${target}" here in the ${this.room.name}.`, "err"); return; }
    if (!item.takeable) { this.print(`The ${item.name} is not going anywhere. Neither, frankly, are you, emotionally.`, "err"); return; }
    this.room.items = this.room.items.filter(i => i !== item);
    this.inventory.push(item);
    this.itemsFound++;
    this.print(`Taken: ${item.name}. It settles into your hands like it was waiting for them.`, "sys");
    this.print(item.description, "dim");
    this.updateUI();
  }

  drop(target) {
    if (!target) { this.print("Drop what? Try 'drop torch'.", "err"); return; }
    const item = this.inventory.find(i => i.matches(target));
    if (!item) { this.print(`You aren't carrying any "${target}".`, "err"); return; }
    this.inventory = this.inventory.filter(i => i !== item);
    this.room.items.push(item);
    this.print(`You set the ${item.name} down here, in the ${this.room.name}. Bold.`, "sys");
    this.updateUI();
  }

  use(target) {
    if (!target) { this.print("Use what? Try 'use lever'.", "err"); return; }
    const feature = this.room.findFeature(target);
    if (feature && feature.onUse) { feature.onUse(this); this.updateUI(); return; }
    if (feature) { this.print(`Using the ${feature.name} produces nothing but a faint sense of impropriety.`, "err"); return; }
    const held = this.inventory.find(i => i.matches(target));
    if (held) {
      if (held.id === "rusted_key") this.print("You hold the rusted key up dramatically. No doors nearby feel addressed. (The Armory is west, then north door territory.)", "dim");
      else if (held.id === "torch") this.print("You consider lighting the torch. Every sign in this building, in increasingly red ink, suggests not here.", "dim");
      else if (held.id === "golden_coin") this.print("The golden coin warms in your palm, quietly pleased with itself. It seems to point, faintly, south.", "dim");
      else this.print(`You fiddle with the ${held.name}. It remains, impressively, exactly itself.`, "dim");
      return;
    }
    this.print(`There is no "${target}" here to use.`, "err");
  }

  examine(target) {
    const t = target.toLowerCase();
    if (t.includes("exit") || t.includes("door") && !this.room.findItem(t)) {
      const exits = this.availableExits();
      this.print("Exits: " + (exits.length ? exits.join(", ") : "none — you have found the edge of things.") + ".", "sys");
      return;
    }
    const feature = this.room.findFeature(target);
    if (feature) { this.print(feature.description, "dim"); return; }
    const inRoom = this.room.findItem(target);
    if (inRoom) { this.print(inRoom.description, "dim"); return; }
    const held = this.inventory.find(i => i.matches(target));
    if (held) { this.print(held.description, "dim"); return; }
    if (t.includes("lever") && this.room.id !== "library") {
      this.print("No lever here. The interesting one is in the Library, north of the Entrance Hall.", "dim"); return;
    }
    this.print(`You squint intently at "${target}", but it declines to reveal anything. It isn't here, in any case.`, "err");
  }
}

/* ============================ Boot ============================ */
(function boot() {
  const output = document.getElementById("output");
  const input = document.getElementById("cmdInput");
  const engine = new GameEngine(output, input);

  input.addEventListener("keydown", (e) => {
    if (e.key === "Enter") {
      if (engine.typing) { engine.skipTyping(); }
      const val = input.value;
      input.value = "";
      engine.handle(val);
    } else if (e.key === "ArrowUp") {
      e.preventDefault();
      if (engine.history.length === 0) return;
      engine.historyIndex = Math.max(0, engine.historyIndex - 1);
      input.value = engine.history[engine.historyIndex] || "";
    } else if (e.key === "ArrowDown") {
      e.preventDefault();
      if (engine.history.length === 0) return;
      engine.historyIndex = Math.min(engine.history.length, engine.historyIndex + 1);
      input.value = engine.history[engine.historyIndex] || "";
    }
  });

  document.getElementById("skipBtn").addEventListener("click", (e) => {
    engine.skipTyping();
    e.stopPropagation();
    input.focus();
  });

  // Clicking the console focuses input (and skips typing if mid-print)
  document.getElementById("consoleArea").addEventListener("click", () => {
    if (engine.typing) engine.skipTyping();
    input.focus();
  });

  document.getElementById("resetBtn").addEventListener("click", () => {
    engine.reset();
  });

  window.addEventListener("load", () => input.focus());
  input.focus();
})();
