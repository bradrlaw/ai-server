LocalMind (Local LLM App).
Single HTML file, zero external dependencies (no CDNs, no fonts, no images). All inline.
Dark mode (#07070a–#12121a), Purple (#7c5cfc), Teal (#00d4aa), White/Muted text. Linear/Vercel style. Responsive.

        *   Sticky Nav (with hamburger, IntersectionObserver active link).
        *   Hero (Canvas particle field/gradient, big headline, CTAs).
        *   Features Grid (2x2, hover glow, lift).
        *   How It Works (3-step flow, connecting lines, responsive).
        *   Pricing (3 tiers, Pro highlighted, checkmarks).
        *   Interactive Demo (Terminal window, fake chat, typing indicator).
        *   Testimonials (3 cards, gradient initials).
        *   FAQ (Accordion, smooth transition, exclusive open).
        *   CTA Banner (Email input, success state).
        *   Footer (Dynamic year).

        *   Smooth scroll.
        *   IntersectionObserver fade-up animations.
        *   Button scaling/shadows.
        *   Link underlines.
        *   Hamburger morphing (3 lines -> X).
        *   Nav blur on scroll.
        *   Typography (system stack, clamp).

    *   *Colors:*
        *   BG: `#07070a`
        *   Surface: `#12121a`
        *   Primary: `#7c5cfc`
        *   Secondary: `#00d4aa`
        *   Text: `#fafafa`
        *   Muted: `#71717a`

    *   *Layout Strategy:*
        *   CSS Grid for features and pricing.
        *   Flexbox for nav and footer.
        *   Relative/Absolute positioning for the "How It Works" lines.
        *   `backdrop-filter: blur(12px)` for the nav.

    *   *Canvas implementation:* A simple particle system. Points moving randomly, connecting with lines if close enough. This creates that "AI/Neural" feel.

    *   *Animations:*
        *   `@keyframes fadeInUp` for the scroll reveal.
        *   `transition` for the accordion `max-height`.
        *   `transition` for the hamburger lines.

    *   *HTML:* Basic boilerplate.
    *   *CSS:*

        *   Specific styles for each section.
        *   Mobile-first media queries.

        *   Canvas animation loop.
        *   IntersectionObserver for active links and fade-ins.
        *   Hamburger menu toggle.
        *   FAQ logic.
        *   CTA success state.
        *   Smooth scrolling.

    *   *Hamburger Morph:* Three spans. Top rotates 45deg, mid fades, bottom -45deg.

        // --- Hero Canvas Animation ---
        const canvas = document.getElementById('hero-canvas');
        const ctx = canvas.getContext('2d');
        let particles = [];
        let mouse = { x: null, y: null };

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        window.addEventListener('mousemove', (e) => {
            mouse.x = e.clientX;
            mouse.y = e.clientY;
        });

        class Particle {
            constructor() {
                this.x = Math.random() * canvas.width;
                this.y = Math.random() * canvas.height;
                this.size = Math.random() * 2 + 1;
                this.speedX = Math.random() * 0.5 - 0.25;
                this.speedY = Math.random() * 0.5 - 0.25;
            }
            update() {
                this.x += this.speedX;
                this.y += this.speedY;

                if (this.x > canvas.width) this.x = 0;
                if (this.x < 0) this.x = canvas.width;
                if (this.y > canvas.height) this.y = 0;
                if (this.y < 0) this.y = canvas.height;

                // Subtle mouse attraction
                if (mouse.x) {
                    let dx = mouse.x - this.x;
                    let dy = mouse.y - this.y;
                    let dist = Math.sqrt(dx*dx + dy*dy);
                    if (dist < 200) {
                        this.x += dx * 0.001;
                        this.y += dy * 0.001;
                    }
                }
            }
            draw() {
                ctx.fillStyle = 'rgba(124, 92, 252, 0.5)';
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        function initParticles() {
            particles = [];
            for (let i = 0; i < 80; i++) {
                particles.push(new Particle());
            }
        }
        initParticles();

        function animate() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            particles.forEach((p, i) => {
                p.update();
                p.draw();
                
                for (let j = i + 1; j < particles.length; j++) {
                    let p2 = particles[j];
                    let dx = p.x - p2.x;
                    let dy = p.y - p2.y;
                    let dist = Math.sqrt(dx*dx + dy*dy);
                    if (dist < 150) {
                        ctx.strokeStyle = `rgba(124, 92, 252, ${0.2 * (1 - dist/150)})`;
                        ctx.lineWidth = 1;
                        ctx.beginPath();
                        ctx.moveTo(p.x, p.y);
                        ctx.lineTo(p2.x, p2.y);
                        ctx.stroke();
                    }
                }
            });
            requestAnimationFrame(animate);
        }
        animate();

        // --- Navigation & Scroll ---
        const nav = document.getElementById('main-nav');
        const hamburger = document.getElementById('hamburger');
        const overlay = document.getElementById('mobile-overlay');
        const navItems = document.querySelectorAll('.nav-item');

        window.addEventListener('scroll', () => {
            if (window.scrollY > 50) {
                nav.classList.add('scrolled');
            } else {
                nav.classList.remove('scrolled');
            }
        });

        hamburger.addEventListener('click', () => {
            hamburger.classList.toggle('open');
            overlay.classList.toggle('open');
        });

        navItems.forEach(item => {
            item.addEventListener('click', () => {
                hamburger.classList.remove('open');
                overlay.classList.remove('open');
            });
        });

        // Intersection Observer for Active Links & Fade-ins
        const observerOptions = { threshold: 0.2 };
        const sectionObserver = new IntersectionObserver((entries) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    entry.target.classList.add('visible');
                    
                    // Update active nav link
                    const id = entry.target.getAttribute('id');
                    document.querySelectorAll('.nav-links a').forEach(link => {
                        link.classList.remove('active');
                        if (link.getAttribute('href') === `#${id}`) {
                            link.classList.add('active');
                        }
                    });
                }
            });
        }, observerOptions);

        document.querySelectorAll('section').forEach(section => {
            sectionObserver.observe(section);
        });

        // --- FAQ Accordion ---
        const faqQuestions = document.querySelectorAll('.faq-question');
        faqQuestions.forEach(q => {
            q.addEventListener('click', () => {
                const item = q.parentElement;
                const isOpen = item.classList.contains('open');
                
                // Close all other items
                document.querySelectorAll('.faq-item').forEach(i => i.classList.remove('open'));
                
                if (!isOpen) {
                    item.classList.add('open');
                }
            });
        });
