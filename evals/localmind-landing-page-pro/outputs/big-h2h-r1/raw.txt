(function () {
  'use strict';

  /* ---------- Year ---------- */
  document.getElementById('year').textContent = new Date().getFullYear();

  /* ---------- Nav scroll state ---------- */
  var nav = document.getElementById('nav');
  function onScroll() {
    if (window.scrollY > 50) nav.classList.add('scrolled');
    else nav.classList.remove('scrolled');
  }
  window.addEventListener('scroll', onScroll, { passive: true });
  onScroll();

  /* ---------- Mobile menu ---------- */
  var hamburger = document.getElementById('hamburger');
  var mobileMenu = document.getElementById('mobileMenu');
  function setMenu(open) {
    hamburger.classList.toggle('open', open);
    mobileMenu.classList.toggle('open', open);
    hamburger.setAttribute('aria-expanded', String(open));
    document.body.style.overflow = open ? 'hidden' : '';
  }
  hamburger.addEventListener('click', function () {
    setMenu(!mobileMenu.classList.contains('open'));
  });
  mobileMenu.querySelectorAll('a').forEach(function (a) {
    a.addEventListener('click', function () { setMenu(false); });
  });
  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && mobileMenu.classList.contains('open')) setMenu(false);
  });

  /* ---------- Smooth scroll (JS fallback + offset) ---------- */
  function smoothScrollTo(target) {
    if (!target) return;
    var top = target.getBoundingClientRect().top + window.scrollY - 70;
    if ('scrollBehavior' in document.documentElement.style) {
      window.scrollTo({ top: top, behavior: 'smooth' });
    } else {
      var start = window.scrollY, dist = top - start, dur = 500, t0 = null;
      function step(t) {
        if (t0 === null) t0 = t;
        var p = Math.min((t - t0) / dur, 1);
        var ease = p < 0.5 ? 2 * p * p : -1 + (4 - 2 * p) * p;
        window.scrollTo(0, start + dist * ease);
        if (p < 1) requestAnimationFrame(step);
      }
      requestAnimationFrame(step);
    }
  }
  document.querySelectorAll('a[href^="#"]').forEach(function (a) {
    a.addEventListener('click', function (e) {
      var id = a.getAttribute('href');
      if (id.length > 1) {
        e.preventDefault();
        smoothScrollTo(document.querySelector(id));
      }
    });
  });

  /* ---------- Active nav link via IntersectionObserver ---------- */
  var navLinks = document.querySelectorAll('.nav-links a.nav-link');
  var sections = [];
  navLinks.forEach(function (link) {
    var sec = document.querySelector(link.getAttribute('href'));
    if (sec) sections.push({ link: link, sec: sec });
  });
  var activeObs = new IntersectionObserver(function (entries) {
    entries.forEach(function (entry) {
      if (entry.isIntersecting) {
        navLinks.forEach(function (l) { l.classList.remove('active'); });
        sections.forEach(function (s) {
          if (s.sec === entry.target) s.link.classList.add('active');
        });
      }
    });
  }, { rootMargin: '-40% 0px -55% 0px' });
  sections.forEach(function (s) { activeObs.observe(s.sec); });

  var groups = [];

    var g = groups.find(function (x) { return x.sec === sec; });
    if (!g) { g = { sec: sec, els: [] }; groups.push(g); }
    g.els.push(el);

  var fadeObs = new IntersectionObserver(function (entries) {

  var canvas = document.getElementById('heroCanvas');
  var ctx = canvas.getContext('2d');
  var W = 0, H = 0, DPR = Math.min(window.devicePixelRatio || 1, 2);
  var particles = [];
  var mouse = { x: 0.5, y: 0.5, tx: 0.5, ty: 0.5 };
  var COUNT = 70;

  function resize() {
    var rect = canvas.parentElement.getBoundingClientRect();
    W = rect.width; H = rect.height;
    canvas.width = W * DPR;

  function initParticles() {
    particles = [];
    for (var i = 0; i < COUNT; i++) {
      particles.push({
        x: Math.random() * W,
        y: Math.random() * H,
        vx: (Math.random() - 0.5) * 0.25,
        vy: (Math.random() - 0.5) * 0.25,

      });
    }
  }

    // ease mouse toward target (subtle clamp)
    mouse.x += (mouse.tx - mouse.x) * 0.04;
    mouse.y += (mouse.ty - mouse.y) * 0.04;

    ctx.clearRect(0, 0, W, H);

    // soft gradient orbs
    var gx = W * 0.5 + (mouse.x - 0.5) * 90;
    var gy = H * 0.35 + (mouse.y - 0.5) * 60;

    g1.addColorStop(0, 'rgba(124,92,252,0.14)');
    g1.addColorStop(1, 'rgba(124,92,252,0)');
    ctx.fillStyle = g1;
    ctx.fillRect(0, 0, W, H);

    var gy2 = H * 0.8 - (mouse.y - 0.5) * 50;
    var g2 = ctx.createRadialGradient(gx2, gy2, 0, gx2, gy2, Math.max(W, H) * 0.5);
    g2.addColorStop(0, 'rgba(0,212,170,0.08)');
    g2.addColorStop(1, 'rgba(0,212,170,0)');
    ctx.fillStyle = g2;
    ctx.fillRect(0, 0, W, H);

    // particles
    var mx = (mouse.x - 0.5) * 26; // subtle parallax
    var my = (mouse.y - 0.5) * 18;

    for (var i = 0; i < particles.length; i++) {
      var p = particles[i];
      p.x += p.vx; p.y += p.vy;
      if (p.x < -10) p.x = W + 10; if (p.x > W + 10) p.x = -10;
      if (p.y < -10) p.y = H + 10; if (p.y > H + 10) p.y = -10;

      var px = p.x + mx, py = p.y + my;
      ctx.beginPath();
      ctx.arc(px, py, p.r, 0, Math.PI * 2);
      ctx.fillStyle = 'rgba(' + p.hue + ',' + p.a + ')';
      ctx.fill();
    }

    // connect nearby particles
    var LINK = W < 640 ? 110 : 140;
    for (var a = 0; a < particles.length; a++) {
      for (var b = a + 1; b < particles.length; b++) {
        var pa = particles[a], pb = particles[b];
        var dx = (pa.x + mx) - (pb.x + mx);
        var dy = (pa.y + my) - (pb.y + my);
        var d2 = dx * dx + dy * dy;
        if (d2 < LINK * LINK) {
          var alpha = (1 - Math.sqrt(d2) / LINK) * 0.14;
          ctx.beginPath();
          ctx.moveTo(pa.x + mx, pa.y + my);
          ctx.lineTo(pb.x + mx, pb.y + my);
          ctx.strokeStyle = 'rgba(124,92,252,' + alpha.toFixed(3) + ')';
          ctx.lineWidth = 1;
          ctx.stroke();
        }
      }
    }
    requestAnimationFrame(draw);
  }

  canvas.parentElement.addEventListener('mousemove', function (e) {
    var rect = canvas.getBoundingClientRect();
    mouse.tx = (e.clientX - rect.left) / rect.width;
    mouse.ty = (e.clientY - rect.top) / rect.height;
  });
  canvas.parentElement.addEventListener('mouseleave', function () {
    mouse.tx = 0.5; mouse.ty = 0.5;
  });
  // touch support (subtle)
  canvas.parentElement.addEventListener('touchmove', function (e) {
    if (e.touches.length) {
      var rect = canvas.getBoundingClientRect();
      mouse.tx = (e.touches[0].clientX - rect.left) / rect.width;
      mouse.ty = (e.touches[0].clientY - rect.top) / rect.height;
    }
  }, { passive: true });

  window.addEventListener('resize', resize);
  resize();
  requestAnimationFrame(draw);
})();
