<script>
(function () {
  "use strict";

  /* ---------- Footer year ---------- */
  document.getElementById("year").textContent = new Date().getFullYear();

  /* ---------- Nav scroll state ---------- */
  var nav = document.getElementById("nav");
  function onScroll() {
    nav.classList.toggle("scrolled", window.scrollY > 50);
  }
  window.addEventListener("scroll", onScroll, { passive: true });
  onScroll();

  /* ---------- Hamburger / mobile menu ---------- */
  var hamburger = document.getElementById("hamburger");
  var mobileMenu = document.getElementById("mobile-menu");
  function setMenu(open) {
    hamburger.classList.toggle("open", open);
    mobileMenu.classList.toggle("open", open);
    hamburger.setAttribute("aria-expanded", String(open));
    mobileMenu.setAttribute("aria-hidden", String(!open));
    document.body.style.overflow = open ? "hidden" : "";
  }
  hamburger.addEventListener("click", function () {
    setMenu(!mobileMenu.classList.contains("open"));
  });
  mobileMenu.querySelectorAll("a").forEach(function (a) {
    a.addEventListener("click", function () { setMenu(false); });
  });

  /* ---------- Smooth scroll (JS fallback + offset for fixed nav) ---------- */
  document.querySelectorAll('a[href^="#"]').forEach(function (link) {
    link.addEventListener("click", function (e) {
      var id = link.getAttribute("href");
      if (id.length < 2) return;
      var target = document.querySelector(id);
      if (!target) return;
      e.preventDefault();
      var top = target.getBoundingClientRect().top + window.scrollY - 60;
      if ("scrollBehavior" in document.documentElement.style) {
        window.scrollTo({ top: top, behavior: "smooth" });
      } else {
        window.scrollTo(0, top);
      }
    });
  });

  /* ---------- Scroll-triggered reveals (staggered 100ms) ---------- */
  var revealEls = Array.prototype.slice.call(document.querySelectorAll(".reveal"));
  if ("IntersectionObserver" in window) {
    var revealObserver = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        if (entry.isIntersecting) {
          var el = entry.target;
          // Stagger siblings within the same section
          var parent = el.closest("section") || document.body;
          var siblings = Array.prototype.slice.call(parent.querySelectorAll(".reveal"));
          var idx = siblings.indexOf(el);
          var delay = Math.max(0, idx) * 100;
          el.style.transitionDelay = delay + "ms";
          el.classList.add("visible");
          revealObserver.unobserve(el);
        }
      });
    }, { threshold: 0.12, rootMargin: "0px 0px -40px 0px" });
    revealEls.forEach(function (el) { revealObserver.observe(el); });
  } else {
    revealEls.forEach(function (el) { el.classList.add("visible"); });
  }

  /* ---------- Active nav link via IntersectionObserver ---------- */
  var navLinks = document.querySelectorAll('.nav-links a[data-section]');
  var sections = ["features", "how-it-works", "pricing", "faq"]
    .map(function (id) { return document.getElementById(id); })
    .filter(Boolean);
  if ("IntersectionObserver" in window && sections.length) {
    var activeMap = {};
    var navObserver = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        activeMap[entry.target.id] = entry.isIntersecting;
      });
      // Pick the topmost intersecting section
      var current = null;
      for (var i = 0; i < sections.length; i++) {
        if (activeMap[sections[i].id]) { current = sections[i].id; break; }
      }
      navLinks.forEach(function (l) {
        l.classList.toggle("active", l.getAttribute("data-section") === current);
      });
    }, { rootMargin: "-30% 0px -55% 0px" });
    sections.forEach(function (s) { navObserver.observe(s); });
  }

  /* ---------- FAQ accordion (one open at a time) ---------- */
  var faqItems = Array.prototype.slice.call(document.querySelectorAll(".faq-item"));
  faqItems.forEach(function (item) {
    var btn = item.querySelector(".faq-q");
    var panel = item.querySelector(".faq-a");
    btn.addEventListener("click", function () {
      var isOpen = item.classList.contains("open");
      faqItems.forEach(function (other) {
        other.classList.remove("open");
        other.querySelector(".faq-a").style.maxHeight = "0px";
        other.querySelector(".faq-q").setAttribute("aria-expanded", "false");
      });
      if (!isOpen) {
        item.classList.add("open");
        panel.style.maxHeight = panel.scrollHeight + "px";
        btn.setAttribute("aria-expanded", "true");
      }
    });
  });
  window.addEventListener("resize", function () {
    var open = document.querySelector(".faq-item.open .faq-a");
    if (open) open.style.maxHeight = open.scrollHeight + "px";
  });

  /* ---------- Waitlist form ---------- */
  var form = document.getElementById("waitlist-form");
  var emailInput = document.getElementById("waitlist-email");
  var waitBtn = document.getElementById("waitlist-btn");
  var ctaNote = document.getElementById("cta-note");
  form.addEventListener("submit", function (e) {
    e.preventDefault();
    var val = emailInput.value.trim();
    if (!val || val.indexOf("@") < 1 || val.indexOf(".") === -1) {
      ctaNote.textContent = "That doesn't look like an email — mind checking?";
      ctaNote.style.color = "#ff8a8a";
      emailInput.focus();
      return;
    }
    waitBtn.textContent = "You're in! 🎉";
    waitBtn.disabled = true;
    waitBtn.style.background = "linear-gradient(92deg, #00d4aa, #2be0bb)";
    waitBtn.style.boxShadow = "0 8px 28px rgba(0,212,170,0.4)";
    emailInput.value = "";
    ctaNote.textContent = "You're on the list. Watch your inbox for an invite within the week.";
    ctaNote.style.color = "#00d4aa";
    setTimeout(function () {
      waitBtn.textContent = "Join the Waitlist";
      waitBtn.disabled = false;
      waitBtn.style.background = "";
      waitBtn.style.boxShadow = "";
      ctaNote.textContent = "Beta runs on macOS 13+, Windows 11, and Ubuntu 22.04+.";
      ctaNote.style.color = "";
    }, 4000);
  });

  /* ---------- Pricing buttons feedback ---------- */
  document.querySelectorAll("[data-plan]").forEach(function (btn) {
    btn.addEventListener("click", function () {
      var original = btn.textContent;
      btn.textContent = btn.getAttribute("data-plan") === "Studio" ? "We'll be in touch ✓" : "Added — check out ✓";
      setTimeout(function () { btn.textContent = original; }, 2200);
    });
  });

  /* ---------- Hero particle field (canvas, mouse-reactive) ---------- */
  var canvas = document.getElementById("particle-canvas");
  var hero = document.getElementById("hero");
  if (canvas && hero && canvas.getContext) {
    var ctx = canvas.getContext("2d");
    var particles = [];
    var W = 0, H = 0, dpr = Math.min(window.devicePixelRatio || 1, 2);
    var mouse = { x: 0.5, y: 0.5, tx: 0.5, ty: 0.5 };
    var reduced = window.matchMedia && window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    var running = true;

    function resize() {
      W = hero.offsetWidth;
      H = hero.offsetHeight;
      canvas.width = W * dpr;
      canvas.height = H * dpr;
      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
      buildParticles();
    }

    function buildParticles() {
      var count = Math.max(36, Math.min(110, Math.floor((W * H) / 14000)));
      particles = [];
      for (var i = 0; i < count; i++) {
        particles.push({
          x: Math.random() * W,
          y: Math.random() * H,
          vx: (Math.random() - 0.5) * 0.22,
          vy: (Math.random() - 0.5) * 0.22,
          r: Math.random() * 1.6 + 0.6,
          hue: Math.random() < 0.5 ? "124,92,252" : "0,212,170",
          base: Math.random() * 0.35 + 0.15
        });
      }
    }

    function step() {
      if (!running) return;
      ctx.clearRect(0, 0, W, H);

      // Ease mouse
      mouse.x += (mouse.tx - mouse.x) * 0.04;
      mouse.y += (mouse.ty - mouse.y) * 0.04;

      // Subtle parallax offset from mouse (clamped ±14px)
      var px = (mouse.x - 0.5) * 14;
      var py = (mouse.y - 0.5) * 14;

      // Soft glow orbs
      var g1 = ctx.createRadialGradient(W * (0.3 + (mouse.x - 0.5) * 0.06), H * (0.35 + (mouse.y - 0.5) * 0.06), 0, W * 0.3, H * 0.35, Math.max(W, H) * 0.45);
      g1.addColorStop(0, "rgba(124,92,252,0.10)");
      g1.addColorStop(1, "rgba(124,92,252,0)");
      ctx.fillStyle = g1;
      ctx.fillRect(0, 0, W, H);

      var g2 = ctx.createRadialGradient(W * (0.72 + (mouse.x - 0.5) * 0.05), H * (0.6 + (mouse.y - 0.5) * 0.05), 0, W * 0.72, H * 0.6, Math.max(W, H) * 0.4);
      g2.addColorStop(0, "rgba(0,212,170,0.07)");
      g2.addColorStop(1, "rgba(0,212,170,0)");
      ctx.fillStyle = g2;
      ctx.fillRect(0, 0, W, H);

      // Update + draw particles
      for (var i = 0; i < particles.length; i++) {
        var p = particles[i];
        p.x += p.vx;
        p.y += p.vy;
        if (p.x < -20) p.x = W + 20; else if (p.x > W + 20) p.x = -20;
        if (p.y < -20) p.y = H + 20; else if (p.y > H + 20) p.y = -20;

        ctx.beginPath();
        ctx.arc(p.x + px, p.y + py, p.r, 0, Math.PI * 2);
        ctx.fillStyle = "rgba(" + p.hue + "," + p.base + ")";
        ctx.fill();
      }

      // Connecting lines
      ctx.lineWidth = 1;
      for (var a = 0; a < particles.length; a++) {
        for (var b = a + 1; b < particles.length; b++) {
          var pa = particles[a], pb = particles[b];
          var dx = (pa.x + px) - (pb.x + px);
          var dy = (pa.y + py) - (pb.y + py);
          var d2 = dx * dx + dy * dy;
          if (d2 < 12100) { // 110px
            var alpha = (1 - d2 / 12100) * 0.12;
            ctx.strokeStyle = "rgba(124,92,252," + alpha + ")";
            ctx.beginPath();
            ctx.moveTo(pa.x + px, pa.y + py);
            ctx.lineTo(pb.x + px, pb.y + py);
            ctx.stroke();
          }
        }
      }

      requestAnimationFrame(step);
    }

    hero.addEventListener("mousemove", function (e) {
      var rect = hero.getBoundingClientRect();
      mouse.tx = Math.min(1, Math.max(0, (e.clientX - rect.left) / rect.width));
      mouse.ty = Math.min(1, Math.max(0, (e.clientY - rect.top) / rect.height));
    });
    hero.addEventListener("mouseleave", function () {
      mouse.tx = 0.5; mouse.ty = 0.5;
    });

    window.addEventListener("resize", resize);

    // Pause when hero off-screen or tab hidden
    document.addEventListener("visibilitychange", function () {
      if (document.hidden) {
        running = false;
      } else if (!reduced && !running) {
        running = true;
        requestAnimationFrame(step);
      }
    });

    resize();
    if (reduced) {
      // Draw one static frame
      running = true;
      ctx.clearRect(0, 0, W, H);
      particles.forEach(function (p) {
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
        ctx.fillStyle = "rgba(" + p.hue + "," + p.base + ")";
        ctx.fill();
      });
      running = false;
    } else {
      requestAnimationFrame(step);
    }
  }
})();
</script>
